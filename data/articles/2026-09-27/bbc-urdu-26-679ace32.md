# 26 سالہ شخص کے جسم سے بچہ دانی برآمد، بانجھ پن کے ٹیسٹ کے دوران نایاب حالت کا انکشاف

26 سالہ شخص کے جسم سے بچہ دانی برآمد، بانجھ پن کے ٹیسٹ کے دوران نایاب حالت کا انکشاف

دہلی میں ایک 26 سالہ شخص نے اپنی شادی کے دو سال بعد اولاد کے نہ ہونے کی وجہ سے ڈاکٹر سے رجوع کیا۔ معائنے کے دوران ڈاکٹروں کو معلوم ہوا کہ اس کے جسم کے اندر رحم یعنے بچہ دانی اور فیلوپین ٹیوبز یعنی بچہ دانی کی نالیاں بھی موجود ہیں۔
تاہم علاج کرنے والے ڈاکٹر کے مطابق باہر سے اس شخص کا جسم عام مردوں جیسا تھا اور اس میں مردانہ جسمانی خصوصیات بھی موجود تھیں۔ اس کے علاوہ ایسی کوئی واضح علامت نہیں تھی جس سے اس طبی حالت کا اندازہ ہو سکتا۔
دہلی کے ایک نجی ہسپتال کے چیف یورولوجسٹ ڈاکٹر سشیل کھر بندہ، جنھوں نے اس مریض کا علاج کیا، کے مطابق اولاد نہ ہونے کی وجہ جاننے کے لیے کیے گئے ٹیسٹوں میں رحم اور فیلوپین ٹیوبز کی موجودگی کا انکشاف ہوا۔
ڈاکٹروں کے مطابق یہ ایک انتہائی نایاب پیدائشی حالت ہے جسے ’پرسیسٹنٹ مولیرین ڈکٹ سنڈروم‘ کہا جاتا ہے۔ اب تک دنیا بھر میں اس کے 300 سے بھی کم کیسز سامنے آئے ہیں۔
’نیچر ریویوز یورولوجی‘ میں شائع ایک رپورٹ کے مطابق ’پرسیسٹنٹ مولیرین ڈکٹ سنڈروم‘ کی صورت میں کسی شخص کے کروموسوم ’46ایکس وائے‘ ہوتے ہیں اور جسم عام طور پر مرد کی طرح نشوونما پاتا ہے، لیکن مولیرین ڈکٹ سے بننے والے اعضا، جیسے رحم اور فیلوپین ٹیوبز جسم میں موجود رہتے ہیں۔
جنوری سنہ 2026 میں ’گلوبل جرنل آف سرجری اینڈ کیس رپورٹس‘ میں شائع ایک رپورٹ کے مطابق ایک 48 سالہ شخص کے بار بار پیشاب اور منی میں خون آنے کی وجہ جاننے کے لیے کیے گئے سی ٹی سکین میں رحم جیسی ساخت موجود ہونے کا پتا چلا۔
سرجری کے دوران پروسٹیٹ سے جڑی دو حصوں پر مشتمل ایک ساخت نکالی گئی، جس کے ٹشوز کے طبی معائنے سے تصدیق ہوئی کہ وہ رحم کے ٹشوز تھے۔ اس شخص کی جسمانی نشوونما عام مردوں جیسی تھی۔
اسی طرح سنہ 2019 کی ایک رپورٹ میں ڈیڑھ سالہ بچے کے جسم میں رحم، فیلوپین ٹیوبز اور بیضہ دانی پائے جانے کا ذکر کیا گیا تھا۔
End of سب سے زیادہ پڑھی جانے والی
لیکن یہ کیسے ممکن ہے کہ کسی شخص میں مردانہ جسمانی خصوصیات کے ساتھ رحم اور فیلوپین ٹیوبز بھی موجود ہوں؟ یہ حالت کیوں پیدا ہوتی ہے اور اس کی تشخیص کیسے کی جاتی ہے؟
جواب جاننے کے لیے بی بی سی نے مریض کا علاج کرنے والے ڈاکٹر سشیل کھر بندہ سے بات کی اور یہی سوالات پوچھے:
اس کیس میں کیا سامنے آیا اور طبی زبان میں اسے کیسے سمجھایا جا سکتا ہے؟
اب اس سوال کا جواب یہ ہے کہ 26 سالہ شخص کی شادی کو دو سال ہو چکے تھے۔ وہ مکمل طور پر جنسی طور پر فعال تھا اور اس میں مردوں میں بلوغت کے دوران پیدا ہونے والی تمام ثانوی جنسی خصوصیات موجود تھیں۔
اولاد نہ ہونے کی وجہ جاننے کے لیے کیے گئے معائنے میں معلوم ہوا کہ اس کے دونوں خصیے اپنی معمول کی جگہ پر موجود نہیں تھے۔
ہارمونز کے ٹیسٹ میں ٹیسٹوسٹیرون، ایف ایس ایچ (فولیکل سٹیمیولیٹنگ ہارمون) اور ایل ایچ (لیوٹینائزنگ ہارمون) کی سطح بھی معمول کے مطابق تھی۔
اس طرح کے نتائج ان مریضوں میں بھی دیکھے جا سکتے ہیں جن کے خصیے جسم کے اندر رہ جاتے ہیں اور اپنی معمول کی جگہ پر نہیں پہنچ پاتے۔ اسے طبی زبان میں کرپٹورکڈزم یا خصیوں کا اپنی جگہ پر نہ اترنا کہا جاتا ہے۔
لیکن ایم آر آئی کے دوران ایک انتہائی غیر معمولی بات سامنے آئی۔ معائنے میں رحم اور رحم کی نالیاں موجود پائی گئیں، جو خصیوں سے جڑی ہوئی تھیں۔
یہ اعضا اندر سے مکمل طور پر نشوونما پا چکے تھے اور ان کی ساخت عام طور پر خواتین کے جسم میں پائے جانے والے اعضا جیسی تھی۔
کیا یہ پیدائش کے وقت سے موجود حالت ہو سکتی ہے؟
جب بچہ ماں کے رحم میں ہوتا ہے تو اس کی جنسی نشوونما کروموسومز اور ہارمونز کے زیرِ اثر ہوتی ہے۔ عام طور پر مردوں میں ’46 ایکس وائے‘ اور خواتین میں ’46 ایکس ایکس‘ کروموسومز ہوتے ہیں۔
اسی دوران جسم میں اینٹی مولیرین ہارمون پیدا ہوتا ہے، جو ان اندرونی تولیدی اعضا کی نشوونما کو روکتا ہے جو عام طور پر مردوں کے جسم میں نہیں بنتے۔
اگر کسی شخص کے کروموسومز ’46 ایکس وائے‘ ہوں تو اس کی نشوونما مرد کے طور پر ہوتی ہے۔ اس دوران اس کے جسم میں اینٹی مولیرین ہارمون پیدا ہوتا ہے۔
اینٹی مولیرین ہارمون رحم اور رحم کی نالیوں جیسے اعضا کی نشوونما کو روکتا ہے۔
لیکن پرسیسٹنٹ مولیرین ڈکٹ سنڈروم میں یا تو یہ ہارمون موجود نہیں ہوتا یا پھر اس کے ریسیپٹر میں خرابی ہوتی ہے۔ اس کے نتیجے میں رحم اور رحم کی نالیوں جیسی ساخت جسم کے اندر موجود رہ جاتی ہیں۔
اس سے جسم کے اندرونی تولیدی اعضا کی ساخت مختلف ہو جاتی ہے، تاہم بیرونی جسمانی ساخت مردوں جیسی ہی رہتی ہے۔ عضوِ تناسل اور مردانہ جسمانی خصوصیات معمول کے مطابق نشوونما پاتی ہیں۔
کیا اس فرد میں خصیوں کے ساتھ بیضہ دانیاں بھی موجود تھیں؟
اس سوال کے جواب میں یہ کہا جا سکتا ہے کہ اس مریض کے جسم میں دونوں طرف رحم اور رحم کی نالیاں موجود تھیں، لیکن بیضہ دانیاں نہیں تھیں۔
بیضہ دانی کی جگہ اس کے خصیے موجود تھے۔ خصیے اور بیضہ دانیاں تولیدی غدود ہوتے ہیں، جو مردوں اور خواتین میں مختلف ہوتے ہیں۔ اس مریض کے دونوں تولیدی غدود خصیے ہی تھے، بیضہ دانیاں نہیں۔
پہلے کسی سکین یا طبی معائنے میں کیوں نہیں پتا چلا؟
اس سوال کا جواب یہ ہے کہ جب مریض ہمارے پاس آیا تو اس کے دونوں خصیے اپنی معمول کی جگہ پر نہیں تھے۔ اس لیے ہارمونز کے ٹیسٹ کے ساتھ اس کے منی کے ٹیسٹ بھی کیے گئے۔ ان ٹیسٹوں میں ’ایزو سپرمیا‘ سامنے آیا، یعنی منی میں سپرمز موجود نہیں تھے۔
اس کے بعد ایم آر آئی کی گئی، جس میں رحم، رحم کی نالیاں اور ان سے جڑے ہوئے خصیے نظر آئے۔
اس کے بعد کروموسومز کا تجزیہ کیا گیا، جس میں ’46 ایکس وائے‘ کروموسومز کی تصدیق ہوئی۔ اس حالت کی تصدیق کے لیے ایم آر آئی اور کروموسومز کا تجزیہ بنیادی ٹیسٹ تھے۔
ایسا نہیں ہے کہ اس حالت کا عموماً 26 سال کی عمر تک پتا نہیں چلتا۔ اگر بروقت سکین کیا جاتا تو اس حالت کی پہلے ہی تشخیص ہو سکتی تھی۔ اس مریض کو ابتدا میں یہ معلوم نہیں تھا کہ اس کے دونوں خصیے اپنی معمول کی جگہ پر موجود نہیں ہیں۔
اس حالت میں عام طور پر درد، الٹی یا بخار جیسی کوئی خاص علامت نہیں ہوتی۔ البتہ ایک علامت یہ ہو سکتی ہے کہ خصیے اپنی معمول کی جگہ پر موجود نہ ہوں۔
مریض کو کن نفسیاتی مشکلات یا زندگی میں کن تبدیلیوں کا سامنا کرنا پڑ سکتا ہے؟
26 سالہ شخص کو جب اچانک معلوم ہوا کہ اس کے جسم میں خواتین کے تولیدی اعضا موجود ہیں تو وہ پریشان اور ذہنی دباؤ کا شکار ہو گیا۔ اسے اپنے مستقبل کے بارے میں بھی فکر تھی۔
ہم نے کافی وقت تک اس کی رہنمائی کی اور اسے سمجھایا کہ یہ ایک نایاب طبی حالت ہے۔ بلاشبہ یہ ایک نایاب حالت ہے، لیکن اس کا مطلب یہ نہیں کہ اس سے مریض کی زندگی میں کوئی بڑی تبدیلی آ جائے گی۔
ان اعضا کو نکالنا ضروری تھا کیونکہ ان میں کینسر کا خطرہ ہو سکتا تھا۔ اب ان اعضا کو نکالنے کے بعد وہ معمول کے مطابق زندگی گزار سکتا ہے۔
کیا ایسے کیسز میں سرجری یا علاج ہمیشہ ضروری ہوتا ہے؟
اس مریض کے معاملے میں سرجری ضروری تھی۔ رحم اور رحم کی نالیاں پوری طرح نشوونما نہیں پا سکی تھیں اور جسم میں ان کا کوئی کام نہیں تھا۔
سرجری کے بعد کی گئی بایوپسی میں یہ بھی معلوم ہوا کہ ایک خصیے میں ایسی تبدیلیاں موجود تھیں جو آگے چل کر کینسر کا سبب بن سکتی تھیں۔ اس لیے ان اعضا کو نکالنا ضروری تھا اور سرجری علاج کا حصہ تھی۔
سرجری کے بعد مریض کی صحت پر کیا اثر پڑا اور آئندہ کیا اثرات ہوں گے؟
مریض آپریشن کے سات دن بعد دوبارہ کام پر چلا گیا۔
چونکہ خصیے ٹیسٹوسٹیرون ہارمون بناتے ہیں، اس لیے انھیں نکالنے کے بعد اس کے جسم میں ٹیسٹوسٹیرون نہیں بنے گا۔ اسے ہر تین ماہ بعد ٹیسٹوسٹیرون کا انجیکشن لگوانا ہوگا۔
اس کے علاوہ اس کی جنسی یا سماجی زندگی میں کسی بڑی تبدیلی کی ضرورت نہیں ہوگی۔
ہم نے سرجری لیپروسکوپی کے ذریعے کی، یعنی دوربین نما آلے کی مدد سے۔ اس طریقے میں جسم پر بڑا چیرا لگانے کی ضرورت نہیں پڑتی اور مریض جلد کام پر واپس آ سکتا ہے۔
ایسے معاملات کے بارے میں لوگوں میں پائی جانے والی عام غلط فہمیاں کیا ہیں؟
اس کیس کے سامنے آنے کے بعد یہ سوال بھی اٹھ رہے ہیں کہ کیا مریض حاملہ ہو سکتا تھا، کیا اسے ماہواری آتی تھی یا باہر سے وہ عورت جیسا دکھائی دیتا تھا۔
ان میں سے کوئی بات درست نہیں ہے۔ اس کے جسم میں رحم اور رحم کی نالیاں موجود تھیں، لیکن وہ پوری طرح نشوونما نہیں پا سکی تھیں اور حمل ٹھہرنا ممکن نہیں تھا۔
اس کے جسم میں خواتین جیسا ہارمونل نظام بھی موجود نہیں تھا۔
ڈاکٹر کا کہنا ہے کہ اس کیس سے یہ سبق ملتا ہے کہ جسم میں ہونے والی کسی بھی غیر معمولی تبدیلی کو نظر انداز نہیں کرنا چاہیے اور نہ ہی اس پر شرمندہ ہونا چاہیے۔
ایسی صورت میں بروقت کسی مستند ڈاکٹر سے رجوع کرنا چاہیے۔ اگر نومولود بچے میں جسم کی ساخت یا کسی عضو کی نشوونما میں کوئی غیر معمولی بات نظر آئے تو فوری طور پر طبی مشورہ لینا چاہیے، کیونکہ بروقت تشخیص سے اس حالت کو بہتر طریقے سے سنبھالنے میں مدد مل سکتی ہے۔
