# اسلام آباد میں شنگھائی تعاون تنظیم کے اجلاس میں انڈیا کی شرکت کیوں اہم ہے؟

پاکستان اور انڈیا کے درمیان تعلقات میں جاری کشیدگی کے باوجود شنگھائی تعاون تنظیم (ایس سی او) کے قومی رابطہ کاروں کا اجلاس اسلام آباد میں جاری ہے، جس میں انڈیا سمیت تنظیم کے تمام رکن ممالک کے نمائندے شریک ہیں۔
یہ اجلاس ایسے وقت میں ہو رہا ہے جب پاکستان نے رواں ماہ ایس سی او کی 2026-27 کے لیے سربراہی سنبھالی ہے اور اگست 2027 میں تنظیم کے سربراہانِ مملکت کے اجلاس کی میزبانی بھی اسلام آباد کو کرنا ہے۔
28 سے 30 ستمبر تک جاری رہنے والے اس اجلاس میں پاکستان کی صدارت کے دوران ایس سی او کی سرگرمیوں کے پروگرام، آئندہ اجلاسوں اور اہم تقریبات کی تیاریوں سمیت تنظیم کے ایجنڈے سے متعلق دیگر امور پر غور کیا جا رہا ہے۔
پاکستان نے اپنی صدارت کے لیے ’تصور کو عمل میں بدلنا: رابطہ کاری، جدت اور مشترکہ خوشحالی‘ کا موضوع منتخب کیا ہے، جس کے تحت علاقائی رابطہ کاری، اقتصادی تعاون، جدت اور رکن ممالک کے عوام کے درمیان روابط کو ترجیح دینے کا اعلان کیا گیا ہے۔
اس اجلاس میں انڈیا کی شرکت خاص طور پر قابلِ توجہ ہے کیونکہ یہ دونوں ملکوں کے درمیان دوطرفہ تعلقات میں کشیدگی کے ماحول میں ہو رہی ہے۔
انڈیا اس سے قبل بھی رواں سال جولائی میں اسلام آباد میں ہونے والے ایس سی او کے سرحدی خدمات کے سربراہان کے اجلاس میں شریک ہو چکا ہے۔ موجودہ اجلاس میں گفتگو کا مرکز ایس سی او کی سرگرمیوں، باہمی رابطہ کاری اور آئندہ اجلاسوں کی تیاری ہے۔
اسلام آباد میں ہونے والا یہ اجلاس اگلے سال ہونے والے ایس سی او سربراہی اجلاس کی تیاری کے عمل کا ابتدائی مرحلہ بھی ہے۔
پاکستان نے کہا ہے کہ اگست 2027 میں ہونے والے سربراہی اجلاس میں ایس سی او کے تمام رکن ممالک کے رہنماؤں کو دعوت دی جائے گی، جس میں انڈیا کے وزیر اعظم نریندر مودی بھی شامل ہوں گے۔

انڈیا شنگھائی تعاون تنظیم (ایس سی او) کے قومی رابطہ کاروں کی کونسل کے اجلاس میں اسلام آباد میں شریک ہے، جہاں قومی رابطہ کار الوک دمری انڈین وفد کی قیادت کر رہے ہیں۔
انڈین وفد کی اسلام آباد میں موجودگی دونوں ممالک کے درمیان کشیدہ تعلقات کے پس منظر میں ہو رہی ہے۔ یہ گذشتہ سال انڈیا اور پاکستان کے درمیان ہونے والی فوجی کشیدگی کے بعد انڈیا کا پاکستان کا پہلا ایسا دورہ بھی ہے۔
اس سے قبل انڈیا کے وزیر خارجہ ایس جے شنکر نے اکتوبر سنہ 2024 میں ایس سی او کی کونسل آف ہیڈز آف گورنمنٹ کے اجلاس میں شرکت کے لیے اسلام آباد کا دورہ کیا تھا۔
انڈیا اور پاکستان نے تقریباً ایک عشرے سے باہمی مذاکرات نہیں کیے
نئی دہلی سے نامہ نگار شکیل اختر کے مطابق انڈیا نے اسلام آباد میں ہونے والی شنگھائی کو آپریشن آرگنائزیشن (ایس سی او) کے کوآرڈینیٹرز کی میٹنگ میں شرکت کے لیے وزارت خارجہ کے سینئر اہلکار آلوک دیمری کی قیادت میں انڈین ڈیلیگیؤن اسلام آباد بھیجا ہے۔ آلوک دیمری وزارت خارجہ میں جوائنٹ سکریٹری ہیں اور وہ ایس سی او کے لیے انڈیا کے کو آرڈینیٹر ہیں۔
انڈیا اور پاکستان نے تقریبا ایک عشرے سے باہمی مذاکرات نہیں کیے ہیں۔ دونوں ملکوں کے درمیان آخری بات چیت دسمبر سنہ 2015 میں ہوئی تھی جب اس وقت کی انڈیا کی وزیر خارجہ سشما سوراج نے ’جامع باہمی مذاکرات‘ کے عمل کو دوبارہ شروع کرنے کی عرض سے اسلام آباد کا دورہ کیا تھا۔ تاہم جنوری سنہ 2016 میں پٹھان کوٹ پر حملے کے بعد یہ عمل ایک بار پھر رک گیا۔
انڈیا کی حکومت نے یہ موقف اختیار کر رکھا ہے کہ پاکستان سے بات چیت صرف اسی صورت میں ہو سکے گی جب ’فضا دہشت گردی سے پوری طرح صاف ہو۔‘ تاہم وہ ایس سی او جیسے ملٹی لیٹرل پلیٹ فارم پر پاکستان کے ساتھ شرکت کے لیے تیار ہے۔
’انڈیا کی ہمیشہ یہ کوشش رہی ہے کہ وہ ایس سی او کے عمل کو خراب نہ کرے‘
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
بین الاقوامی امور کی تجزیہ کار نروپما سبرامنین کہتی ہیں کہ انڈیا اور پاکستان شنگھائی کوآپریشن گروپ میں بہت باقاعدگی کے ساتھ شریک ہوتے رہے ہیں۔
بی بی سی سے بات کرتے ہوئے انھوں نے کہا کہ ’دونوں ممالک اسے ایک ملٹی لیٹرل پلیٹ فارم سمجھتے ہیں جس میں چین اور روس بھی شامل ہیں۔ اس لیے دونوں کے لیے یہ اہم ہے کہ وہ اس میں شریک ہوں۔‘
ان کا کہنا تھا کہ ’پاکستان اور انڈیا دونوں ہی چین اور روس سے وسیع تر پارٹنرشپ کے خواہاں ہیں۔ انڈیا ایس سی او اور برکس جیسے فورم کے ذریعے یہ اشارہ بھی دینا چاہتا ہے کہ وہ امریکہ پر منحصر نہیں ہے اور وہ دوسرے پارٹنر بھی چن رہا ہے اور چین جیسے اپنے مخالف ملکوں کے ساتھ بھی پارٹنر شپ کے لیے تیار ہے۔‘
’تاہم دباؤ کے باوجود وہ روس کے ساتھ اپنے تجارتی تعلقات برقرار رکھنا چاہے گا۔ ایس سی او میں جو وسطی ایشیا کے ممالک شامل ہیں ان سے انڈیا کے تعلقات سویت یونین کے زمانے سے چلے آ رہے ہیں۔ انڈیا ایس سی او کے ذریعے ان تعلقات کو مزید وسعت دینے کی کوشش کر رہا ہے۔‘
نروپما مزید کہتی ہیں کہ ’انڈیا کی ہمیشہ یہ کوشش رہی ہے کہ وہ ایس سی او کے عمل کو خراب نہ کرے۔ جب اسلام آباد کا سربراہی اجلاس آئے گا اس وقت یہ معلوم ہو گا کہ انڈیا کا اس اجلاس کے تئیں کتنا سنجیدہ ہے۔‘
ان کا کہنا ہے کہ ’پہلے کے سربراہی اجلاس میں وزیر اعظم مودی جاتے رہے ہیں، اسلام آباد کے اجلاس میں وہ جاتے ہیں کہ نہیں یہ دیکھنا ہوگا۔ کیونکہ سارک دس برس سے رکا ہوا ہے۔ سارک کا سربراہی اجلاس نومبر 2016 میں اسلام آباد میں میں ہونا تھا لیکن اوڑی میں حملے کے بعد وزیر اعظم وہاں نہیں گئے۔‘
’اگر مودی ایس سی او کے سربراہی اجلاس میں شرکت کرتے ہیں تو کیا وہ سارک میں بھی جا سکتے ہیں؟ یہ سوال بھی اٹھے گا۔‘
انڈین وفد کی ایس سی او کے قومی رابطہ کاروں کی کونسل کے اجلاس میں شرکت پر دونوں ملکوں کے صحافیوں کے تبصرے
اسلام آباد میں شنگھائی تعاون تنظیم (ایس سی او) کے قومی رابطہ کاروں کی کونسل کے اجلاس کے دوران پاکستان اور انڈیا کے صحافی بھی سوشل میڈیا پر اس اجلاس اور دونوں ممالک کے تعلقات کے تناظر میں اپنے خیالات کا اظہار کر رہے ہیں۔
بعض صحافیوں نے انڈین وفد کی اسلام آباد میں موجودگی کو دونوں ممالک کے درمیان کشیدہ تعلقات کے پس منظر میں اہم قرار دیا، جبکہ کچھ نے سوشل میڈیا پر دونوں ملکوں کے ایک دوسرے کے اکاؤنٹس تک رسائی بحال ہونے کی امید بھی ظاہر کی ہے۔
انڈیا کی معروف صحافی سوہاسنی حیدر نے ایکس پر لکھا کہ ’اقوامِ متحدہ کی جنرل اسمبلی میں انڈیا اور پاکستان کے درمیان لفظی تکرار کے چند روز بعد پاکستانی صحافیوں کے مطابق، انڈین وفد اگلے سال ہونے والے ایس سی او کے کثیر ملکی سربراہی اجلاس کی تیاری کے لیے اسلام آباد میں موجود ہے۔‘
انھوں نے مزید کہا کہ ’امید ہے کہ انڈیا اور پاکستان سوشل میڈیا پر ایک دوسرے کے اکاؤنٹس تک رسائی بحال کرنے پر بھی بات کریں گے۔‘
پاکستانی سنئیر صحافی اعزاز سید نے ایکس پر لکھا ’پاکستان کی سربراہی میں ایس سی او کے قومی رابطہ کاروں کا پہلا اجلاس شروع ہو چُکا ہے اور دو سال بعد انڈین وفد بھی اسلام آباد میں موجود ہے۔ یہ اس بات کی عکاسی کرتا ہے کہ دونوں ممالک کے درمیان اب بھی امن کی ایک معمولی سی امید موجود ہے۔‘
انھوں نے مزید لکھا کہ ’پاکستان کے قومی رابطہ کار ڈاکٹر ایم فیصل تین روزہ مذاکرات کی صدارت کر رہے ہیں، جبکہ انڈیا کے قومی رابطہ کار الوک دمری انڈین وفد کی قیادت کر رہے ہیں۔‘
امریکی تھنک ٹینک سٹمسن سینٹر میں جنوبی ایشیا پروگرام کی ڈائریکٹر اور سینیئر فیلو الزبتھ تھریلکیلڈ نے بھی انڈیا کے وفد کے پاکستان آنے پر اپنے خیالات کا اظہار ایکس پر کیا۔
الزبتھ تھریلکیلڈ نے لکھا کہ ’غیر سرکاری طور پر انڈیا اور پاکستان کے درمیان رابطوں سے متعلق سامنے آنے والی متعدد رپورٹس کے درمیان یہ ایک حوصلہ افزا پیش رفت ہے، انڈیا نے جمعے کو پاکستان کی سربراہی میں اسلام آباد میں ہونے والے شنگھائی تعاون تنظیم (ایس سی او) کے سرحدی خدمات کے سربراہان کے 12ویں اجلاس میں شرکت کی۔‘
ایس سی او کا موجودہ اجلاس 30 ستمبر تک جاری رہے گا، جس میں رکن ممالک کے درمیان تنظیم کی آئندہ سرگرمیوں، باہمی رابطہ کاری اور اگلے سال ہونے والے سربراہانِ مملکت کے اجلاس کی تیاریوں پر بات چیت کی جا رہی ہے۔
پاکستان نے یکم ستمبر کو ایس سی او کی 2026-27 کے لیے صدارت سنبھالی تھی اور اگست 2027 میں تنظیم کے سربراہانِ مملکت کے اجلاس کی میزبانی بھی اسلام آباد کرے گا۔ انڈین وفد کی شرکت سے یہ بھی ظاہر ہوتا ہے کہ دونوں ممالک کے درمیان کشیدگی کے باوجود ایس سی او کے کثیر ملکی فورم پر دونوں کی شرکت جاری ہے۔
