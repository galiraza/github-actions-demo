# وفاقی حکومت کس قانون کے تحت خیبرپختونخوا میں ایمرجنسی یا گورنر راج نافذ کر سکتی ہے؟

وفاقی حکومت کس قانون کے تحت خیبرپختونخوا میں ایمرجنسی یا گورنر راج نافذ کر سکتی ہے؟
پاکستان تحریک انصاف (پی ٹی آئی) نے گذشتہ ہفتے 27 ستمبر کو اپنا اسلام آباد کی طرف لانگ مارچ مؤخر کر دیا تھا اور چار اکتوبر کی نئی تاریخ کا اعلان کیا تھا۔ تاہم بظاہر ایسا معلوم ہوتا ہے کہ دارالحکومت میں کنٹینرز لگانے اور ملک بھر سے پولیس اہلکاروں کو بُلانے کے علاوہ وفاقی حکومت اس احتجاج کو روکنے کے لیے دیگر آپشنز پر بھی غور کر رہی ہے۔
گذشتہ دو، تین دنوں سے متعدد حکومتی شخصیات بشمول وفاقی وزرا اور خیبر پختونخوا کے گورنر ’ایمرجنسی‘ اور ’گورنر راج‘ کے الفاظ دہراتے نظر آتے ہیں۔
گذشتہ روز وفاقی وزیرِ قانون اعظم نذیر تارڑ نے جیو نیوز سے گفتگو میں کہا تھا کہ ’اگر حالات آئینی تقاضا کریں تو صوبے میں گورنر راج یا ایمرجنسی نافذ کی جا سکتی ہے۔‘
اسی طرح کا بیان منگل کو خیبر پختونخوا کے گورنر فیصل کریم کنڈی نے بھی دیا جب انھوں نے صوبے میں پی ٹی آئی کی حکومت کو مخاطب کرتے ہوئے کہا کہ ’میں چیلنج کرتا ہوں کہ چار تاریخ کو یہ اسلام آباد پہنچ کر دکھائیں، آپ صرف صوابی پہنچ کر دکھائیں۔‘
جب صحافیوں نے ان سے گورنر راج یا ایمرجنسی کے نفاذ کے حوالے سے سوال کیا، تو ان کا کہنا تھا کہ ’میں خواب تو نہیں دیکھ سکتا کہ ایسا کب ہوگا اور کب نہیں ہوگا، جب صوبے کی صورتحال ایسی ہوگی، جس میں ضرورت ہوگی کہ گورنر راج لگایا جائے یا ایمرجنسی لگائی جائے تو آئین میں شق موجود ہے۔‘
انھوں نے مزید کہا کہ ’جب صدر پاکستان، وزیر اعظم پاکستان اور باقی اداروں کے لوگ بیٹھیں گے اور سمجھیں گے کہ صوبے میں گورنر راج کی گنجائش ہے یا ایمرجنسی کی گنجائش ہے تو وہ لگائیں گے۔‘
وفاقی حکومت کی جانب سے خیبر پختونخوا کی سکیورٹی صورتحال کو بھی ایمرجنسی اور گورنر راج کے آپشنز پر غور کرنے کی وجہ بتایا جا رہا ہے۔
ادھر پی ٹی آئی کے اندر چار اکتوبر کے لانگ مارچ کے حوالے سے آرا منقسم نظر آتی ہیں۔

بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
ایک طرف وزیر اعلیٰ سہیل آفریدی عوام کے مینڈیٹ کے خلاف کسی بھی اقدام کو ’ناقابلِ قبول‘ اور بظاہر لانگ مارچ کی تیاریاں کرتے ہوئے نظر آتے ہیں، تو دوسری جانب پی ٹی آئی کے رہنما شوکت یوسفزئی نے جماعت کے چیئرمین بیرسٹر گوہر کو ایک خط لکھ کر لانگ مارچ ملتوی کرنے کا مطالبہ کیا ہے۔
انھوں نے اپنے خط میں کہا ہے وفاقی حکومت ’کچھ معاملات کو جواز بنا کر‘ خیبر پختونخوا میں ایمرجنسی نافذ کرنا چاہتی ہے، جس میں افغانستان کی طرف سے پاکستان پر ممکنہ حملے کا جواز دیا جاسکتا ہے۔
انھوں نے مزید لکھا کہ ’جس الزام کو بنیاد بنایا جا رہا ہے وہ دہشت گردی کو سپورٹ کرنا ہے اور اس کے لیے ایک شق موجود ہے کہ سیاست پر پابندی بھی لگ سکتی ہے۔‘
’میری گزارش ہے کہ ان حالات میں پارٹی اپنے اور ملک کے وسیع تر مفاد میں بہتر حکمت عملی وضع کرے اور سوچے کہ اس صورتحال میں ایک ذمہ دار اور پاپولر سیاسی جماعت ہونے کے ناطے ہمیں کیا کرنا چاہئے۔ میری رائےمیں ہمیں لانگ مارچ ملک کے وسیع تر مفاد میں فوری طور پر ملتوی کرنا چاہیے۔‘
یہاں یہ سوال پیدا ہوتا ہے کہ وفاقی حکومت کن قوانین اور کس صورتحال میں کسی صوبے میں ایمرجنسی یا گورنر راج نافذ کر سکتی ہے؟
ایمرجنسی کا نفاذ کیسے ممکن ہے؟
کسی صوبے میں ہنگامی حالت یا ایمرجنسی کا نفاذ پاکستان کے آئین کی شق 232 کے تحت کیا جا سکتا ہے اور اس کے نفاذ کا حق صدرِ پاکستان کے پاس ہے۔
آئین کی اس شق کے مطابق ایمرجنسی جنگ اور بیرونی جارحیت کے علاوہ داخلی انتشار یا خلل کی وجہ سے بھی لگائی جا سکتی ہے۔
اس شق میں کہا گیا ہے کہ اگر صدر اس بات پر مطمئن ہوں کہ کوئی سنگین ہنگامی صورتحال موجود ہے، جس میں پاکستان یا اس کے کسی حصے کی سلامتی کو جنگ، بیرونی جارحیت یا اندرونی انتشار سے ایسا خطرہ لاحق ہے جس پر قابو پانا صوبائی حکومت کے اختیار میں نہیں رہا تو وہ ایمرجنسی کے نفاذ کا اعلان کر سکتے ہیں۔
آئین کے مطابق صوبے میں ہنگامی حالت کے نفاذ کے لیے صوبائی اسمبلی کی قرارداد کی ضرورت ہوتی ہے، تاہم اگر صدر اپنی مرضی سے عمل کرے تو ایمرجنسی کے نفاذ کا فیصلہ دس دن کے اندر پاکستانی پارلیمان کے دونوں ایوانوں یعنی قومی اسمبلی اور سینیٹ کے سامنے منظوری کے لیے پیش کیا جانا لازمی ہے۔
آئین کے مطابق ملک کے جس علاقے میں جب تک ایمرجنسی کا نفاذ رہے گا اس وقت تک مجلسِ شوریٰ یا پارلیمان کو اختیار حاصل ہو گا کہ وہ کسی صوبے یا اس کے کسی حصے کے لیے ایسے معاملے پر قوانین بنا سکتی ہے، جو وفاقی قانون ساز فہرست یا متوازی قانون ساز فہرست میں درج نہ ہوں۔
کسی صوبے میں ایمرجنسی کے نفاذ کی مدت کے دوران وفاق کا انتظامی اختیار ہدایات دینے تک ہو گا کہ مذکورہ صوبے کے انتظامی اختیار کو کس طرح استعمال کیا جائے۔
آرٹیکل 232 کے مطابق ایمرجنسی کے نفاذ کی صورت میں وفاقی حکومت صوبے کا انتظام خود سنبھال سکتی ہے یا صوبے کے گورنر کو ہدایت دے سکتی ہے کہ وہ وفاقی حکومت کی طرف سے صوبائی حکومت کے تمام یا چنندہ فرائض اور صوبے میں صوبائی اسمبلی کے علاوہ کسی بھی ادارے یا اختیار کو دیے گئے یا قابل استعمال اختیارات سنبھالے۔
تاہم آئین میں واضح ہے کہ اس صورتحال میں کسی بھی صورت میں وفاقی حکومت کو یہ اختیار نہیں دیا جا سکتا کہ وہ خود ہائی کورٹ کے اختیارات سنبھالے یا صوبے کے گورنر کو اپنی طرف سے ایسا کرنے کو کہے یا ہائیکورٹ سے متعلق آئین کی کسی بھی شق کو مکمل یا جزوی طور پر معطل کرے۔
آئینِ پاکستان کے جس آرٹیکل کے تحت صوبے میں ہنگامی حالت نافذ کی جا سکتی ہے اس میں یہ بھی کہا گیا ہے کہ اس آرٹیکل میں کوئی بھی چیز صوبائی اسمبلی کے کسی بھی قانون کو بنانے کے اختیار کو محدود نہیں کرے گی جو اسے آئین کے تحت بنانے کا اختیار ہے۔
لیکن اگر کسی صوبائی قانون کی کوئی شق اس ایکٹ آف پارلیمنٹ کی کسی شق کے خلاف ہو جس کا پارلیمان کو ہنگامی حالت کے نفاذ کے ایکٹ کے تحت بنانے کا اختیار حاصل ہے تو پارلیمنٹ کا بنایا گیا قانون چاہے صوبائی قانون سے پہلے منظور ہو یا بعد میں، اس وقت تک غالب رہے گا جب تک ہنگامی حالت کا نفاذ باقی رہتا ہے۔
ایمرجنسی کے نفاذ کا اعلان پارلیمان کے اس مشترکہ اجلاس کے سامنے رکھا جاتا ہے، جو صدر کی طرف سے اعلان کے اجرا کے 30 دن کے اندر بلایا جانا لازم ہے اور یہ ایمرجنسی دو ماہ کی میعاد مکمل ہونے پر ختم بھی ہو جائے گی، تاوقتیکہ میعاد ختم ہونے سے پہلے مشترکہ اجلاس کی قرارداد کے ذریعے اس کا نفاذ دوبارہ منظور نہ کیا جائے۔
آئین کے مطابق اگر پارلیمان کا مشترکہ اجلاس دونوں ایوانوں کے کل اراکین کی اکثریت سے ایمرجنسی کے نفاذ کو مسترد کر دے تو اس صورت میں بھی ایمرجنسی کا نفاذ باقی نہیں رہے گا۔
گورنر راج کیوں لگتا ہے اور اسے لگانے کا اختیار کس کا ہے؟
کسی بھی صوبے میں وزیراعلیٰ اور صوبائی اسمبلی کے اراکین عوام کے منتخب کردہ نمائندے ہوتے ہیں، جو اس صوبے کے حکومتی اور انتظامی امور چلاتے ہیں۔ اس کے ساتھ ساتھ گورنر صوبوں میں وفاق کے نمائندے کے طور پر موجود ہوتا ہے۔
عام حالات میں گورنر کا عہدہ محض علامتی ہوتا ہے جبکہ صوبے میں حکومتی پالیسیوں پر عملدرآمد کروانا منتخب صوبائی حکومت کے ذمے ہوتا ہے۔
مگر جب گورنر راج لگتا ہے تو ایک محدود مدت کے لیے تمام تر اختیارات گورنر کے پاس آ جاتے ہیں اور وزیراعلیٰ اور ان کی اسمبلی بے اختیار ہو جاتی ہے۔
آئین پاکستان کے دسویں باب میں دو شقیں ایسی ہیں، جن کی بنیاد پر صوبے میں ایمرجنسی یا گورنر راج لگایا جا سکتا ہے۔ ان میں آرٹیکل 232 اور 234 شامل ہیں۔
اگر صدرِ پاکستان سمجھتے ہیں کہ پاکستان یا اس کے کسی حصے کو جنگ یا بیرونی خطرات لاحق ہیں، یا ایسی اندرونی شورش ہے جو صوبائی حکومت کے بس کی بات نہیں، تو وہ ایمرجنسی اور گورنر راج نافذ کر سکتے ہیں۔
تاہم اندرونی خلفشار کے باعث لگنے والی ایمرجنسی کی صورت میں اس فیصلے کے نفاذ سے پہلے اس کی متعلقہ صوبائی اسمبلی سے توثیق لازم ہے۔
اسی شق میں یہ بھی کہا گیا ہے کہ اگر صدر اپنے طور پر ہی گورنر راج لگانے کا فیصلہ کر لیں تو اس صورت میں انھیں پارلیمان کے دونوں ایوانوں سے دس روز کے اندر اپنے فیصلے کی توثیق کروانا ہو گی۔
ایسی ایمرجنسی کی صورت میں اس صوبے کے انتظامی امور اور قانون سازی کا اختیار وفاق میں پارلیمان کے پاس منتقل ہو جائے گا۔
گورنر راج کی مدت اور توثیق کا عمل
وفاقی حکومت خود یا صوبائی گورنر کو اختیار دے سکتی ہے کہ وہ وفاقی حکومت کے نمائندے کے طور پر وہاں تمام امور سرانجام دے، تاہم صوبے کی ہائی کورٹ کے اختیارات وفاقی حکومت یا گورنر کو منتقل نہیں ہوتے۔
اب آرٹیکل 234 کے تحت اگر صدر کو صوبائی گورنر یہ رپورٹ پیش کر دے کہ صوبے میں ایسی صورتحال پیدا ہو گئی ہے جس میں صوبائی حکومت آئین کے وضع کردہ طریقہ کار کے تحت کام نہیں کر سکتی، تو وہ یہاں گورنر راج نافذ کرتے ہوئے تمام اختیارات اپنے پاس رکھ سکتے ہیں یا صوبے میں اپنے نمائندے گورنر کے حوالے کر سکتے ہیں۔
آئین کے مطابق اس صورت میں صوبائی اسمبلی کے اختیارات پارلیمنٹ کو منتقل ہو جاتے ہیں جبکہ ہائی کورٹ کے اختیارات نہیں لیے جا سکتے۔
گورنر راج کی مدت صرف دو ماہ ہو سکتی ہے اور اس کی توثیق بھی پارلیمان کے مشترکہ اجلاس سے کروانا ہوتی ہے جبکہ گورنر راج میں مشترکہ اجلاس کے ذریعے ایک وقت میں صرف دو ماہ کی توسیع ہی کی جا سکتی ہے اور اس کی کل میعاد چھ ماہ سے زیادہ نہیں ہو سکتی۔
اگر اس دوران قومی اسمبلی تحلیل ہو بھی جائے تو صوبے میں گورنر راج تین ماہ تک نافذ رہے گا جب تک کہ قومی اسمبلی کے انتخابات نہ ہو جائیں یا سینیٹ اس ضمن میں کوئی فیصلہ کر لے۔
کیا گورنر راج کو عدالت میں چیلنج کیا جا سکتا ہے؟
پاکستانی آئین کی شق نمبر 236 کے تحت اس کے 10ویں باب میں موجود کسی بھی آرٹیکل کو عدالت میں چیلنج نہیں کیا جا سکتا۔
تاہم قانونی ماہر احمد بلال محبوب نے ماضی میں بی بی سی اردو سے بات کرتے ہوئے کہا تھا کہ اب تک لگنے والی تقریباً تمام ہی ایمرجنسیز اور گورنر راج کو عدالتوں میں چیلنج کیا گیا۔
اس بارے میں بات کرتے ہوئے انھوں نے کہا تھا کہ اس باب کے تحت تو ایسے کسی حکم نامے پر سوال نہیں اٹھایا جا سکتا، تاہم کوئی بھی شخص عدالت میں رٹ دائر کر سکتا ہے کہ آئین میں موجود دیگر شقوں کی خلاف ورزی ہوئی یا اس کے بنیادی حقوق سلب کیے گئے۔
پاکستان میں گورنر راج کی تاریخ
پاکستان کی تاریخ میں مختلف ادوار میں اب تک 15 بار مرکز یا وفاق نے صوبوں میں گورنر راج لگایا گیا۔ یہ عمل تین، تین بار پنجاب، سندھ اور صوبہ سرحد(خیبرپختونخوا) ، چار بار بلوچستان اور دو بار مشرقی بنگال میں ہوا۔
ایسا تب ہوا جب صوبائی اسمبلیاں تحلیل کی گئیں اور انتظامی اختیار براہِ راست مرکز کے ماتحت آ گیا، جیسا کہ 1958–1972 اور 1977–1985 کے مارشل لا کے ادوار میں ہوا یا پھر جب مرکز نے کسی نہ کسی جواز کے تحت گورنر راج نافذ کیا۔
صوبہ سرحد (موجودہ خیبرپختونخوا ) میں بھی تین بار گورنر راج نافذ کیا گیا۔ یہ راج پہلی بار 16 فروری 1975 سے 3 مئی 1975 تک، دوسری بار 25 فروری 1994 سے 24 اپریل 1994 تک اور تیسری بار 12 اکتوبر 1999 سے 30 نومبر 2002 تک نافذ العمل رہا۔
