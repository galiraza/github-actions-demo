# وہ مسلمان خاتون جس نے آکسفورڈ سے صدیوں قبل دنیا کی قدیم ترین یونیورسٹی تعمیر کروائی

وہ مسلمان خاتون جس نے آکسفورڈ سے صدیوں قبل دنیا کی قدیم ترین یونیورسٹی تعمیر کروائی

فاس شمالی افریقہ کے مسلم اکثریتی ملک مراکش کا ایک قدیم شہر ہے۔ اس پہاڑی شہر کی گلیاں اتنی تنگ اور پیچیدہ ہیں کہ وہاں گاڑیاں تک نہیں جا سکتیں۔ یہی وجہ ہے کہ لوگ آج بھی آمدورفت اور سامان ڈھونے کے لیے گدھوں اور بیل گاڑیوں پر انحصار کرتے ہیں۔
شہر کے مرکزی حصے میں داخل ہوں تو ایک دوسرے سے جڑی بے شمار تنگ گلیاں دکھائی دیتی ہیں، جنھیں دیکھ کر ایسا لگتا ہے جیسے آپ کسی بھول بھلیاں میں داخل ہو گئے ہوں۔
تاہم اگر شہر کے کسی بلند مقام سے اس کے مرکز کی طرف دیکھا جائے تو سب سے پہلے سبز رنگ کی خوبصورت ٹائلوں والی چھتیں نظر آتی ہیں۔
انھی چھتوں کے نیچے ایک ایسا ادارہ موجود ہے جو 1150 سال سے زیادہ عرصے سے علم کی روشنی پھیلا رہا ہے۔ آج اسے جامعہ القرویین (University of Al-Qarawiyyin) کہا جاتا ہے۔ بعض لوگ اسے القرویین بھی کہتے ہیں۔
859 میں قائم ہونے والے اس تعلیمی ادارے کو یونیسکو اور گنیز بک آف ورلڈ ریکارڈز نے دنیا کی قدیم ترین یونیورسٹی قرار دیا ہے۔
’القرویین‘ کے قیام کے دو سو سال سے بھی زیادہ عرصے بعد اٹلی میں یونیورسٹی آف بولونیا نے 1088 میں کام شروع کیا، جسے یورپ کی قدیم ترین جامعہ تسلیم کیا جاتا ہے۔
انگلینڈ میں یونیورسٹی آف آکسفورڈ کی بنیاد 12ویں صدی کے اواخر میں پڑی، جبکہ یونیورسٹی آف کیمبرج تقریباً ایک صدی بعد قائم ہوئی۔
لیکن اس سے بھی زیادہ حیران کن بات یہ ہے کہ دنیا کی قدیم ترین جامعہ، القرویین، ایک مسلم خاتون فاطمہ الفہری نے قائم کی تھی۔
End of سب سے زیادہ پڑھی جانے والی
یہ فاطمہ آخر کون تھیں؟ انھوں نے نویں صدی میں ایک یونیورسٹی کیوں اور کیسے قائم کی؟ ان سوالوں کے جواب ہم آگے چل کر جانیں گے، لیکن اس سے پہلے ایک بات واضح کرنا ضروری ہے۔
یونیسکو اور گنیز بک آف ورلڈ ریکارڈز نے جامعہ القرویین کو اُن جامعات میں دنیا کی قدیم ترین یونیورسٹی قرار دیا ہے جو آج بھی مسلسل تعلیمی سرگرمیاں جاری رکھے ہوئے ہیں۔
اس کا مطلب یہ ہے کہ القرویین سے پہلے بھی مختلف علاقوں میں تعلیمی مراکز موجود رہے ہوں گے۔ مثال کے طور پر قدیم انڈیا کی تکششیلا اور نالندا کا ذکر ملتا ہے، لیکن یہ ادارے اب فعال نہیں ہیں۔
تاہم ایک ہزار سال سے زیادہ عرصہ گزر جانے کے باوجود جامعہ القرویین آج بھی دنیا بھر کے طلبہ کو اپنی جانب راغب کرتی ہے۔
فاطمہ الفہری کے بارے میں معلومات مختلف تاریخی کتابوں میں ملتی ہیں تاہم ان میں سے زیادہ تر معلومات 14ویں صدی میں لکھی گئی ایک قدیم تاریخی کتاب سے ماخوذ ہیں۔
یہ کتاب ’روض القرطاس‘ کے نام سے معروف ہے جسے مؤرخ ابن ابی زرع نے تحریر کیا تھا۔ یہ بنیادی طور پر فاس شہر کی قدیم تاریخ پر مبنی کتاب ہے۔
مؤرخ ابن ابی زرع کے مطابق فاطمہ الفہری کا تعلق عرب کے معزز قریش قبیلے سے تھا۔ ان کے آباؤ اجداد پیشے کے اعتبار سے تاجر تھے اور ایشیا اور افریقہ کے مختلف شہروں کا سفر کیا کرتے تھے۔
کاروبار کے سلسلے میں وہ ایک موقع پر قیروان (موجودہ تیونس) پہنچے اور وہیں آباد ہو گئے۔ فاطمہ الفہری کی پیدائش تقریباً 800 عیسوی میں اسی شہر میں ہوئی تھی۔
ان کا پورا نام فاطمہ بنت محمد الفہری القریشیہ تھا۔ ان کے والد محمد الفہری بھی تجارت کے پیشے سے وابستے تھے۔
کاروبار میں طویل عرصے تک مندی رہنے کے باعث وہ مالی مشکلات کا شکار ہو گئے۔ بہتر مواقع کی تلاش میں انھوں نے نویں صدی کے اوائل میں مراکش اس امید کے ساتھ نقل مکانی کی کہ وہاں ان کی قسمت بدل جائے گی۔
مال و دولت کا حصول
مراکش نقل مکانی کے بعد محمد الفہری کے مالی حالات بتدریج بہتر ہونے لگے۔ کچھ عرصے بعد انھوں نے اپنے خاندان کے ساتھ فاس شہر میں مستقل طور پر آباد ہونے کا فیصلہ کیا۔
اس وقت فاس آہستہ آہستہ تجارت اور کاروبار کا ایک اہم مرکز بنتا جا رہا تھا۔ بعد میں محمد الفہری ایک خوش حال تاجر بن گئے۔
اس تاجر نے اپنے تمام بیٹوں اور بیٹیوں کو دینی تعلیم دلوائی۔
فاطمہ الفہری نے بچپن میں اپنے خاندان سے اسلامی تعلیمات اور مذہبی اصول سیکھے۔ بعد میں انھوں نے قرآن و حدیث کے ساتھ ساتھ فقہ، یعنی اسلامی قانون، سمیت کئی دیگر علوم بھی حاصل کیے۔
نوجوانی میں فاطمہ کی شادی فاس کے ایک معزز خاندان میں ہوئی۔ تاہم چند ہی برس بعد ان کی زندگی میں یکے بعد دیگر کئی صدمے آئے۔
فاطمہ کم عمری میں اپنے شوہر کی وفات کے بعد بیوہ ہو گئیں۔ اس کے کچھ ہی عرصے بعد ان کے والد اور بھائی بھی وفات پا گئے۔
نتیجتاً خاندان کی بڑی وراثت فاطمہ الفہری اور ان کی بہن مریم الفہری کے حصے میں آئی۔
جامعہ القرویین کا قیام
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
وراثت میں بڑی دولت ملنے کے بعد فاطمہ اور ان کی بہن مریم نے فیصلہ کیا کہ وہ یہ دولت عیش و عشرت پر خرچ نہیں کریں گی۔ انھوں نے طے کیا کہ اپنے وسائل ایسے کاموں کے لیے وقف کریں گی جو اسلام اور انسانیت کے لیے فائدہ مند ہوں۔
اس زمانے میں فاس میں مسلمانوں کی تعداد تیزی سے بڑھ رہی تھی۔ اسی ضرورت کو مدنظر رکھتے ہوئے دونوں بہنوں نے شہر میں بڑی مسجدیں اور دینی تعلیمی مراکز قائم کرنے کا فیصلہ کیا۔
مریم الفہری نے اپنی دولت کا بڑا حصہ خرچ کر کے ایک بڑی مسجد تعمیر کروائی، جو الاندلس مسجد کے نام سے جانی جاتی ہے۔
خوبصورت نقش و نگار سے مزین اس مسجد کا رقبہ اتنا وسیع ہے کہ یہاں ایک وقت میں تقریباً 20 ہزار نمازی نماز ادا کر سکتے ہیں۔
دوسری جانب فاطمہ الفہری نے بھی اپنی دولت سے ایک اور بڑی مسجد تعمیر کروائی، تاہم یہ مسجد دوسری مساجد کے مقابلے میں کچھ مختلف تھی۔
فاطمہ نے مسجد کے منصوبے میں نماز کی جگہ کے ساتھ ایک بڑا اسلامی تعلیمی مرکز، یعنی مدرسہ، بھی شامل کیا۔
مؤرخین کے مطابق اس کی تعمیر یکم رمضان 245 ہجری (859 عیسوی) میں شروع ہوئی تھی۔ اگلے دو سال کے دوران مسجد کی مرکزی عمارت مکمل کر لی گئی۔
تاہم مختلف تاریخی روایات کے مطابق، مدرسے سمیت اس پورے تعلیمی و مذہبی مرکز کی تعمیر مکمل ہونے میں 12 سے 18 سال لگے تھے۔
مؤرخ ابن ابی زرع نے اپنی کتاب روض القرطاس میں لکھا ہے کہ فاطمہ خود مسجد کے اس پورے منصوبے کی تعمیر کی مسلسل نگرانی کرتی تھیں۔
کہا جاتا ہے کہ انھوں نے اس کام کے لیے خود کو اس قدر وقف کر رکھا تھا کہ مسجد اور تعلیمی مرکز کی تعمیر مکمل ہونے تک وہ مسلسل روزے رکھتی رہیں۔
تعمیر مکمل ہونے کے بعد فاطمہ نے اپنے آبائی شہر قیروان کی نسبت سے اس مسجد اور مدرسے کے مجموعے کا نام القرویین رکھا۔
مختلف تاریخی روایات کے مطابق فاس کے جس علاقے میں فاطمہ اور ان کا خاندان رہتا تھا، وہاں کے بیشتر رہائشی قیروان سے ہجرت کر کے آئے تھے۔
اسی وجہ سے فاس کے دوسرے باشندے اس علاقے کو القرویین کے نام سے جانتے تھے، جس کے معنی ہیں ’قیروان سے آنے والے لوگ‘۔ اسی مناسبت سے اس مسجد اور تعلیمی مرکز کا نام بھی القرویین رکھا گیا۔
ہزار سال سے زیادہ کا سفر
قدیم شہر فاس کے وسط میں واقع اس کمپلیکس میں ابتدا ہی سے ایک مسجد، ایک مدرسہ اور ایک کتب خانہ شامل رہے ہیں۔
اس کمپلیکس کے مرکز میں القرویین مسجد واقع ہے۔ یہاں مردوں کے ساتھ ساتھ خواتین کے لیے بھی نماز کی الگ جگہ مختص کی گئی تھی۔
مسجد کو دیکھیں تو اس کی چھت پر قائم ایک بڑا مینار نظر آتا ہے۔ اسی مینار سے ہر روز نماز سے پہلے اذان دی جاتی ہے۔
مسجد کے بیرونی حصے میں ایک کشادہ صحن ہے۔ وہاں سنگِ مرمر کے تین فوارے موجود ہیں، جو نمازیوں کے وضو کے لیے استعمال ہوتے ہیں۔
ابتدا میں اس مسجد کی لمبائی صرف 100 فٹ تھی، تاہم بعد میں اس میں توسیع کی گئی۔
956 میں مسجد کی پہلی بڑی توسیع اور مرمت کا کام کیا گیا۔ اس وقت قرطبہ کے حکمران عبدالرحمٰن سوم کے دور میں مسجد کو مزید وسیع کیا گیا۔
اس دوران مسجد کے سامنے موجود صحن کو وسعت دی گئی جبکہ سفید رنگ کے مینار کو منتقل کر کے مسجد کی شمالی دیوار کے ساتھ تعمیر کیا گیا۔ اس کے بعد 12ویں اور 16ویں صدی میں بھی مسجد کی مزید توسیع کی گئی۔
مسلسل توسیع کے بعد آج مسجد کی لمبائی تقریباً 272 فٹ ہو چکی ہے، جس کی وجہ سے یہ افریقہ کی بڑی مساجد میں شمار ہوتی ہے۔ یہاں ایک وقت میں تقریباً 22 ہزار نمازی نماز ادا کر سکتے ہیں۔
مسجد کی چھت پر پانچ گنبد ہیں۔ ان میں مرکزی دروازے کے اوپر واقع گنبد کا اندرونی حصہ نہایت خوبصورت نقش و نگار سے مزین ہے۔ عمارت کے دروازوں اور بلند دیواروں پر بھی دلکش آرائش اور نفیس کاریگری دیکھی جا سکتی ہے۔
وسیع لائبریری
مسجد کے علاوہ جامعہ القرویین میں متعدد تعلیمی عمارتیں اور طلبہ کی رہائش کے لیے ہاسٹل بھی موجود ہیں۔ یہاں ایک وسیع و عریض کتب خانہ بھی قائم ہے۔
بتایا جاتا ہے کہ مرینی سلطنت کے دورِ حکومت میں 1359 میں اس کتب خانے کی مرمت اور توسیع کی گئی تھی۔
اس کتب خانے میں قرونِ وسطیٰ کی 20 ہزار سے زیادہ کتابیں محفوظ ہیں جن میں بیشتر ہاتھ سے لکھی گئی ہیں۔
ان میں تقریباً چار ہزار نایاب کتابیں اور مخطوطات شامل ہیں جو اسلامی تاریخ کے حوالے سے خاص اہمیت رکھتے ہیں۔
یہاں محفوظ اہم مخطوطات میں امام مالک بن انس کی مشہور حدیث کی کتاب الموطأ کے نسخے، ابن اسحاق کی تصنیف السیرۃ النبویہ اور ابن خلدون کی معروف کتابوں المقدمہ اور العبر کے اصل مخطوطات شامل ہیں۔
اس کے علاوہ کتب خانے میں قرآن مجید کا ایک قدیم مخطوطہ (ہاتھ سے لکھا گیا قدیم نسخہ) بھی محفوظ ہے جسے 1602 میں سلطان احمد المنصور نے عطیہ کیا تھا۔
2012 سے 2016 کے درمیان کتب خانے کی حالیہ مرمت کے دوران تاریخی مخطوطات کی بحالی کے لیے ایک جدید تحقیقی مرکز بھی قائم کیا گیا۔
کتب خانے میں 16ویں صدی کا ایک محفوظ ذخیرہ خانہ بھی موجود ہے، جس کا دروازہ تانبے کا بنا ہوا ہے۔ اس دروازے پر چار تالے لگے ہیں اور اسے کھولنے کے لیے چار مختلف چابیاں درکار ہوتی ہیں۔
یہ نایاب کتابیں اور مخطوطات اسی محفوظ ذخیرہ خانے میں رکھے گئے ہیں۔
اسناد جاری کرنے کا نظام
فاطمہ کی قائم کردہ القرویین کی بدولت فاس آہستہ آہستہ افریقہ میں اسلامی تعلیم کے ایک اہم مرکز کے طور پر ابھرنے لگا۔
یہاں مردوں کے ساتھ ساتھ خواتین بھی تعلیم حاصل کر سکتی تھیں تاہم ان کے لیے تعلیم کی الگ جگہ مختص تھی۔ کہا جاتا ہے کہ فاطمہ خود بھی اس ادارے میں زیرِ تعلیم رہی تھیں۔
ابتدائی دور میں یہاں تعلیم کا مرکز اسلامی علوم تھے لیکن وقت کے ساتھ اس کے نصاب میں لسانیات، صرف و نحو، قانون، ریاضی، موسیقی، فلسفہ، تصوف، طب اور علمِ نجوم جیسے مضامین بھی شامل ہو گئے۔
صرف مسلمان ہی نہیں بلکہ قرونِ وسطیٰ کے متعدد معروف یہودی اور مسیحی علما نے بھی اس ادارے سے تعلیم حاصل کی۔
اس ادارے کے طلبہ میں پوپ سِلویسٹر دوم بھی شامل تھے۔ کہا جاتا ہے کہ انھوں نے یہاں عربی ہندسوں کے نظام کے بارے میں علم حاصل کیا، جسے بعد میں یورپ میں متعارف کرانے میں انھوں نے کردار ادا کیا۔
اس ادارے میں مختلف ادوار میں تدریسی خدمات انجام دینے والی شخصیات میں معروف مالکی فقیہ ابن العربی، مؤرخ ابن خلدون اور ماہرِ فلکیات نورالدین البطروجی شامل ہیں۔
مراکش کے مؤرخ محمد المنونی کے مطابق تقریباً 1040 سے 1147 کے درمیان یہ مدرسہ ایک باضابطہ اعلیٰ تعلیمی ادارے میں تبدیل ہو چکا تھا، جہاں طلبہ کو ان کی اہلیت کی بنیاد پر اسناد دی جاتی تھیں۔
بیشتر مؤرخین اور محققین اس بات پر متفق ہیں کہ القرویین دنیا کا پہلا ادارہ تھا جہاں باقاعدہ طور پر اسناد جاری کی جاتی تھیں۔
1963 میں مراکش نے ایک شاہی فرمان کے ذریعے اسے جدید جامعہ کی شکل دینے کا عمل شروع کیا۔ 1965 میں اس کے نام کے ساتھ باضابطہ طور پر یونیورسٹی کا لفظ شامل کیا گیا اور یوں یہ جامعہ القرویین کہلانے لگی۔
