انتباہ: اس تحریر میں ریپ سے متعلق تفصیلات شامل ہیں، جو بعض قارئین کے لیے پریشان کن ہو سکتی ہیں۔
امریکہ کی کارنیل یونیورسٹی میں پیش آنے والے ریپ کے ایک مبینہ واقعے کے دو سال بعد تفتیش کا عمل تنقید کی زد میں آ گیا ہے۔
ریاست نیو یارک کی گورنر کیتھی ہوکل نے کہا ہے کہ ریپ کا الزام لگانے والی خاتون کو تحفظ فراہم کرنے کے ذمہ دار افراد اور ادارے ہر مرحلے پر اپنی ذمہ داری نبھانے میں ناکام رہے۔
جمعے کے روز پریس کانفرنس کرتے ہوئے گورنر نے کیس کی تفتیش پر ’گہری تشویش‘ کا اظہار کیا اور تحقیقات کی نگرانی پر مامور مقامی استغاثہ پر ’عدم اعتماد‘ کا اظہار کرتے ہوئے نیویارک کی اٹارنی جنرل لیٹیشیا جیمز کو اس کیس کی سربراہی سونپ دی ہے۔
مبینہ ریپ کا یہ واقعہ 2024 میں پیش آیا تھا، لیکن اس کی تحقیقات دوبارہ شروع کی گئی ہیں اور پولیس، مقامی استغاثہ اور کارنیل یونیورسٹی کے کردار پر سوالات اٹھائے گئے ہیں۔
پریس کانفرنس میں نیو یارک کی گورنر نے اس بات پر برہمی کا اظہار کیا کہ جب خاتون نے ریپ کیے جانے کی شکایت درج کرائی تھی تو مقامی حکام نے خود ان کا انٹرویو کیوں نہ کیا۔
انھوں نے کہا: ’اس نوجوان خاتون نے پہلے ہی ایک ایسے تجربے سے گزرنے کا دعویٰ کیا ہے جسے الفاظ میں بیان کرنا مشکل ہے، اور پھر ہر مرحلے پر وہ ان افراد اور اداروں کی ناکامی کا شکار ہوئیں جن پر ان کے تحفظ کی ذمہ داری عائد ہوتی تھی۔‘
ہوکل نے اس قانونی ’سقم‘ کو ختم کرنے کا بھی عزم ظاہر کیا ہے جو ایسے مقدمات میں استغاثہ کی جانب سے مقدمہ چلانے میں رکاوٹ بنتا ہے جہاں شکایت کنندہ نے اپنی مرضی سے منشیات یا شراب استعمال کی ہو۔
End of سب سے زیادہ پڑھی جانے والی
16 ستمبر کو ایک خاتون نے کارنیل یونیورسٹی کے خلاف مقدمہ دائر کیا، جس میں الزام عائد کیا گیا کہ یونیورسٹی میں چی فی تنظیم (فریٹرنٹی) سے منسلک ایک گھر میں انھیں منشیات دے کر مدہوش کرنے کے بعد ان کا ریپ کیا گیا تھا۔
انھوں نے یہ الزام بھی لگایا کہ یونیورسٹی انھیں تحفظ فراہم کرنے اور متعلقہ افراد کے خلاف مناسب کارروائی کرنے میں ناکام رہی۔ مقدمے کی دستاویزات میں خاتون کی شناخت خفیہ رکھی گئی ہے اور انھیں جین ڈوؤ کہا گیا ہے۔ جبکہ ملزمان کو ’کارنیل سیون‘ کا نام دیا گیا ہے۔
گورنر کیتھی ہوکل نے کہا: ’اگر کوئی شخص نشے کی ایسی کیفیت میں ہو کہ (سیکس کے لیے) رضا مندی دینے کے قابل نہ رہے تو اس سے یہ فرق نہیں پڑنا چاہیے کہ اس نے منشیات یا شراب اپنی مرضی سے استعمال کی تھی یا نہیں۔‘
انھوں نے مزید کہا: ’اپنی مرضی سے نشہ کرنا کسی کو گینگ ریپ کا جواز فراہم نہیں کرتا۔‘
مقدمے کے بارے میں ہم کیا جانتے ہیں؟
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
کارنیل یونیورسٹی کے خلاف دائر مقدمے میں خاتون نے الزام لگایا ہے کہ 19 اکتوبر 2024 کی رات چی فی تنظیم سے تعلق رکھنے والے یونیورسٹی کے سات افراد نے ایک ایسے واقعے میں کئی گھنٹے تک انھیں تشدد کا نشانہ بنایا، جس میں شراب اور کیٹامائن منشیات کا استعمال بھی شامل تھا۔
اُس وقت ان کی عمر 20 سال تھی اور وہ انڈرگریجویٹ طالبہ تھیں۔
عدالتی دستاویزات میں ’جین ڈوؤ‘ کے نام سے شناخت کی جانے والی خاتون نے اپنے مقدمے میں کہا ہے کہ مبینہ واقعے کے وقت وہ نشے کی حالت میں تھیں اور رضا مندی ظاہر کرنے کے قابل نہیں تھیں۔
ان کے مطابق، اُس رات شراب نوشی کے بعد وہ اپنی ساتھیوں سے الگ ہو گئیں اور ایک مرد دوست سے ملنے کے لیے چی فی تنظیم سے منسلک ایک گھر (فریٹرنٹی ہاؤس) تک اکیلے پہنچیں۔
مقدمے میں مختلف الزامات عائد کیے گئے ہیں جن میں جنسی تشدد، مار پیٹ، غفلت، معاہدے کی خلاف ورزی اور نیو یارک کے انسانی حقوق کے قانون کی خلاف ورزیاں شامل ہیں۔
یہ مقدمہ یونیورسٹی، متعلقہ تنظیم اور مبینہ واقعے میں نامزد افراد کے خلاف دائر کیا گیا ہے۔
مقدمے میں ایک ایسی تصویر بھی شامل ہے، جس کے بارے میں بتایا گیا ہے کہ تنظیم کی ایک گروپ چیٹ سے لی گئی ہے۔ مقدمے کے مطابق اس چیٹ میں شریک فرد نے نا مناسب زبان استعمال کرتے ہوئے دیگر اراکین کو جین ڈوؤ کے ساتھ ممکنہ جنسی ملاقات کے بارے میں آگاہ کیا تھا۔
خاتون کے وکیل کا کہنا ہے کہ واقعے کے بعد کے دنوں میں وہ ’شدید ذہنی صدمے کا شکار تھیں اور جو کچھ ان کے ساتھ ہوا اسے قبول کرنے یا معمول کی زندگی جاری رکھنے کے قابل نہیں تھیں۔‘
وکیل کے مطابق انھوں نے 8 نومبر 2024 کو اس معاملے کی اطلاع یونیورسٹی کو دی تھی۔
حالیہ ہفتوں میں اس مقدمے پر عوامی ردعمل میں اس وقت شدت آئی جب جین ڈوؤ نے اپنے دعوے میں الزام عائد کیا کہ یونیورسٹی نے بعض افراد کو اپنے طرزِ عمل کی سنگینی کم ظاہر کرنے کے لیے مضامین جمع کرانے کا موقع دیا۔
خاتون کے وکیل کا کہنا ہے کہ اس معاملے میں نامزد دو افراد کو یونیورسٹی سے نکال دیا گیا تھا۔ انھوں نے بتایا کہ مبینہ واقعے کے بعد خاتون نے بھی یونیورسٹی چھوڑ دی تھی۔
21 ستمبر کو کارنیل اخبار کے ادارتی بورڈ نے مقدمے کے نامزد ملزمان کے نام شائع کر دیے، جس کے بعد سوشل میڈیا پر ردعمل اور بحث میں مزید اضافہ ہوا۔
چی فی تنظیم کے سنیپ چیٹ اور ٹیکسٹ پیغامات میں کیا تھا؟
کارنیل یونیورسٹی کے محکمہ پولیس نے نومبر 2024 میں استغاثہ کو ایک سنیپ چیٹ گروپ کی تصویر بطور ثبوت فراہم کی۔ اس گروپ کا نام چی فی ایکٹیوز بتایا گیا تھا۔
چیٹ میں شامل افراد کے نام موجود ہیں اور وہ نا شائستہ زبان استعمال کرتے ہوئے جین ڈوؤ کے ساتھ ’مفت‘ جنسی ملاقات پر گفتگو کر رہے ہیں۔
بی بی سی کے امریکہ میں شراکت دار ادارے سی بی ایس نیوز نیو یارک نے ایسے اضافی پیغامات کا متن بھی شائع کیا جن کے بارے میں کہا گیا کہ وہ مقدمے میں نامزد ایک شخص اور الزام عائد کرنے والی خاتون نے ایک دوسرے کو بھیجے۔
اطلاعات کے مطابق یہ پیغامات مبینہ واقعے کے دو دن بعد بھیجے گئے تھے۔
ایک پیغام میں ایک شخص ’جو کچھ ہوا‘ اس پر معذرت کا اظہار کرتے دکھائی دیتے ہیں۔
وہ کہتے ہیں کہ ان کی یادداشت دھندلا رہی تھی، اور یہ کہ وہ اور ایک اور شخص اس بات پر افسوس محسوس کر رہے تھے کہ نشے کی وجہ سے صورت حال بے قابو ہونے سے روکنے اور دیگر افراد کو وہاں سے جانے کے لیے کہنے کے قابل نہیں رہے۔
پیغام کے متن کے مطابق خاتون جواب میں کہتی ہیں کہ ان کی یادداشت بھی بہت دھندلی تھی اور اس وقت تمام افراد نشے کی حالت میں تھے۔
انھوں نے یہ بھی لکھا کہ ’جنسی نوعیت کا کوئی بھی معاملہ غیر قانونی نہیں تھا۔‘
خاتون نے ٹیکسٹ پیغام میں یہ بھی لکھا کہ انھیں اس شخص اور ایک اور شخص کے ساتھ موجود رہنا اچھا لگا۔
ڈوؤ نے یہ تجویز بھی دی کہ وہ دوبارہ مل کر سموک کریں، تاہم اس بار ’دیگر افراد اور منشیات کو اس میں شامل نہ کیا جائے۔‘
کارنیل یونیورسٹی کا ردِعمل کیا تھا؟
خاتون کا کہنا ہے کہ واقعے کے تقریباً دو ہفتے بعد انھوں نے کارنیل یونیورسٹی پولیس کو اس کی اطلاع دی۔
اس کے کچھ ہی عرصے بعد کارنیل یونیورسٹی نے ’چی فی فریٹرنٹی ہاؤس میں منشیات کے استعمال اور جنسی تشدد سے متعلق سنگین اور نہایت تشویش ناک الزامات‘ کے بارے میں ایک عوامی بیان جاری کیا۔
یونیورسٹی نے کہا کہ کیمپس پولیس اس معاملے کی تحقیقات کر رہی ہے اور چی فی فریٹرنٹی کی مقامی شاخ کے ساتھ ساتھ متعدد طلبہ کو بھی معطل کر دیا گیا ہے۔
کارنیل نے یہ بھی بتایا کہ اس نے داخلی سطح پر تحقیقات شروع کر دی ہیں۔
ڈوؤ کے مقدمے کے بعد یونیورسٹی نے اس الزام کو مسترد کیا کہ اس نے معاملے کی مناسب تحقیقات نہیں کیں یا مبینہ طور پر ملوث افراد کے خلاف مناسب کارروائی نہیں کی۔
یونیورسٹی کے مطابق یہ معاملہ تربیت یافتہ اساتذہ اور عملے پر مشتمل ایک پینل کے سپرد کیا گیا تھا، جس نے مختلف تادیبی سزائیں تجویز کیں جن میں یونیورسٹی سے اخراج اور معطلی بھی شامل تھی۔
کارنیل نے اس بات کی تردید نہیں کی کہ بعض افراد کے لیے تادیبی کارروائیوں میں مضمون لکھنے کی شرط بھی شامل تھی، تاہم اس کا کہنا تھا کہ کسی بھی ملزم کو صرف مضمون لکھنے کو ہی واحد سزا کے طور پر اختیار کرنے کا موقع نہیں دیا گیا۔
یونیورسٹی نے ایک بیان میں کہا کہ جب اس طرزِ عمل کی اطلاع ملی تو اس وقت عارضی معطلیوں اور دیگر پابندیوں پر مبنی اقدامات نافذ کیے گئے تھے۔
کارنیل کا یہ بھی کہنا تھا کہ 2024 میں اس کے کیمپس میں پیش آنے والے واقعات اُس وقت بھی تشویش ناک تھے اور آج بھی انھیں اسی نظر سے دیکھا جاتا ہے۔
بیان میں کہا گیا کہ یونیورسٹی ’ایسے ماحول کو فروغ دینے کی کوشش کر رہی ہے جہاں نا منساب جنسی رویہ کبھی بھی برداشت نہ کیا جائے۔‘
چی فی تنظیم کا مؤقف کیا ہے؟
جین ڈوؤ نے اپنے مقدمے میں چی فی فریٹرنٹی (یا تنظیم) کو بھی فریق بنایا ہے۔ ان کا الزام ہے کہ تنظیم میں شامل افراد کے اقدامات کے نتیجے میں انھیں جسمانی اور نفسیاتی نقصان پہنچا۔
چی فی تنظیم کے عبوری ایگزیکٹو ڈائریکٹر ٹرے روب نے ایک بیان میں کہا کہ تنظیم نے کارنیل یونیورسٹی کی مقامی شاخ کے ان اراکین کو خارج کر دیا ہے جن پر الزامات عائد کیے گئے تھے۔
روب نے کہا: ’ہمارے ریکارڈ سے تصدیق ہوتی ہے کہ جین ڈوؤ کی جانب سے پولیس رپورٹ درج کرانے کے صرف تین دن بعد تنظیم کی مقامی شاخ نے نامزد ملزمان کو اپنی رکنیت سے خارج کر دیا تھا۔
’مقامی شاخ نے یہ سفارش بھی کی تھی کہ ہماری قومی سطح کی تنظیم انھیں مستقل طور پر خارج کر دے، اور بعد کے ایک اجلاس میں ہماری گرینڈ کونسل نے ایسا ہی کیا۔‘
روب کا کہنا تھا کہ تنظیم عدالتی کارروائی یا زیرِ تفتیش معاملات پر تبصرہ نہیں کرے گی۔
تحقیقات دوبارہ کیوں شروع کی گئیں؟
کاؤنٹی کے استغاثہ کے اعلیٰ عہدے دار میتھیو وان ہاؤٹن نے کہا تھا کہ ان کا دفتر اس معاملے کی تحقیقات دوبارہ شروع کرے گا۔ ٹومپکنز کاؤنٹی کے ڈسٹرکٹ اٹارنی میتھیو وان ہاؤٹن کے مطابق، الزامات کے تعین کے بعد کیس گرینڈ جیوری کے سامنے پیش کیا جانا تھا اور قانونی کارروائی ممکنہ حد تک غیر جانب دار اور معروضی انداز میں آگے بڑھائی جانی تھی۔
تاہم جمعے کو ریاست نیو یارک کی گورنر کیتھی ہوکل نے اعلان کیا کہ انھیں مقامی استغاثہ کی جانب سے تحقیقات کی نگرانی پر اعتماد نہیں رہا، جس کے بعد انھوں نے ریاست کی اٹارنی جنرل لیٹیشیا جیمز کو خصوصی پراسیکیوٹر مقرر کر دیا۔ اب جیمز کا دفتر اس معاملے کی تحقیقات کی قیادت کرے گا اور اگر ضرورت پڑی تو قانونی کارروائی بھی شروع کی جا سکتی ہے۔
ٹومپکنز کاؤنٹی استغاثہ کے مطابق دو برس قبل جب یہ معاملہ پہلی مرتبہ ان کے سامنے آیا تھا تو دفتر نے یہ نتیجہ اخذ کیا تھا کہ یونیورسٹی حکام کو دیے گئے بیان کی بنیاد پر مقدمہ چلانے کے لیے مطلوب قانونی معیار پورا نہیں ہوتا۔
وان ہاؤٹن نے کہا کہ گرینڈ جیوری کے سامنے معاملہ پیش کرنے کی تیاری کے دوران انھوں نے مزید معلومات کے لیے جین ڈوؤ اور ان کے وکلا سے رابطہ کیا۔
انھوں نے اپنے بیان میں یہ بھی کہا کہ 2024 میں جین ڈوؤ کی جانب سے یونیورسٹی پولیس کو دی گئی معلومات اور حالیہ مقدمے میں شامل بعض الزامات میں فرق پایا جاتا ہے۔
وان ہاؤٹن کے مطابق ’جین ڈوؤ نے 2024 کے حلفیہ بیان میں یہ الزام نہیں لگایا تھا کہ انھیں مرضی کے خلاف نشہ دیا گیا یا گینک ریپ کیا گیا۔
’اس کے برعکس، جین ڈوؤ کے بیان میں کہا گیا تھا کہ منشیات کے استعمال اور جنسی نوعیت کے عمل میں ان کی شرکت رضا کارانہ، شعوری اور باہمی رضا مندی پر مبنی تھی۔‘
کارنیل یونیورسٹی کے ترجمان نے بی بی سی کو بتایا کہ یونیورسٹی پولیس نے اپنی تفصیلی تحقیقاتی رپورٹ ٹومپکنز کاؤنٹی ڈسٹرکٹ اٹارنی کے دفتر کو فراہم کی تھی۔
حالیہ رپورٹس میں اس بات کی مزید تفصیلات سامنے آئی ہیں کہ مبینہ واقعے کے بعد جین ڈوؤ نے یونیورسٹی پولیس کو کیا بتایا تھا۔
سی بی ایس نیوز نیو یارک نے انٹرویوز کا ایک متن حاصل کیا، جو 14 نومبر کو شروع ہوئے تھے، یعنی واقعے کے تین ہفتوں سے زیادہ عرصے بعد۔
جنسی نوعیت کے واقعے کے بارے میں ابتدائی طور پر ڈوؤ نے پولیس کو بتایا: ’اس وقت مجھے اس سب پر کوئی خاص اعتراض نہیں تھا۔‘
تاہم، سی بی ایس نیوز کے مطابق، بعد میں خاتون نے کہا کہ ’میں 100 فیصد یقین کے ساتھ کہہ سکتی ہوں کہ میرا ریپ کیا گیا تھا۔‘
اے بی سی نیوز کی جانب سے حاصل کی گئی 15 نومبر کی ایک پولیس رپورٹ کے مطابق، جین ڈوؤ نے اپنے بیان میں کہا کہ ایک مرحلے میں وہ نشے کی کیفیت کے باعث خود کو فیصلے کرنے کے قابل محسوس نہیں کر رہی تھیں۔
انھوں نے اپنے بیان میں کہا: ’میرے لیے یہ دباؤ ڈالنے یا مجبور کیے جانے کے مترادف تھا۔ مجھے یہ بھی محسوس ہوا کہ میری نشے کی کیفیت کی وجہ سے انھیں اندازہ ہو جانا چاہیے تھا کہ میں کس حد تک مدہوش تھی، کیوں کہ گفتگو کے دوران میری زبان لڑکھڑا رہی تھی۔‘
وان ہاؤٹن نے سی بی ایس نیوز کو بتایا کہ انھوں نے کیمپس پولیس کے انٹرویو کا متن پہلے نہیں دیکھا تھا۔
کیمپس میں کیا ردِعمل سامنے آیا؟
ستمبر میں جین ڈو کی جانب سے مقدمہ دائر کیے جانے کے بعد سے کارنیل یونیورسٹی کے کیمپس میں نمایاں کشیدگی محسوس کی جا رہی ہے۔
بی بی سی نے متعدد طلبہ اور ایک استاد سے بات کی، جن میں سے کئی نے نام ظاہر نہ کرنے کی درخواست کی۔ ان کا کہنا تھا کہ یونیورسٹی پر ایک ’سیاہ بادل‘ چھایا ہوا محسوس ہوتا ہے۔
2024 میں کیمپس میں موجود ایک سینیئر طالبہ نے بی بی سی کو بتایا کہ اس وقت حالات مشکل تھے کیونکہ جین ڈوؤ کے بارے میں افواہیں گردش کرنے لگی تھیں۔
ان کے مطابق دو سال بعد بھی وہ اور ان کے دوست خود کو شدید دباؤ اور مایوسی کا شکار محسوس کرتے ہیں کیونکہ ان کے خیال میں یونیورسٹی اس معاملے کو مناسب انداز میں نہیں سنبھال سکی۔ انھوں نے کہا کہ اس صورتِ حال کے باعث پڑھائی پر توجہ مرکوز کرنا بھی مشکل ہو گیا تھا۔
ایک استاد نے بھی کیمپس کی فضا کو مایوس کن قرار دیا۔
ان کے بقول تدریسی عملہ تربیت یافتہ تھراپسٹ نہیں ہوتا، اس لیے طلبہ کے ساتھ ایسے حساس موضوعات پر گفتگو کرنا آسان نہیں تھا۔
انھوں نے کہا: ’یہ ایک غیر معمولی صورت حال ہے۔ مجھے اب بھی سمجھ نہیں آتا کہ یہ سب کیسے ہوا۔‘
