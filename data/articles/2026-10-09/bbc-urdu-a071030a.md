# مکہ معاہدے کے بعد کیا سعودی عرب امریکہ کے ساتھ دفاعی شراکت داری ختم کرنے جا رہا ہے؟

مکہ معاہدے کے بعد کیا سعودی عرب امریکہ کے ساتھ دفاعی شراکت داری ختم کرنے جا رہا ہے؟

سعودی عرب کی جانب سے اپنی سکیورٹی شراکت داریوں میں تنوع پیدا کرنے کا تصور اب محض ایک بحث نہیں رہا بلکہ عملی شکل اختیار کرتا دکھائی دے رہا ہے۔ 5 اکتوبر کو سعودی عرب، ترکی اور پاکستان نے اعلان کیا کہ وہ ’مکہ معاہدے‘ کے تحت کیے گئے اجتماعی دفاعی وعدوں پر فوری عمل درآمد شروع کریں گے اور اس مقصد کے لیے مشترکہ فوجی صلاحیتوں اور دستوں کی فراہمی اور ضرورت پڑنے پر ان کی سعودی عرب میں فوری تعیناتی کے اقدامات بھی کیے جائیں گے۔ یہ فیصلہ یمن میں حوثی تحریک انصار اللہ کی جانب سے سعودی عرب پر مسلسل حملوں کے تناظر میں سامنے آیا۔
یہ اعلان اتحاد کی سیاسی اور دفاعی تزویراتی کمیٹی کے ہنگامی اجلاس کے بعد کیا گیا، جو رواں سال اگست میں مکہ مکرمہ میں قائم کیے گئے اتحاد کے تحت منعقد ہوا۔ اجلاس میں سعودی عرب کو درپیش سکیورٹی خطرات کا جائزہ لیا گیا، ایسے وقت میں جب یمن کی جنگ دوبارہ شدت اختیار کر رہی ہے اور ایران کی حمایت یافتہ حوثی تحریک سعودی سرزمین کو نشانہ بنا رہی ہے۔
اس سے قبل بعض میڈیا رپورٹس میں دعویٰ کیا گیا تھا کہ امریکی صدر ڈونلڈ ٹرمپ نے حوثیوں کے خلاف براہ راست فوجی کارروائی کے لیے سعودی عرب کی متعدد درخواستیں مسترد کر دی تھیں۔ ان رپورٹس کے مطابق واشنگٹن نے سعودی عرب کے ساتھ مل کر حوثیوں پر براہ راست حملوں کی منظوری نہیں دی، البتہ اہداف کے تعین اور آپریشنل منصوبہ بندی کے لیے انٹیلی جنس اور معلوماتی معاونت فراہم کرتا رہا۔
ایسے میں اہم سوال یہ ہے کہ کیا یہ پیش رفت سعودی عرب کو اپنی سلامتی کے لیے امریکہ پر انحصار کم کرنے کی جانب لے جائے گی؟ اور کیا نام نہاد ’مکہ اتحاد‘ وہ سکیورٹی ضمانتیں فراہم کر سکتا ہے جن کے لیے ریاض طویل عرصے سے واشنگٹن پر انحصار کرتا آیا ہے؟ یہ ایسے سوالات ہیں جن کے جواب مشرقِ وسطیٰ کی بدلتی ہوئی تزویراتی سیاست اور اس نئے دفاعی انتظام کے عملی نتائج سے وابستہ ہیں۔
امریکی حمایت کی حدود
اقوامِ متحدہ کی نگرانی میں 2022 میں طے پانے والی جنگ بندی کے بعد، جس نے یمن میں چار سال سے زائد عرصے تک نسبتاً سکون برقرار رکھا، اب وہاں ایک بار پھر جنگ کی شدت میں نمایاں اضافہ دیکھا جا رہا ہے۔
صورتِ حال میں اہم تبدیلی جولائی میں آئی، جب سعودی حمایت یافتہ یمنی سرکاری افواج نے حوثیوں کے زیرِ کنٹرول صنعا ایئرپورٹ پر بمباری کی۔ حکومتی فورسز کا کہنا تھا کہ کارروائی کا مقصد ایک ایرانی طیارے کو دارالحکومت میں اترنے سے روکنا تھا۔
اس حملے کے چند ہی گھنٹوں بعد حوثیوں نے جنوبی سعودی عرب کی جانب میزائل داغ دیے۔ بعد ازاں بحران اس وقت مزید گہرا ہو گیا، جب حوثیوں نے یمن کے مغربی ساحلی علاقوں میں اپنی پوزیشن مضبوط کر لی، جس سے باب المندب کے قریب بحری جہاز رانی کو خطرے میں ڈالنے کی ان کی صلاحیت بڑھ گئی۔ باب المندب بحیرہ احمر کا جنوبی دروازہ ہے اور بحرِ ہند و بحیرہ روم کے درمیان تجارت کی اہم ترین گزرگاہوں میں شمار ہوتا ہے، جس سے نہرِ سویز کے راستے عالمی تجارت کا بڑا حصہ گزرتا ہے۔
پانچ اکتوبر کو سعودی حمایت یافتہ یمنی افواج نے حوثیوں کے زیرِ قبضہ علاقوں کا کنٹرول واپس لینے کے لیے ایک بڑے فوجی آپریشن کا آغاز کیا، جسے سعودی فضائیہ کی مدد بھی حاصل تھی۔
End of سب سے زیادہ پڑھی جانے والی
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
فاؤنڈیشن فار مڈل ایسٹ پیس کے سابق نائب صدر اور مشرقِ وسطیٰ کے امور کے ماہر مچل پلیٹنک کے مطابق حالیہ واقعات نے سعودی عرب اور امریکہ کے سکیورٹی تعلقات کو ایک نئے امتحان سے دوچار کر دیا ہے۔
ان کے بقول، ’اس معاملے میں امریکہ نے سعودی عرب کو مایوس کیا ہے۔‘
مچل پلیٹنک کا کہنا ہے کہ دوسری عالمی جنگ کے بعد سعودی عرب اور امریکہ کے تعلقات ایک غیر تحریری سکیورٹی مفاہمت پر استوار رہے، جس کے تحت سعودی عرب توانائی کی مسلسل فراہمی اور تیل کی عالمی منڈیوں کے استحکام میں کردار ادا کرتا رہا، جبکہ امریکہ اور مغربی ممالک مملکت کو سکیورٹی تحفظ فراہم کرتے رہے۔
تاہم ان کے مطابق ٹرمپ انتظامیہ کے پہلے دورِ اقتدار ہی سے یہ واضح ہونا شروع ہو گیا تھا کہ جب سعودی سلامتی کے تقاضے اور امریکی ترجیحات ایک دوسرے سے متصادم ہوں تو یہ شراکت داری مکمل اور غیر مشروط سکیورٹی ضمانت فراہم نہیں کرتی۔
وہ ستمبر 2019 میں سعودی آرامکو سمیت اہم تیل تنصیبات پر ہونے والے حملوں کی مثال دیتے ہیں، جن سے سعودی تیل کی پیداوار عارضی طور پر شدید متاثر ہوئی تھی۔ اگرچہ امریکہ اور سعودی عرب نے ان حملوں کا الزام ایران پر عائد کیا تھا، جس کی تہران نے تردید کی، تاہم واشنگٹن نے ایران کے خلاف کوئی براہِ راست فوجی کارروائی نہیں کی۔
مچل پلیٹنک کے نزدیک اس واقعے نے سعودی قیادت میں یہ تاثر مزید مضبوط کیا کہ امریکہ کے ساتھ سکیورٹی شراکت داری کا مطلب یہ نہیں کہ واشنگٹن ہر اس تنازع میں فوجی طور پر شریک ہوگا جسے ریاض اپنی قومی سلامتی کے لیے ضروری سمجھتا ہو۔
دوسری جانب امریکی محکمۂ خارجہ میں تقریباً 25 برس تک مشیر رہنے والے آرون ڈیوڈ ملر کا مؤقف کچھ مختلف ہے۔ ان کے مطابق ٹرمپ انتظامیہ کے پاس سعودی عرب کی مدد کرنے کے اسباب ضرور موجود تھے، لیکن وہ ایران کے ساتھ وسیع تر تصادم اور یمن میں ایک نئے فوجی محاذ کے آغاز سے گریز کرنا چاہتی تھی۔
آرون ڈیوڈ ملر کے مطابق یہ احتیاط خاص طور پر 2025 میں حوثیوں کے خلاف امریکی آپریشن ’راف رائیڈر‘ کے تجربے کے بعد اختیار کی گئی، جو 52 دن تک جاری رہا اور جس دوران ایک ہزار سے زیادہ حملے کیے گئے، مگر اس کے باوجود حوثیوں کی فوجی صلاحیتوں پر فیصلہ کن ضرب نہیں لگائی جا سکی۔
ان کے خیال میں کانگریس کے وسط مدتی انتخابات سے قبل امریکی داخلی سیاسی عوامل بھی ایسے تھے جنھوں نے واشنگٹن کو مزید فوجی کشیدگی سے دور رہنے پر آمادہ کیا۔
کارنیگی اینڈومنٹ فار انٹرنیشنل پیس کے مشرقِ وسطیٰ پروگرام کے ڈائریکٹر عمرو حمزاوی کے مطابق سعودی عرب کو براہِ راست فوجی مدد نہ دینا لازماً اس بات کا ثبوت نہیں کہ امریکہ خطے سے پیچھے ہٹ رہا ہے یا سعودی عرب کے ساتھ اپنے تعلقات کی نوعیت تبدیل کر رہا ہے۔
ان کے مطابق امریکہ نے فوجی مداخلت کے بجائے انٹیلی جنس تعاون کا راستہ اپنایا، جس میں حوثیوں کی صلاحیتوں اور اہداف سے متعلق معلومات کا تبادلہ شامل تھا۔ حمزاوی کے نزدیک یہ پالیسی امریکی فوجی کردار کی حدود کو ظاہر کرتی ہے، نہ کہ سعودی عرب کے لیے اس کی وابستگی میں کسی بنیادی تبدیلی کو۔
اسی دوران امریکی حکام کے بیانات بھی مملکت کے لیے مسلسل امریکی حمایت کے عزم کی عکاسی کرتے رہے۔ 23 ستمبر کو امریکی وزیرِ خارجہ مارکو روبیو نے کہا تھا کہ امریکہ اور سعودی عرب کے درمیان ’بہت مضبوط دفاعی تعلقات‘ موجود ہیں اور واشنگٹن اپنی ذمہ داریوں سے پیچھے نہیں ہٹے گا۔
چند روز بعد سعودی وزیرِ دفاع شہزادہ خالد بن سلمان اور ان کے امریکی ہم منصب پیٹ ہیگستھ کے درمیان بھی رابطہ ہوا، جس میں دونوں ممالک کے درمیان تزویراتی دفاعی شراکت داری اور خطے میں امن و استحکام کے فروغ کے لیے تعاون پر تبادلۂ خیال کیا گیا۔
سعودی وزارتِ خارجہ کے سابق مشیر سالم الیامی بھی ’امریکی دستبرداری‘ کے تصور سے اتفاق نہیں کرتے۔ ان کے مطابق امریکہ خود بھی سعودی عرب جیسے اہم اتحادی سے دوری کا متحمل نہیں ہو سکتا، کیونکہ اس سے اس کے اپنے سیاسی، اقتصادی اور تزویراتی مفادات متاثر ہوں گے۔
سالم الیامی کا کہنا ہے کہ حالیہ اختلافات نے اگرچہ بعض عملی اور وقتی معاملات پر اثر ڈالا ہے، لیکن اس سے سعودی عرب اور امریکہ کے درمیان کئی دہائیوں پر محیط تزویراتی شراکت داری کی بنیاد متزلزل نہیں ہوئی۔
تاہم وہ اس بات کی نشاندہی بھی کرتے ہیں کہ سعودی عرب 2015 سے ہی اپنی خارجہ اور دفاعی پالیسی میں تنوع لانے کی کوشش کر رہا ہے۔ چنانچہ موجودہ حالات کو واشنگٹن سے مکمل دوری کے بجائے ریاض کی اس وسیع حکمتِ عملی کے تناظر میں دیکھنا زیادہ مناسب ہوگا، جس کا مقصد مختلف عالمی اور علاقائی شراکت داروں کے ساتھ متوازن تعلقات استوار کرنا ہے۔
مکہ اتحاد: امریکہ کا متبادل یا ایک اضافی حفاظتی تہہ؟
سالم الیامی کے مطابق اگست میں مکہ میں سعودی عرب، ترکی اور پاکستان کے درمیان طے پانے والا دفاعی معاہدہ ریاض کی سکیورٹی شراکت داریوں میں تنوع پیدا کرنے کی پالیسی کی نمایاں مثال ہے۔
معاہدے کے تحت تینوں ممالک اس بات پر متفق ہوئے کہ اگر کسی ایک رکن ملک پر مسلح حملہ ہوتا ہے تو اسے تمام اراکین کے خلاف حملہ تصور کیا جائے گا، اور اس صورت میں مشترکہ دفاعی تعاون کو بروئے کار لایا جائے گا۔
تاہم معاہدے پر دستخط کے بعد سے ایک اہم سوال مسلسل زیرِ بحث رہا: اسے عملی شکل کس طرح دی جائے گی؟ یہ سوال اس لیے بھی اٹھا کہ سعودی عرب پر حوثی حملے جاری رہنے کے باوجود ترکی اور پاکستان کی جانب سے کسی براہ راست فوجی مداخلت کا اعلان سامنے نہیں آیا۔
ستمبر میں پاکستان نے واضح کیا تھا کہ حوثی حملوں کے جواب میں معاہدے کے تحت فوری فوجی اقدامات پر غور نہیں کیا گیا، اگرچہ اسلام آباد نے اس دفاعی انتظام سے اپنی وابستگی کا اعادہ کیا تھا۔
تاہم 5 اکتوبر کا اجلاس اس حوالے سے ایک اہم پیش رفت ثابت ہوا۔ اجلاس کے بعد تینوں ممالک نے مشترکہ دفاعی وعدوں پر ’فوری‘ عمل درآمد، متفقہ فوجی قوتوں اور صلاحیتوں کی فراہمی، اور ضرورت پڑنے پر ان کی سعودی عرب میں تیز رفتار تعیناتی کے لیے اقدامات کرنے کا اعلان کیا۔
اجلاس میں ایک اور اہم فیصلہ یہ بھی کیا گیا کہ پاکستان کے ریٹائرڈ جنرل نعمان محمود کو تین برس کے لیے اتحاد کا پہلا سیکریٹری جنرل مقرر کیا جائے۔ مبصرین کے مطابق یہ قدم اتحاد کو باقاعدہ ادارہ جاتی شکل دینے کی جانب پیش رفت کی نشاندہی کرتا ہے۔
آرون ڈیوڈ ملر کے مطابق اس اتحاد کو معمولی نہیں سمجھا جا سکتا، کیونکہ اس میں دنیا کے بڑے تیل پیدا کرنے والے ممالک میں شامل سعودی عرب، جوہری طاقت پاکستان اور نیٹو کے اہم رکن ترکی شامل ہیں۔
تاہم ان کے خیال میں اس کا یہ مطلب ہرگز نہیں کہ تینوں ممالک سعودی عرب کے دفاع کے لیے ایران کے ساتھ کھلی محاذ آرائی پر آمادہ ہیں، یا یہ اتحاد فی الحال وہ فوجی، انٹیلی جنس اور فضائی دفاعی صلاحیتیں فراہم کرنے کی پوزیشن میں ہے جو امریکہ کئی دہائیوں سے سعودی عرب کو مہیا کرتا آیا ہے۔
حمزاوی کے مطابق ’مکہ اتحاد‘ دراصل سعودی عرب کی سلامتی کی حکمتِ عملی میں ہونے والی تبدیلی کی عکاسی کرتا ہے۔ ان کے نزدیک ریاض ایک ساتھ دو راستوں پر چل رہا ہے: ایک جانب محدود تعداد میں ممالک کے ساتھ دو طرفہ اور سہ فریقی دفاعی شراکت داریاں، اور دوسری جانب وسیع تر اجتماعی سلامتی کے انتظامات۔
ان کے بقول سعودی عرب ان دونوں راستوں کو بیک وقت آگے بڑھا رہا ہے۔ وہ اس اتحاد کو بنیادی طور پر ایک دفاعی بندوبست قرار دیتے ہیں، جس کا مقصد سعودی سرزمین، سرحدوں اور اہم تنصیبات کا تحفظ ہے، نہ کہ یمن کی جنگ میں براہِ راست شمولیت۔
حمزاوی یاد دلاتے ہیں کہ سعودی عرب میں پاکستانی فوجی عملہ پہلے ہی مختلف ذمہ داریاں انجام دے رہا ہے، لیکن اس حقیقت سے یہ نتیجہ اخذ نہیں کیا جا سکتا کہ پاکستان یا ترکی ایران کے ساتھ براہِ راست فوجی تصادم یا یمن کی جنگ میں شرکت کے لیے تیار ہیں۔
پلیٹنک بھی اسی مؤقف سے اتفاق کرتے ہیں۔ ان کے مطابق معاہدہ ابھی ابتدائی مرحلے میں ہے اور اس کے عملی خدوخال مکمل طور پر واضح نہیں ہوئے۔ ان کا کہنا ہے کہ سعودی عرب، ترکی اور پاکستان تینوں بخوبی جانتے ہیں کہ یہ اتحاد کیا کر سکتا ہے اور اس کی حدود کیا ہیں، اسی لیے اس کا بنیادی مقصد سعودی سرزمین، سرحدوں اور اہم تنصیبات کا دفاع ہے، نہ کہ یمن میں کسی وسیع فوجی مہم کا آغاز۔
پلیٹنک کے مطابق، اگر مستقبل میں یہ اتحاد مزید مضبوط اور مؤثر بھی ہو جائے، تب بھی اسے امریکہ کے مکمل متبادل کے طور پر دیکھنا قبل از وقت ہوگا۔ ان کے بقول ریاض کی نئی شراکت داریاں اس کے دفاعی اختیارات میں اضافہ ضرور کر سکتی ہیں، لیکن فی الحال وہ واشنگٹن کی فراہم کردہ سکیورٹی چھتری کا مکمل نعم البدل نہیں بن سکتیں۔
یاد رہے کہ ترکی کے وزیرِ خارجہ ہاکان فیدان نے وضاحت کی کہ اس معاہدے کے تحت ترکی پر لازم نہیں ہے کہ وہ اپنے فوجی دستے کسی دوسرے ملک میں جارحانہ کارروائیوں کے لیے تعینات کرے۔
اسرائیل: کیا وہ سعودی حساب کتاب کا حصہ بن سکتا ہے؟
امریکی ماہر مچل پلیٹنک کے بقول، ’آخرکار تمام راستے اسرائیل ہی کی طرف جاتے ہیں۔‘
ان کے مطابق، جیسے جیسے سعودی عرب کو یہ احساس ہوتا جا رہا ہے کہ امریکہ کی جانب سے فراہم کی جانے والی سکیورٹی چھتری پہلے جیسی قابلِ اعتماد نہیں رہی، ویسے ویسے واشنگٹن کی نظر میں اسرائیل کے ساتھ تعلقات کی اہمیت بڑھتی جا رہی ہے۔ اس کی ایک بڑی وجہ یہ ہے کہ سعودی عرب اور اسرائیل کے درمیان تعلقات کا معمول پر آنا مشرقِ وسطیٰ کے سکیورٹی ڈھانچے کی نئی تشکیل اور ایران کے مقابلے میں وسیع علاقائی شراکت داری کے قیام میں اہم کردار ادا کر سکتا ہے۔
یہ سوچ کوئی نئی نہیں۔ امریکی حکومتیں برسوں سے سعودی عرب کو اسرائیل کے ساتھ تعلقات معمول پر لانے پر آمادہ کرنے کی کوشش کرتی رہی ہیں۔ اس مقصد کے لیے مختلف اوقات میں ایسے پیکیج زیرِ غور رہے ہیں جن میں امریکی سکیورٹی ضمانتیں، پرامن جوہری تعاون اور سعودی عرب کے لیے دیگر تزویراتی و اقتصادی مراعات شامل تھیں۔
آرون ڈیوڈ ملر کے مطابق، سعودی ولی عہد محمد بن سلمان ہمیشہ سے یہ چاہتے رہے ہیں کہ مملکت امریکہ کے اہم ترین علاقائی شراکت داروں میں اپنی حیثیت مزید مضبوط کرے۔ تاہم وہ اسرائیل کے ساتھ تعلقات کی بحالی کو امریکہ سے بہتر تعلقات حاصل کرنے کی قیمت کے طور پر پیش کرنے سے گریز کرتے رہے ہیں۔
اسی پس منظر میں پرامن جوہری تعاون کا معاملہ بھی خاص اہمیت رکھتا ہے۔ واشنگٹن اور ریاض طویل عرصے سے ایسے قانونی فریم ورک پر کام کر رہے ہیں جو دونوں ممالک کے درمیان جوہری توانائی کے شعبے میں تعاون کی راہ ہموار کر سکے۔ تاہم اس تعاون کا مستقبل صرف توانائی کے شعبے تک محدود نہیں بلکہ سکیورٹی انتظامات اور اسرائیل سے متعلق وسیع تر مفاہمتوں سے بھی جڑا ہوا ہے۔
امریکی قوانین کے تحت اس نوعیت کے کسی بھی معاہدے کی کانگریس سے منظوری ضروری ہوگی، جس کے باعث داخلی سیاسی عوامل، خصوصاً انتخابات کے قریب، مزید اہمیت اختیار کر جاتے ہیں۔
ستمبر میں بعض میڈیا رپورٹس میں دعویٰ کیا گیا کہ جرمنی میں امریکہ، اسرائیل، سعودی عرب اور کئی عرب ممالک کے اعلیٰ فوجی عہدیداروں کا ایک خفیہ اجلاس ہوا، جس میں ایران کے ساتھ ممکنہ جنگ اور خطے کی سکیورٹی صورتحال پر تبادلہ خیال کیا گیا۔ تاہم اس اجلاس کی متعلقہ حکومتوں نے باضابطہ تصدیق نہیں کی۔
اسی طرح اسرائیلی میڈیا نے یہ بھی رپورٹ کیا کہ متحدہ عرب امارات میں اسرائیلی وزیرِ اعظم بنیامین نتن یاہو کے دورے کے دوران سعودی اور اسرائیلی حکام کے درمیان ملاقاتیں ہوئیں۔ بعد ازاں ایک سینیئر سعودی عہدیدار نے ان اطلاعات کی تردید کی، جبکہ اسرائیلی اخبار یدیعوت احرونوت نے بھی اپنی ابتدائی رپورٹ پر نظرِ ثانی کرتے ہوئے کہا کہ وہ اس دعوے کی تصدیق نہیں کر سکا۔
اسرائیل کے سابق نائب قومی سلامتی مشیر اور کولمبیا و تل ابیب یونیورسٹی کے پروفیسر چک فریلیخ کے مطابق، سعودی عرب کی جانب سے اپنی سکیورٹی شراکت داریوں میں تنوع پیدا کرنے کی کوششیں اسرائیل کے لیے زیادہ فعال کردار ادا کرنے کے امکانات پیدا کرتی ہیں، خصوصاً اس لیے کہ دونوں ممالک ایران کو اپنا مشترکہ حریف سمجھتے ہیں۔
چک فریلیخ کا کہنا ہے کہ امریکہ کے برعکس اسرائیل خطہ چھوڑ کر نہیں جا سکتا، کیونکہ ایران کے ساتھ اس کا تنازع اس کی قومی اور وجودی سلامتی سے براہِ راست جڑا ہوا ہے۔ اسی وجہ سے اسرائیل فضائی دفاع، میزائل شکن نظام اور دیگر سکیورٹی شعبوں میں سعودی عرب کا ممکنہ شراکت دار بن سکتا ہے۔
ان کے نزدیک سعودی عرب اور اسرائیل کے درمیان قربت کا عمل ضروری نہیں کہ کھلے سفارتی تعلقات سے ہی شروع ہو۔ سلامتی، انٹیلی جنس اور سیاسی روابط اکثر باضابطہ تعلقات کے اعلان سے پہلے بھی فروغ پا سکتے ہیں، جیسا کہ ماضی میں بعض دیگر عرب ممالک کے ساتھ ہوا۔
تاہم ماہر حمزاوی اس راستے میں ایک بڑی سیاسی رکاوٹ کی نشاندہی کرتے ہیں۔ ان کے مطابق سعودی عرب نے اسرائیل کے ساتھ تعلقات کی بحالی کو فلسطینی مسئلے کے حل میں پیش رفت سے مشروط کر رکھا ہے، اور ریاض کے لیے اس مؤقف سے پیچھے ہٹنا آسان نہیں ہوگا۔
دوسری جانب سالم الیامی کا کہنا ہے کہ مختلف علاقائی اور بین الاقوامی فریقوں کے ساتھ روابط بڑھانے کا مطلب سعودی پالیسی اصولوں سے دستبرداری نہیں۔ ان کے مطابق اسرائیل کے ساتھ تعلقات میں کسی بھی ممکنہ پیش رفت کا انحصار سعودی عرب کے وسیع تر تزویراتی مفادات اور ان شرائط پر ہوگا جنھیں ریاض کسی بھی سفارتی پیش رفت کے لیے ضروری سمجھتا ہے۔

گذشتہ چند برسوں کے تجربے نے سعودی قیادت کو یہ احساس دلایا ہے کہ اگرچہ امریکہ مشرقِ وسطیٰ میں ایک اہم اور بااثر شراکت دار ہے، لیکن واشنگٹن کی ترجیحات اور پالیسی ترجیحات مختلف حکومتوں کے ساتھ تبدیل ہوتی رہتی ہیں۔ یہی وجہ ہے کہ امریکی سکیورٹی ضمانتوں پر مکمل انحصار کرنا ریاض کے لیے پہلے کی نسبت زیادہ پیچیدہ ہو گیا ہے۔
امریکی وزارتِ خارجہ میں ریپبلکن اور ڈیموکریٹ دونوں ادوار میں چھ صدور کے ساتھ کام کرنے والے آرون ڈیوڈ ملر کے مطابق سعودی عرب ایک ایسے اتحادی کے ساتھ تعلقات رکھتا ہے جس کے بنیادی مفادات تو نسبتاً مستقل رہتے ہیں، لیکن ان مفادات کے حصول کے طریقے، ترجیحات اور سیاسی حساب کتاب وقت کے ساتھ بدلتے رہتے ہیں۔
مچل پلیٹنک کے نزدیک سعودی عرب اس حقیقت کو پہلے ہی بخوبی سمجھ چکا ہے۔ ان کے مطابق ریاض نے باراک اوباما، ڈونلڈ ٹرمپ، جو بائیڈن اور پھر دوبارہ ٹرمپ انتظامیہ کے ساتھ کام کیا ہے اور وہ جانتا ہے کہ کوئی بھی امریکی حکومت دائمی نہیں ہوتی۔ اسی لیے سعودی پالیسی ساز اب اپنے تعلقات کو کسی ایک شخصیت یا انتظامیہ کے بجائے طویل المدتی مفادات کی بنیاد پر ترتیب دینے کی کوشش کرتے ہیں۔
مچل پلیٹنک کا خیال ہے کہ اسی تجربے نے سعودی قیادت کو مختلف امریکی صدور اور ان کی بدلتی ترجیحات کے ساتھ خود کو ہم آہنگ کرنے میں پہلے سے زیادہ مہارت فراہم کی ہے۔
دوسری جانب عمرو حمزاوی کے مطابق اصل بحث یہ نہیں کہ سعودی عرب اور امریکہ کے درمیان شراکت داری برقرار رہے گی یا نہیں، بلکہ یہ ہے کہ واشنگٹن ہر بحران اور ہر تنازع میں کس حد تک عملی کردار ادا کرنے پر آمادہ ہوگا۔ ان کے نزدیک یہ مسئلہ کسی ایک امریکی انتظامیہ تک محدود نہیں بلکہ امریکی خارجہ پالیسی کے ایک وسیع تر رجحان کی عکاسی کرتا ہے۔
اسی تناظر میں چک فریلیخ کا مؤقف ہے کہ سعودی عرب امریکہ کا متبادل تلاش کرنے کے بجائے حفاظتی توازن کی حکمتِ عملی اپنا رہا ہے۔ اس پالیسی کا مقصد امریکہ سے دوری اختیار کرنا نہیں بلکہ اپنے دفاعی اور تزویراتی آپشنز کو وسیع کرنا ہے۔
