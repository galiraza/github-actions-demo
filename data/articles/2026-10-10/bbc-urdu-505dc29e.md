کیا واقعی دو لڑکیوں کے درمیان ’ایک کنگھی‘ کا تنازع ایک ہزار مربع میل پر پھیلی سلطنت کے زوال کا سبب بنا؟

گجرات کی تاریخ کے شواہد ابتدائی موریہ دور تک ملتے ہیں، تاہم اس کی تفصیلی اور معتبر تاریخ کا آغاز والابھی پور سے ہوتا ہے۔
سلطنتِ گپتا کے زوال کے بعد 470 عیسوی میں والابھی میں میئتریا سلطنت قائم ہوئی۔ ڈاکٹر ہری پرساد گنگادھر شاستری اپنی کتاب ’تاریخِ گجرات: قدیم دور‘ میں لکھتے ہیں میئتریا حکمرانوں کا عہد گجرات کی قدیم تاریخ کا دوسرا طویل ترین دور تھا۔
اس خاندان کی حکمرانی 470 سے 788 عیسوی تک، یعنی تین صدیوں سے زیادہ عرصے تک قائم رہی۔ تاریخی حوالوں کے مطابق میئتریا خاندان کے بانی فوجی کمانڈر بھٹارک تھے۔ بھٹارک کی حکومت پورے سوراشٹر، نیز کچھ، شمالی گجرات اور وسطی گجرات کے وسیع علاقوں تک پھیلی ہوئی تھی۔
میئتریا دورِ حکومت میں والابھی گجرات کا دارالحکومت تھا۔ گرینگر کے بعد والابھی میئتریا حکمرانوں کے عہد میں گجرات کا اہم سیاسی مرکز بن کر ابھرا۔ تاریخی ریکارڈ میں اس شہر کی خوشحالی اور غیر معمولی دولت کا ذکر موجود ہے۔
پروفیسر مولانا سید ابو جعفر ندوی اپنی کتاب ’تاریخِ گجرات: آریاؤں کی آمد سے مسلمانوں کے حملے تک‘ میں لکھتے ہیں کہ ’اس شہر کی معاشی حالت نہایت مستحکم تھی۔ ہنوں کے دور سے لے کر آخری عرب سیاحوں تک سب نے اس کی دولت و ثروت کی تعریف کی ہے، جبکہ تجارت بھی یہاں خوب پھلی پھولی۔ اس شہر میں متعدد دولت مند خاندان آباد تھے اور یہاں ایک سو سے زیادہ کروڑ پتی موجود تھے۔‘
اشرافیہ کی دولت اور علمی عظمت کے اس مرکز کا وجود جتنا شان دار تھا، اس کے زوال کی داستان اتنی ہی حیران کن اور چونکا دینے والی سمجھی جاتی ہے۔
تاریخی روایات کے مطابق، جنرل بھٹارک کے قائم کردہ اس عظیم شہر کے زوال کی وجہ ایک ’کنگھی‘ بنی۔ یہ شہر ایک ہزار مربع میل کے رقبے پر پھیلی میِتریا سلطنت کا دارالحکومت تھا۔
میئتریا خاندان کا عروج اور والابھی شہر کی شان و شوکت
ڈاکٹر ہری پرساد گنگادھر شاستری لکھتے ہیں کہ ’سکندر گپتا کی وفات کے بعد گپتا سلطنت کی طاقت کمزور پڑ گئی۔ جب دور دراز کے صوبوں نے خود مختاری حاصل کرنا شروع کی تو فوجی کمانڈر بھٹارک نے والابھی میں اپنی آزاد ریاست قائم کر لی۔‘
End of سب سے زیادہ پڑھی جانے والی
تاریخی روایات کے مطابق میئتری خاندان میں فوجی کمانڈر بھٹارک سے لے کر شیلادتیہ تک متعدد نامور حکمران گزرے۔ ان کے عہدِ حکومت میں ’والابھی کی شہرت ملک کے اندر ہی نہیں بلکہ بیرونی دنیا تک پھیل گئی۔‘
مؤرخ رسیک لال سی۔ پاریکھ لکھتے ہیں ’میئتریا حکمرانوں نے اپنی وسیع سلطنت میں ایک مؤثر نظامِ حکومت قائم کیا تھا۔ انھوں نے مذہب، تعلیم، ادب، فنون، تجارت اور صنعت کے فروغ کی سرپرستی کی، جبکہ اپنی سلطنت کی معاشی اور ثقافتی زندگی کو بھی ترقی دی۔‘
میئتریا دور میں والابھی نہ صرف گجرات کا سیاسی مرکز تھا بلکہ تعلیم، ثقافت اور تجارت کا بھی ایک نمایاں مرکز بن گیا تھا۔ حکمرانوں کی سرپرستی کے باعث یہاں علمی سرگرمیوں کو فروغ ملا، تجارت پھلی پھولی اور شہر خوشحالی کی علامت کے طور پر ابھرا۔ اسی وجہ سے والابھی کو اپنے زمانے کے اہم ترین تہذیبی مراکز میں شمار کیا جاتا ہے۔
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
والابھی کی خوشحالی کا ذکر محققین اور مؤرخین نے تاریخی کتب میں بڑی تفصیل سے کیا ہے۔ تاریخِ سوراشٹر کے مصنف شمبو پرساد دیسائی لکھتے ہیں کہ ’640 عیسوی میں چینی مسافر ہیون سانگ والابھی پور پہنچے۔ انھوں نے اپنے مشاہدات کی بنیاد پر والابھی کی تفصیلی تصویر پیش کی، جس سے اس شہر کی خوشحالی اور وسعت کا کافی حد تک اندازہ ہوتا ہے۔‘
پروفیسر مولانا سید ابو جعفر ندوی لکھتے ہیں ’ہیون سانگ کے سفرنامے کے مطابق یہ ایک گنجان آباد شہر تھا۔ اس بیان میں یہ بھی ذکر ملتا ہے کہ یہاں سینکڑوں معزز اور بااثر خاندان آباد تھے۔ شہر میں بے شمار تعلیمی ادارے، خانقاہیں اور مندر موجود تھے۔ بعض اندازوں کے مطابق اس کی آبادی کئی لاکھ نفوس پر مشتمل تھی۔‘
وہ مزید لکھتے ہیں کہ ہیون سانگ کے مطابق اس شہر میں متعدد انتہائی خوشحال خاندان رہتے تھے۔
’یہاں ایک سو سے زیادہ کروڑ پتی رہتے تھے۔‘
یہ تفصیلات ظاہر کرتی ہیں کہ والابھی اپنے دور میں نہ صرف گجرات کا سیاسی مرکز تھا بلکہ تجارت، تعلیم اور ثقافت کا بھی ایک بڑا مرکز شمار ہوتا تھا۔ اس کی خوشحالی، تعلیمی اداروں کی کثرت اور امیر تاجروں کی موجودگی نے اسے برصغیر کے اہم ترین شہروں میں شامل کر دیا تھا۔
ایک کنگھی کا تنازع جو والابھی کے زوال کی وجہ بنا
پروفیسر مولانا سید ابو جعفر ندوی اپنی کتاب میں لکھتے ہیں ’یہ ناقابلِ فہم ہے کہ وہ عظیم شہر، جو ڈھائی سو برس سے زیادہ عرصے تک تمام شہروں میں ممتاز مقام رکھتا تھا، آخر کس طرح تباہی سے دوچار ہوا۔ اس کے بارے میں ہارون کے خزانے کی طرح کئی افسانوی اور نیم تاریخی کہانیاں مشہور ہیں۔
ان کے مطابق ’ایک روایت ایسی ہے جس کا ذکر عام تاریخی کتابوں میں ملتا ہے اور تقریباً تمام ہندو مؤرخین نے اسے نقل کیا ہے۔‘
اس روایت کے مطابق والابھی پور میں کاکو نامی ایک نہایت دولت مند شخص رہتا تھا۔ اس کی شہر میں بڑی بڑی حویلیاں تھیں۔ کہا جاتا ہے کہ اس کی بیٹی کے پاس ہیروں سے جڑی ہوئی ایک کنگھی تھی۔ جب بادشاہِ وقت کی بیٹی کو اس کنگھی کا علم ہوا تو اس نے اسے حاصل کرنے کی خواہش ظاہر کی، لیکن کاکو کی بیٹی نے انکار کر دیا۔
روایت کے مطابق شہزادی نے ناراض ہو کر اپنے والد سے شکایت کی۔ اس پر بادشاہ نے خود کاکو سے کنگھی کا مطالبہ کیا، مگر اس نے بھی کنگھی دینے سے انکار کر دیا۔ کہا جاتا ہے کہ کاکو کی جانب سے انکار پر بادشاہ غضب ناک ہو گیا اور اس نے اپنے سپاہیوں کو کاکو کی بیٹی کا ریپ کرنے کا حکم دیا۔
روایات کے مطابق کاکو اس ظلم کو برداشت نہ کر سکا اور انھوں نے انتقام لینے کا فیصلہ کیا۔
انھوں نے ایک غیر ملکی فوجی سردار کو بھاری رقم دی اور اسے والابھی پر حملے کے لیے آمادہ کیا۔ بعد ازاں اس فوج نے والابھی پور کو لوٹ لیا اور شہر تباہی کا شکار ہو گیا، جبکہ بادشاہ غم و اندوہ میں ڈوب گیا۔
ڈاکٹر ہری پرساد گنگادھر شاستری لکھتے ہیں ’والابھی کی تباہی کا ذکر جین تواریخ میں تفصیل سے ملتا ہے۔ اس روایت کے مطابق والابھی کے حکمران شیلادتیہ نے شہر کے ایک بااثر خاندان کی خاتون، کاکو کی بیٹی، سے جواہرات سے مزین سونے کا زیور چھین لیا اور اس کا ریپ بھی کیا۔ انتقام لینے کے لیے یہ خاتون مِلیچھ علاقے گئی، وہاں کے سردار کو رقم ادا کر کے والابھی پر حملہ کروایا۔‘
تاہم مؤرخین عموماً اس قصے کو ایک روایتی یا نیم افسانوی روایت کے طور پر دیکھتے ہیں۔ والابھی کے زوال کے حقیقی اسباب کے بارے میں تاریخی شواہد محدود ہیں، جبکہ مختلف ذرائع اس کی تباہی کو بیرونی حملوں، سیاسی کمزوری یا علاقائی تبدیلیوں سے بھی جوڑتے ہیں۔ اس لیے کنگھی والی داستان کو زیادہ تر ایک مشہور روایت سمجھا جاتا ہے، نہ کہ ثابت شدہ تاریخی حقیقت۔
روایت کے مطابق جب ’مِلیچھ‘ فوج والابھی پور کے قریب پہنچی تو کاکو کی چال کے نتیجے میں بادشاہ شیلادتیہ کا الٰہی گھوڑا آسمان کی طرف اڑ گیا۔ کہا جاتا ہے کہ اس غیر معمولی واقعے نے شیلادتیہ کو سخت پریشان اور بے بس کر دیا۔ بعد ازاں وہ جنگ میں مارا گیا اور مِلیچھ فوج نے والابھی کو تباہ کر دیا۔ اس روایت کے مطابق یہ واقعہ 788-789 عیسوی میں پیش آیا۔
عرب مؤرخ البیرونی کے حوالے سے معلوم ہوتا ہے کہ جین متون میں جس ’مِلیچھ سردار‘ کا ذکر کیا گیا ہے، اس سے مراد المنصورہ کا حکمران تھا، یعنی سندھ میں عرب اقتدار کا نمائندہ۔
تاریخِ سوراشٹر میں شیلادتیہ کے ’سورج کنڈ‘ سے متعلق ایک اور روایت بھی بیان کی گئی ہے۔ اس داستان کے مطابق مصیبت کے وقت سورج کنڈ سے سورج دیوتا کا ایک گھوڑا نمودار ہوتا تھا، جس پر سوار ہو کر شیلادتیہ جنگ کرتا اور ہمیشہ فتح یاب رہتا۔ روایت کے مطابق کاکو نے یہ راز عربوں کو بتا دیا۔ پھر حملہ آوروں نے گائیں ذبح کر کے ان کی باقیات اس تالاب میں پھینک دیں، جس کے نتیجے میں وہ گھوڑا غائب ہو گیا اور شیلادتیہ بے یار و مددگار رہ گیا۔
کتاب کے مطابق والابھی پور کے باشندے اور سپاہی اس اچانک حملے سے خوف زدہ ہو گئے۔ یہ واضح نہیں کہ جنگ کتنی شدید تھی یا کتنے عرصے تک جاری رہی، لیکن روایت میں بتایا گیا ہے کہ خلیفہ کے کمانڈر عمر بن جمال کی فوج نے شیلادتیہ کو جنگ میں قتل کر دیا۔ اس کے ساتھ ہی والابھی کی سلطنت تاریخ کے صفحات سے مٹ گئی۔
جین ماخذ سے یہ بھی معلوم ہوتا ہے کہ شیلادتیہ اور دارالحکومت والابھی کی تباہی کے بعد عرب فوج واپس لوٹ گئی، لیکن میئتریا خاندان کی سیاسی طاقت مکمل طور پر ختم ہو گئی۔ بعض روایات کے مطابق شیلادتیہ کی ظالمانہ پالیسیوں اور طرزِ حکومت نے بھی ریاست کے المناک انجام میں کردار ادا کیا۔
والابھی کی تباہی کے بارے میں جین متون سمیت متعدد تاریخی ماخذ مختلف روایات بیان کرتے ہیں۔ آج تک یہ واضح نہیں ہو سکا کہ ان میں سے کون سی روایت مکمل طور پر مستند تاریخی حقیقت کی نمائندگی کرتی ہے۔ تاہم بیشتر روایات میں کاکو یا ’رنگا کاکو‘ سے منسوب قصہ کسی نہ کسی شکل میں ضرور موجود ہے، جس کی وجہ سے یہ داستان والابھی کے زوال کے حوالے سے سب سے زیادہ مشہور روایت بن گئی ہے۔
