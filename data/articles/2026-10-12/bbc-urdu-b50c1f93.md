مبینہ ویزا فراڈ کے باعث طالبہ لائبہ اعجاز کی خودکشی: بیرونِ ملک تعلیم کے نام پر دھوکے سے کیسے بچا جا سکتا ہے؟

  - عہدہ, بی بی سی اُردو، اسلام آباد

اس رپورٹ میں جرم اور خودکشی سے متعلق تفصیلات قارئین کے لیے تکلیف دہ ہو سکتی ہیں
وفاقی دارالحکومت اسلام آباد میں مبینہ ویزا فراڈ کا نشانہ بننے والی انٹرنیشنل اسلامک یونیورسٹی کی 23 سالہ طالبہ کی خودکشی کے بعد حکومت نے بیرونِ ملک بھجوانے کا جھانسہ دے کر رقوم بٹورنے والے عناصر کے خلاف گھیرا تنگ کرنے کا اعلان کیا ہے۔
اُدھر 23 سالہ طالبہ لائبہ اعجاز کی مبینہ خودکشی کے کیس میں گرفتار ملزم کو پانچ روزہ جسمانی ریمانڈ پر پولیس کے حوالے کر دیا گیا ہے۔
ملزم کو پیر کی صبح جوڈیشل مجسٹریٹ زنیرہ ظفر کی عدالت میں چہرہ ڈھانپ کر پیش کیا گیا جہاں تفتیشی حکام نے بتایا کہ انھیں دستیاب شواہد اور ابتدائی تفتیش کی روشنی میں ملزم مزید تفتیش کے لیے مطلوب ہے۔ پولیس کی جانب سے ملزم کے 10 روز کے جسمانی ریمانڈ کی استدعا کی گئی تھی۔
یاد رہے کہ جمعہ (نو اکتوبر) کے روز صوبہ پنجاب کے وسطی شہر سرگودھا سے تعلق رکھنے والی طالبہ لائبہ اعجاز کی لاش اسلام آباد کے سیکٹر جی ٹین میں واقع ہفتہ وار بازار میں درخت سے لٹکی ہوئی ملی تھی۔ پولیس کے مطابق لائبہ نے بظاہر گلے میں دوپٹہ ڈال کر مبینہ طور پر خودکشی کی تھی۔
خودکشی کی مبینہ وجہ بعدازاں یہ سامنے آئی کہ ملزم نے لائبہ اور اُن کے اہلخانہ سے امریکہ کی یونیورسٹی میں داخلہ دلوانے کا جھانسہ دے کر 50 لاکھ روپے بٹورے تھے۔ بعدازاں جب ملزم کی جانب سے فراہم کردہ دستاویزات کو چیک کروایا گیا تو وہ جعلی نکلے۔
پولیس کا دعویٰ ہے کہ اسی معاملے پر دلبرداشتہ لائبہ نے موت کو گلے لگا لیا کیونکہ اس رقم کی ادائیگی کے لیے اُن کے والد نے ناصرف اپنی جمع پونجی خرچ کی تھی بلکہ اُس پلاٹ کو بھی فروخت کر دیا تھا جس پر وہ اپنا مکان بنانے کا ارادہ رکھتے تھے۔
لائبہ اعجاز کی مبینہ خود کشی کا وزیراعظم شہباز شریف نے نوٹس لیا تھا، جس کے بعد ملزم کو حراست میں لیا گیا تھا۔
End of سب سے زیادہ پڑھی جانے والی
ہم نے اس تحریر میں اس کیس میں ہونے والی پیش رفت کے ساتھ ساتھ طلبہ کو بیرونِ ملک بھجوانے والے اداروں اور وفاقی تفتیشی ایجنسی (ایف آئی اے) کے عہدے دار سے بات کر کے یہ جاننے کی کوشش کی ہے کہ طلبہ اس نوعیت کے فراڈ سے کیسے بچ سکتے ہیں۔
بیرونِ ملک تعلیم کے نام پر دھوکے سے کیسے بچا جا سکتا ہے؟
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
ایف آئی اے کے ایک سینئر عہدے دار نے نام ظاہر نہ کرنے کی شرط پر بی بی سی کو بتایا کہ ’سب سے تشویش ناک بات یہ ہے کہ پاکستان میں ویزا کنسلٹنٹس کی رجسٹریشن کا کوئی نظام موجود نہیں، جس کے باعث طلبہ کے دھوکہ دہی اور فراڈ کا شکار ہونے کے امکانات بڑھ جاتے ہیں۔‘
اُن کا مزید کہنا تھا کہ ’آج کے دور میں بین الاقوامی تعلیمی اداروں تک رسائی آسان ہو چکی ہے۔ ہر ادارے کی اپنی ویب سائٹ اور ہیلپ ڈیسک موجود ہوتی ہے، جہاں سے داخلے کے خواہشمند طلبہ براہِ راست معلومات حاصل کر سکتے ہیں۔‘
اُنھوں نے بتایا کہ ’آسٹریلیا اور برطانیہ کی بعض جامعات پاکستان میں موجود اپنے کنسلٹنٹس کے ذریعے داخلے کے لیے درخواست دینے کی ہدایت کرتی ہیں۔ بعض صورتوں میں ان کنسلٹنٹس کے ذریعے درخواست دینے پر داخلہ فیس میں رعایت بھی دیے جانے کے وعدے ہوتے ہیں اور ایسے میں اس کاروبار میں چھپے بعض جرائم پیشہ افراد لوگوں کو دھوکہ دینے میں کامیاب ہو جاتے ہیں۔‘
ایف آئی اے کے سنئیر افسر نے طلبہ کو مشورہ دیا کہ ’اگر وہ کسی دھوکے یا فراڈ کا شکار ہو جائیں تو مزید نقصان سے بچنے کے لیے انتظار کرنے کے بجائے فوری طور پر ایف آئی اے سے رابطہ کریں تاکہ مزید کسی بڑے نقصان سے بچا جا سکے۔‘
’مستند کنسلٹنٹ کا باقاعدہ دفتر اور رجسٹرڈ کاروبار ہونا چاہیے‘
نعمان چٹھہ لاہور میں سٹوڈنٹ ویزا کنسلٹنسی فراہم کرنے والی کمپنی چلاتے ہیں اور اب تک کئی طلبہ کو بیرونِ ملک بھجوا چکے ہیں۔
بی بی سی نیوز اُردو سے گفتگو کرتے ہوئے اُن کا کہنا تھا کہ طلبہ کسی کنسلٹنٹ کی ساکھ جانچنے کے لیے اس کے سوشل میڈیا پیجز پر موجود تبصرے اور ریویوز دیکھ سکتے ہیں۔
ان کا کہنا تھا کہ ’مستند کنسلٹنٹ کا باقاعدہ دفتر اور رجسٹرڈ کاروبار ہونا چاہیے، جبکہ متعلقہ پیشہ ورانہ رجسٹریشن بھی دیکھی جا سکتی ہے۔‘
انھوں نے مشورہ دیا کہ ’یونیورسٹی کا آفر لیٹر یا داخلے کی دستاویز ملنے پر طالب علم اسے متعلقہ یونیورسٹی کو ای میل کر کے اس کی تصدیق کروائے۔‘
ان کے مطابق، طلبہ کو درج ذیل پانچ باتوں کا خاص خیال رکھنا چاہیے:
- داخلے کی ضمانت کے نام پر بھاری رقم سے محتاط رہیں: اگر کوئی کنسلٹنٹ لاکھوں روپے لے کر کسی یونیورسٹی میں داخلہ دلوانے کا دعویٰ کرے تو اس کی اچھی طرح جانچ پڑتال کریں۔
- یونیورسٹی کی فیس خود ادا کریں: کنسلٹنٹ کے بجائے یونیورسٹی کے سرکاری طریقۂ ادائیگی کے ذریعے فیس جمع کروائیں تاکہ رقم کی ادائیگی کا ریکارڈ آپ کے پاس ہو۔
- جعلی دستاویزات سے گریز کریں: بینک سٹیٹمنٹ یا دیگر مالی دستاویزات بنوانے کی پیشکش قبول نہ کریں۔ جعلی کاغذات کے نتیجے میں ویزا مسترد ہونے اور دیگر قانونی مسائل کا سامنا کرنا پڑ سکتا ہے۔
- اپنی ای میل اور رابطہ نمبر استعمال کریں: یونیورسٹی میں داخلے اور ویزا درخواست کے لیے اپنے ای میل ایڈریس اور فون نمبر دیں تاکہ تمام اہم پیغامات براہِ راست آپ کو موصول ہوں اور یونیورسٹی کی جانب سے آپ کا داخلہ ہونے اور نہ ہونے کی براہِ راست ہر بات آپ کے علم میں ہو۔
- ویزا درخواست پر اپنا کنٹرول رکھیں: درخواست جمع ہونے سے لے کر فیصلے تک اس کی پیش رفت خود معلوم کرتے رہیں۔ اگر ای ویزا جاری ہو تو اس تک رسائی بھی اپنے پاس رکھیں، تاکہ کنسلٹنٹ معلومات چھپا کر مزید رقم کا مطالبہ نہ کر سکے۔
ایف آئی اے کے سنئیر افسر نے بتایا کہ اس سارے معاملے میں سب سے اہم بات یہ ہے کہ ’پاکستان سے باہر کسی بھی تعلیمی ادارے میں داخلے کے لیے طلبہ کو چاہیے کہ وہ ادارے سے ساتھ خود رابطے میں رہیں تاکہ وہ کسی بھی فراڈ سے بچ سکیں۔‘
نعمان چٹھہ کا کہنا تھا کہ اس واقعے کے بعد اب حکومت کو آگاہی مہم چلانی چاہیے، تاکہ نوجوان طلبہ کو یہ پتہ چل سکے کہ بیرونِ ملک جانے کا درست طریقہ کیا ہے اور فراڈ سے کیسے بچا جا سکتا ہے۔
ایف آئی آر میں کیا بتایا گیا؟
اس واقعے کی ایف آئی آر اسلام آباد کے تھانہ رمنہ میں قتل بسبب (322) اور مجرمانہ نیت (34) کے تحت درج کی گئی ہے۔ اس ایف آئی آر کو ابتدا میں سیل کر دیا گیا تھا تاہم پیر کی صبح یہ ایف آئی آر ڈی سیل کر دی گئی ہے۔
ایف آئی آر لائبہ کے والد کی مدعیت میں درج کی گئی ہے جس میں انھوں نے دعویٰ کیا کہ ملزم نے بیرونِ ملک تعلیم کے لیے ویزا دلوانے کے نام پر مبینہ فراڈ کیا اور مالی دباؤ کے باعث اُن کی بیٹی لائبہ اعجاز نے انتہائی قدم (خودکشی) اٹھایا۔
ایف آئی آر کے مطابق مدعی مقدمہ اسلام آباد میں دودھ دہی کی دکان چلاتے ہیں جبکہ اُن کی 23 سالہ بیٹی لائبہ اعجاز اسلامی یونیورسٹی کی طالبہ اور سکالرشپ ہولڈر تھیں۔
والد کے مطابق تقریباً چھ سے سات ماہ قبل اُن کی بیٹی کی ملاقات ملزم سے ہوئی، جس نے خود کو اسلام آباد میں ایک اعلیٰ عہدیدار ظاہر کیا اور لائبہ کو سٹڈی ویزا پر امریکہ بھجوانے کی پیشکش کی۔ درخواست میں دعویٰ کیا گیا ہے کہ اس مقصد کے لیے 93 لاکھ روپے کی رقم طے پائی، جس میں سے 21 لاکھ روپے نقد جبکہ 29 لاکھ روپے مختلف اکاؤنٹس کے ذریعے ملزم کو ادا کیے گئے۔
والد کے مطابق بعد میں ملزم کی جانب سے اُن کی بیٹی کو ویزا سے متعلق دستاویزات دی گئیں، تاہم جانچ پڑتال پر وہ مبینہ طور پر جعلی ثابت ہوئیں۔ ایف آئی آر میں مزید الزام لگایا گیا ہے کہ بینک سٹیٹمنٹ کے لیے 20 ہزار امریکی ڈالر اکاؤنٹ میں جمع کروانے کی رسید بھی جعلی نکلی۔
والد کا کہنا ہے کہ جب انھوں نے یہ تفصیل سامنے آنے پر ملزم سے رابطہ کیا تو اُس نے فون کالز کا جواب دینا بند کر دیا جبکہ ملزم کی ایک ساتھی کی جانب سے دوران مبینہ طور پر مزید رقم کا مطالبہ جاری رہا۔
درخواست میں مؤقف اختیار کیا گیا ہے کہ مسلسل ذہنی دباؤ اور پریشانی کے باعث لائبہ اعجاز شدید اضطراب کا شکار رہیں اور اسی صورتحال کے نتیجے میں انھوں نے اپنی زندگی کا خاتمہ کر لیا۔
پولیس دستاویزات کے مطابق واقعے کی اطلاع ملنے پر تفتیشی افسر موقع پر پہنچے جبکہ فرانزک ٹیم نے بھی شواہد اکٹھے کیے۔ لائبہ کی لاش کو پوسٹ مارٹم کے لیے پمز ہسپتال منتقل کیا گیا، جہاں قانونی کارروائی مکمل ہونے کے بعد میت ورثا کے حوالے کر دی گئی۔
یاد رہے کہ اس معاملے کا انکوائری ایف آئی اے بھی کر رہی ہے اور ایف آئی اے کے اسلام آباد ریجن کے ڈائریکٹر شہزاد بخاری نے بی بی سی اُردو کو بتایا ’اس واقعہ سے متعلق تمام شواہد جمع کیے جا رہے ہیں اور ملزم کے موبائل کا ریکارڈ بھی حاصل کر لیا گیا ہے۔‘
’میرے ساتھ ہونے والے فراڈ کے بارے میں میرے والد صاحب کو نہ بتانا ورنہ وہ مر جائیں گے‘
یہ الفاظ لائبہ کے ہیں جو انھوں نے مبینہ طور پر خود کشی کرنے سے پہلے اپنے بہنوئی کو فون کرکے کہے تھے۔
لائبہ کے بہنوئی نے بی بی سی اُردو سے بات کرتے ہوئے بتایا تھا کہ لائبہ لائق اور قابل طالبہ تھیں اور اسلامک یونیورسٹی میں سکالر شپ پر پڑھی تھیں جبکہ اُن کی یونیورسٹی نے انھیں مزید سکالر شپ پر سعودی عرب بھیجنے کی آفر بھی کر رکھی تھی۔
اُن کے بہنوئی کے مطابق بی ایس مکمل کرنے کے بعد لائبہ آن لائن کام کرتی تھیں، جہاں اُن کا رابطہ ایک شخص سے ہوا جس پر خاندان کا الزام ہے کہ اس نے امریکہ کی یونیورسٹی میں داخلہ دلوانے کا جھانسہ دے کر لاکھوں روپے وصول کیے۔
بہنوئی کا دعویٰ ہے کہ لائبہ نے اس شخص کو اسلامک یونیورسٹی کی جانب سے اعلی تعلیم کے لیے سعودی عرب بھیجنے کی آفر کے بارے میں بتایا تو اس نے مبینہ طور پر کہا کہ وہ امریکہ کی کسی یونیورسٹی میں داخلہ دلوا سکتا ہے جس کے لیے رقم کی ادائیگی طے پائی تھی۔
انھوں نے کہا کہ چونکہ 'لائبہ کو اعلی تعلیم حاصل کرنے کے لیے باہر جانے کا شوق تھا اور اس نے اپنی اس خواہش کا اظہار اپنے والد سے کیا جنھوں نے بچی کی خواہش کو مدنظر رکھتے ہوئے وہ پلاٹ بھی بیچ دیا جو اپنے گھر کی تعمیر کے لیے خریدا تھا۔'
انھوں نے دعویٰ کیا کہ ملزم گذشتہ مہینوں میں مختلف اوقات میں اس خاندان سے مختلف دستاویزات کی تیاری کے نام پر پیسے بٹورتا رہا، مگر بعد میں اس کی طرف سے دیے گئے تمام ڈاکومینٹس جعلی نکلے، جس پر لائبہ انتہائی پریشان تھیں کیونکہ اُن کی والد اس کام پر اپنی تمام جمع پونجی خرچ کر چکے تھے جبکہ ملزم نے بھی رابطہ کرنا بند کر دیا تھا۔
لائبہ اعجاز کا مبینہ ویزا فراڈ کا نشانہ بننا بیرونِ ملک اعلیٰ تعلیم حاصل کرنے کے خواہشمند پاکستانی طلبہ کے لیے بھی کئی سوالات کو جنم دیتا ہے۔ بیرونِ ملک کسی یونیورسٹی میں داخلے سے لے کر سٹڈی ویزا حاصل کرنے تک کے مراحل میں طلبہ کو کن باتوں کا خیال رکھنا چاہیے اور وہ خود کو جعل سازوں سے کیسے محفوظ رکھ سکتے ہیں؟
