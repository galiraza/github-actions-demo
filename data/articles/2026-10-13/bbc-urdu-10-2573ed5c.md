# ’میں سوشل میڈیا ویڈیوز کو وائرل کر کے ماہانہ 10 ہزار پاؤنڈ کماتا ہوں‘

’میں سوشل میڈیا ویڈیوز کو وائرل کر کے ماہانہ 10 ہزار پاؤنڈ کماتا ہوں‘

لندن سے تعلق رکھنے والا 22 سالہ نوجوان ایک پیشہ ور ’کلپر‘ ہیں۔ انھیں مختلف گاہک مختصر ورٹیکل سوشل میڈیا ویڈیوز بنانے کے لیے معاوضہ دیتے ہیں، اس امید کے ساتھ کہ وہ وائرل ہو جائیں گی۔
وہ کہتے ہیں کہ ’جتنے زیادہ ویوز ویڈیو کلپ کو ملتے ہیں، میں اتنی ہی زیادہ کمائی کرتا ہوں۔‘
عام طور پر اس کا طریقۂ کار یہ ہوتا ہے کہ کلائنٹ، جو کسی کپڑوں کے برانڈ، ریکارڈ لیبل، فلم سٹوڈیو، پوڈکاسٹر یا سوشل میڈیا انفلوئنسر میں سے کوئی بھی ہو سکتا ہے، ایک طویل ویڈیو ایکنور کو فراہم کرتا ہے، جسے وہ پہلے مکمل دیکھتے ہیں۔
اس کے بعد وہ 30 سے 60 سیکنڈ کے ایسے مختصر ویڈیو کلپس تیار کرتے ہیں، جنھیں وہ ویڈیو کے سب سے دلچسپ یا مؤثر حصے سمجھتے ہیں اور پھر انہیں انسٹاگرام، یوٹیوب، ٹک ٹاک یا دیگر پلیٹ فارمز پر اپ لوڈ کر دیتے ہیں۔
اس تصور کے پیچھے یہ خیال کارفرما ہوتا ہے کہ کلپرز، جن کی عمریں عموماً 16 سے 25 سال کے درمیان ہوتی ہیں، اس عمر کے گروہ کی دلچسپیوں اور ترجیحات کو بہتر طور پر سمجھتے ہیں اور اس لیے وہ مختصر ویڈیوز کو زیادہ سے زیادہ ویوز دلانے میں کامیاب ہو سکتے ہیں۔
10 ہزار پاؤنڈ ماہانہ
اگر کسی ویڈیو کلپ کو غیر معمولی مقبولیت مل جائے تو یہ کام بہت منافع بخش ثابت ہو سکتا ہے کیونکہ مختلف پلیٹ فارمز پر کلپرز پر آنے والے ہر 1,000 ویوز کے عوض 3 ڈالر (تقریباً 2.30 پاؤنڈ) تک کمائے جا سکتے ہیں۔
ایکنور کا کہنا ہے کہ وہ گذشتہ سال اس وقت کلپنگ کے اس کام کے عادی ہو گئے تھے، جب انھوں نے پاپ گلوکارہ دعا لیپا کی ایک مختصر ویڈیو انسٹاگرام پر پوسٹ کر کے 300 ڈالر کمائے تھے۔
’میں نے سوچا ’اچھا، صرف ایک ویڈیو پوسٹ کرنے کے عوض یہ خاصی بڑی رقم ہے۔‘ مجھے فوراً اندازہ ہو گیا تھا کہ یہ کام کتنا منافع بخش ہو سکتا ہے۔‘
End of سب سے زیادہ پڑھی جانے والی
ان کا کہنا ہے کہ اب وہ اس کام سے ’ہر مہینے کم از کم 10,000 پاؤنڈ‘ کما رہے ہیں۔
وہ مزید کہتے ہیں کہ وہ ایسی ویڈیو کلپس تلاش کرتے ہیں جو مزاحیہ ہوں، جلدی دیکھی جا سکیں اور ناظرین کی دلچسپی برقرار رکھیں۔
اس کی ایک مثال وہ مختصر ویڈیو ہے، جو انھوں نے مئی میں نیویارک میں ہونے والے میٹ گالا میں گلوکارہ سبرینا کارپنٹر کے گاؤن کے دامن پر کسی کے پاؤں پڑ جانے پر ان کے ردِعمل پر مبنی بنائی تھی۔ سنگھ نے یہ نہیں بتایا کہ اس ویڈیو کا کلائنٹ کون تھا۔
وہ جو کچھ نہیں کماتے
وِچِٹا کینساس سے تعلق رکھنے والے 24 سالہ کلپر ایتھن ویمز کے لیے یہ ان کے دیگر ویڈیو کے کام کے ساتھ ایک منافع بخش اضافی ذریعۂ آمدن ہے۔ ان کا کہنا ہے کہ وہ کلپنگ سے ’کم از کم ایک ہزار ڈالر ماہانہ‘ کماتے ہیں۔
وہ یہ کام یوٹیوبر اور پوڈکاسٹر کیلیب ہیمر کے لیے مختصر ویڈیوز بنا کر کرتے ہیں، جو اپنے پروگرام فنانشل آڈٹ کی میزبانی کے لیے معروف ہیں۔ اس پروگرام میں وہ اپنے مہمانوں کے ذاتی مالی معاملات کا جائزہ لیا جاتا ہے۔
ویمز کا کہنا ہے کہ کلپنگ نے انھیں گھر بیٹھے روزگار کمانے کا موقع فراہم کیا ہے، جس سے انھیں اپنی اہلیہ اور ایک سالہ بیٹے کے ساتھ زیادہ وقت گزارنے کا موقع ملتا ہے۔
وہ کہتے ہیں ’میں روایتی نو سے پانچ والی ملازمت نہیں کرنا چاہتا، جس میں مجھے ہر روز دفتر جانا پڑے۔‘
بعض کلپرز اچھی آمدنی کما سکتے ہیں. تاہم سوشل میڈیا کنسلٹنٹ لیا ہیبرمین خبردار کرتی ہیں کہ ایسے افراد کی تعداد بہت کم ہے۔
وہ کہتی ہیں ’یہ مجموعی طور پر کریئیٹر اکانومی کی عکاسی کرتا ہے۔ بھاری اکثریت یا تو بہت کم کماتی ہے یا کچھ بھی نہیں کماتی۔‘
دوسری جانب سٹریمنگ انڈسٹری کے تجزیہ کار ڈیوِن نیش کا کہنا ہے کہ یہ اندازہ لگانا آسان ہے کہ کلپنگ کے شعبے میں کامیابی حاصل کرنے کے لیے کتنا وقت درکار ہوتا ہے۔ ان کے مطابق کامیاب کلپرز کو ’گھنٹوں، گھنٹوں مواد دیکھنا پڑتا ہے اور وہ اسی دنیا میں جیتے ہیں۔‘
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
کلپرز کو ممکنہ کلائنٹس سے جوڑنے کے لیے متعدد ویب سائٹس موجود ہیں، جہاں ویڈیو سے متعلق دستیاب کاموں کی فہرست دی جاتی ہے۔
ان میں ریڈٹ اور ڈسکارڈ جیسے عمومی آن لائن فورمز کے صفحات سے لے کر خاص طور پر کلپنگ کے لیے مختص ویب سائٹس، مثلاً کلپنگ کلچر، وائرو، وھوپ اور کِک شامل ہیں۔
یہ ویب سائٹس کام کروانے کی خواہش رکھنے والی کمپنیوں سے فیس وصول کرتی ہیں۔ کلپنگ کلچر کے کلائنٹس میں کیپیٹل ریکارڈز، یونیورسل میوزک گروپ اور ایچ بی او میکس شامل ہیں۔
ایسی کمپنیوں کے لیے گھروں کے کمروں سے کام کرنے والے نوجوانوں کو ملازمت دینا سوشل میڈیا پر روایتی اشتہارات چلانے کے مقابلے میں کہیں سستا پڑتا ہے اور صارفین کی توجہ حاصل کرنے کے کے لیے یہ حکمت عملی اکثر زیادہ مؤثر بھی ثابت ہوتا ہے۔
یہی تجربہ لاس ویگاس میں قائم ماڈیولر گھروں کی تعمیر کرنے والی کمپنی باکسابل کا بھی ہے۔ کمپنی کے بانی گیلیانو تیرامانی، جو کمپنی کی بہت سی ویڈیوز میں خود بھی نظر آتے ہیں، کہتے ہیں: ’میرے خیال میں اب تک ہماری ویڈیوز اور کلپس کو مجموعی طور پر تقریباً ایک ارب ویوز مل چکے ہیں۔‘
برطانیہ میں ڈاؤننگ سٹریٹ کے سابق مشیر اور اب جیمز جابز آف دی فیوچر پوڈکاسٹ کے میزبان جمی میک لافلن نے بھی مختصر ویڈیو کلپس کو اپنا لیا ہے۔
وہ کہتے ہیں: ’یہ ویڈیو کلپس تقریباً مرکزی پروگراموں کے ٹریلرز کی طرح ہوتی ہیں۔‘
مثال کے طور پر اینڈی برنہم کے ساتھ ان کے ایک حالیہ پوڈکاسٹ کو ’طویل دورانیے کی مکمل سنے جانے والے تقریباً 50 ہزار پلے‘ ملے ’لیکن کلپس کے لحاظ سے اسے لاکھوں ویوز حاصل ہوئے ہیں۔‘
کلپنگ کلچر کے بانی اور چیف ایگزیکٹو 21 سالہ ایون سٹین فیلڈ نے گذشتہ برس یونیورسٹی آف کینساس چھوڑنے کے بعد یہ کاروبار شروع کیا تھا۔
سٹین فیلڈ کا کہنا ہے کہ ان کی ویب سائٹ دنیا بھر کے 340,000 سے زیادہ کلپرز کو مختلف برانڈز سے جوڑتی ہے۔
مختصر ویڈیوز کی مقبولیت کے بارے میں ان کا نظریہ سادہ ہے: ’لوگوں کی توجہ کا دورانیہ مسلسل کم سے کم ہوتا جا رہا ہے۔‘
وہ مزید کہتے ہیں کہ مختصر ویڈیوز بنانا دراصل نوجوانوں کا کام ہے کیونکہ ’مختصر دورانیے کی ویڈیوز کے زیادہ تر صارفین یہی لوگ ہیں۔‘
لندن میں ایکنور سنگھ کا کہنا ہے کہ وہ تقریباً ’ہر وقت کلپنگ میں مصروف‘ رہتے ہیں، چاہے وہ کہیں بھی ہوں۔
وہ کہتے ہیں ’میں کہیں بھی ہو سکتا ہوں۔ میں چھٹیوں پر بھی جا سکتا ہوں اور ساتھ ہی پیسے بھی کما سکتا ہوں۔‘
