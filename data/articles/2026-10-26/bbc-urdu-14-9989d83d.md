14 روپے کمانے والے بس ڈرائیور کا بیٹا ’راکنگ سٹار یش‘ کیسے بنا؟
کنڑ فلم انڈسٹری میں ’راکنگ سٹار‘ کے نام سے مشہور یش نے ہندی فلم رامائن میں راون کا کردار کیوں قبول کیا، اس بات پر خاصی بحث جاری ہے۔
اس سال دیوالی کے موقع پر ہدایت کار نتیش تیواری کی فلم رامائن ریلیز ہونے جا رہی ہے۔ اس فلم کے بڑے بجٹ، شاندار کاسٹ اور پروڈکشن کے حجم نے اسے پہلے ہی خبروں کا مرکز بنا دیا ہے۔ فلم میں رام کا کردار رنبیر کپور ادا کر رہے ہیں جبکہ سیتا کے روپ میں سائی پلوی نظر آئیں گی۔
تاہم اس فلم میں ایک نام اور بھی غیرمعمولی توجہ حاصل کر رہا ہے اور وہ ہے کنڑ سنیما کے سپر سٹار یش۔
یش نہ صرف اس فلم میں راون کا کردار نبھا رہے ہیں بلکہ وہ اس کے شریک پروڈیوسرز میں بھی شامل ہیں۔ فلم پانچ نومبر کو ریلیز ہو گی۔
یش کا شمار جنوبی انڈیا کے مقبول ترین اداکاروں میں ہوتا ہے۔ کے جی ایف کی غیرمعمولی کامیابی کے بعد ان کی شہرت پورے انڈیا میں پھیل گئی اور وہ قومی سطح پر پہچانے جانے لگے۔ آج وہ کنڑ سنیما کی کامیاب ترین شخصیات میں شمار ہوتے ہیں۔
8 جنوری 1986 کو کرناٹک کے ضلع حسن میں پیدا ہونے والے یش کا اصل نام نوین کمار گوڑا ہے۔ تاہم فلمی دنیا اور مداحوں کے درمیان وہ اپنے اصل نام سے کم اور ’راکنگ سٹار یش‘ کے نام سے زیادہ جانے جاتے ہیں۔
ایک انٹرویو میں جب ان سے پوچھا گیا کہ اگر کوئی انھیں نوین کمار گوڑا کہہ کر پکارے تو ان کا ردِعمل کیا ہوگا، تو انھوں نے مزاحیہ انداز میں جواب دیا: ’یہ کون ہے؟‘
شاید یہی وجہ ہے کہ آج بہت سے مداح ان کے اصل نام سے واقف بھی نہیں۔ لیکن سوال یہ ہے کہ کرناٹک کے ایک نسبتاً چھوٹے شہر سے تعلق رکھنے والا یہ نوجوان کنڑ فلم انڈسٹری کا سب سے بڑا ستارہ کیسے بنا اور پھر اس نے انڈین سینما کی سب سے زیادہ زیرِ بحث فلموں میں سے ایک میں راون کا کردار ادا کرنے تک کا سفر کیسے طے کیا؟
End of سب سے زیادہ پڑھی جانے والی
یش فلموں میں کیسے آئے؟
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
یش کا تعلق کسی فلمی خاندان سے نہیں تھا۔ ان کے والد ایک بس ڈرائیور تھے۔ تاہم یش کو بچپن ہی سے اداکاری، ڈراموں اور سٹیج پرفارمنس کا شوق تھا، جس نے ان کے اندر فلمی ہیرو بننے کا خواب پیدا کر دیا۔
تعلیم مکمل کرنے کے بعد انھوں نے روایتی پیشہ اختیار کرنے کے بجائے اداکاری میں قسمت آزمانے کا فیصلہ کیا۔ اس مقصد کے لیے وہ تھیٹر اور ٹی وی کی دنیا کی طرف آئے اور آہستہ آہستہ اپنی صلاحیتوں کو نکھارنا شروع کیا۔
سنہ 2014 میں کنڑ ٹی وی پروگرام ویک اینڈ ود رمیش میں گفتگو کرتے ہوئے یش نے بتایا کہ وہ بچپن میں ڈراموں اور رقص کے مقابلوں میں بڑھ چڑھ کر حصہ لیتے تھے۔ ان کا کہنا تھا کہ سٹیج پر پرفارم کرنے کے بعد لوگوں کی تالیاں اور سیٹیاں سننا انھیں بے حد پسند تھا اور وقت کے ساتھ یہ شوق ایک خواب میں بدل گیا۔
یش کے مطابق جب ان کے اساتذہ انھیں مذاقاً یا حوصلہ افزائی کے طور پر ’ہیرو‘ کہہ کر پکارتے تھے تو انھیں بہت اچھا لگتا تھا۔ وہ کہتے ہیں کہ ’آہستہ آہستہ میرے ذہن میں ہیرو بننے کا خیال پیدا ہوا۔ پھر ایک وقت ایسا آیا جب میں نے فیصلہ کر لیا کہ مجھے زندگی میں فلمی اداکار ہی بننا ہے۔‘
یہی خواب انھیں کرناٹک کے ایک عام گھرانے سے اٹھا کر کنڑ سنیما کے سب سے بڑے ستاروں میں لے آیا، جہاں سے ان کا سفر ’کے جی ایف‘ جیسے بلاک بسٹر منصوبوں اور اب رامائن میں راون کے اہم کردار تک پہنچا ہے۔
تاہم ان کی ابتدائی زندگی مشکلات سے بھرپور تھی۔ ایک انٹرویو میں یش نے بتایا کہ ان کے والد بنگلورو کی میونسپل بس سروس میں ڈرائیور تھے اور روزانہ صرف 14 روپے کماتے تھے۔ ان کے خاندان میں والدین، وہ خود اور ان کی ایک چھوٹی بہن شامل تھے۔
گھریلو حالات کے باعث انھیں اپنی تعلیم ادھوری چھوڑنا پڑی اور خاندان کا ہاتھ بٹانے کے لیے کریانے کی دکان چلانے میں بھی مدد کرنا پڑتی تھی۔
جب یش نے فلم انڈسٹری میں قدم رکھنے کی خواہش ظاہر کی تو ان کے اہلِ خانہ اس فیصلے سے زیادہ خوش نہیں تھے۔ ان کے والد چاہتے تھے کہ وہ ایک سرکاری افسر بنیں۔
گھر سے نکلتے وقت جیب میں صرف 300 روپے تھے
سنہ 2019 میں دی نیوز منٹ کو دیے گئے ایک انٹرویو میں یش نے بتایا تھا کہ جب وہ قسمت آزمانے کے لیے گھر چھوڑ کر بنگلورو آئے تو ان کی جیب میں صرف 300 روپے تھے۔
انھوں نے کہا کہ ’جب میں بنگلورو پہنچا تو بہت خوف زدہ تھا۔ اتنا بڑا شہر! لیکن مجھے شروع ہی سے اپنے اوپر بھروسہ تھا۔ میں جدوجہد سے نہیں ڈرتا تھا۔ مجھے معلوم تھا کہ اگر واپس گھر چلا گیا تو میرے والدین مجھے دوبارہ یہاں آنے کی اجازت نہیں دیں گے۔‘
یش کے مطابق ان کے والدین نے انھیں واضح طور پر کہہ دیا تھا کہ اگر وہ ہیرو بننے کا خواب پورا کرنے کے لیے قسمت آزمانا چاہتے ہیں تو انھیں مکمل آزادی ہے، لیکن اگر کامیاب نہ ہوئے تو پھر انھیں وہی کرنا ہوگا جو ان کے والدین کہیں گے۔
تھیٹر نے مجھے ایک مختلف انسان بنا دیا
ویک اینڈ ود رمیش میں گفتگو کرتے ہوئے یش نے بتایا کہ ان کی زندگی میں تھیٹر کا دروازہ ان کے دوست ننجنڈا نے کھولا۔ اس سے پہلے ان کی ڈرامے سے واقفیت زیادہ تر دیومالائی کہانیوں پر مبنی سٹیج پرفارمنس تک محدود تھی، جن میں گدا ہاتھ میں لے کر لڑائی کے مناظر پیش کیے جاتے تھے۔ سماجی موضوعات پر مبنی ڈراموں اور جدید تھیٹر کی دنیا سے وہ تقریباً ناواقف تھے۔
یش کے مطابق ننجنڈا ہی وہ شخص تھے جنھوں نے انھیں تھیٹر سے متعارف کروایا اور فنِ اداکاری کو ایک نئے زاویے سے دیکھنے کا موقع دیا۔ اسی رفاقت کے ذریعے وہ تھیٹر کی سرگرمیوں سے جڑے اور اداکاری کے بارے میں ان کی سمجھ مزید گہری ہوئی۔
وہ یاد کرتے ہیں کہ جب وہ پہلی بار بینکا تھیٹر گروپ کا حصہ بنے تو ان کی سوچ اور نقطۂ نظر میں بڑی تبدیلی آئی۔ ان کے بقول تھیٹر نے انھیں نہ صرف اداکاری کے رموز سکھائے بلکہ زندگی اور معاشرے کو سمجھنے کا ایک نیا شعور بھی عطا کیا۔
یش کہتے ہیں، ’جب میں پہلی بار بینکا تھیٹر گروپ سے وابستہ ہوا تو میری دنیا ہی بدل گئی۔ مجھے لگتا ہے کہ تھیٹر نے مجھے ایک مختلف انسان بنا دیا۔ میری شخصیت کی تشکیل میں تھیٹر کا بہت اہم کردار ہے۔‘
شاید یہی وجہ ہے کہ یش آج بھی اپنی کامیابی کی بنیاد فلموں سے پہلے تھیٹر کو قرار دیتے ہیں، کیونکہ ان کے نزدیک ایک بس ڈرائیور کے بیٹے کو ’راکنگ سٹار‘ بنانے میں تھیٹر نے فیصلہ کن کردار ادا کیا۔
پہلا فلمی موقع اتفاقاً ملا
سنہ 2007 میں فلم جمباڈا ہدوگی کے ذریعے یش نے بطور معاون اداکار اپنی فلمی اننگز کا آغاز کیا۔ اس سے قبل وہ اتّرایانا اور نندا گوکلا جیسے ٹی وی ڈراموں میں کام کر چکے تھے۔ خاص طور پر نندا گوکلا ایک مقبول سیریل ثابت ہوا، جس نے یش کو ناظرین میں پہچان دلائی۔ اس ڈرامے میں ان کی اور رادھیکا پنڈت کی جوڑی کو بے حد پسند کیا گیا، اور یہی مقبولیت بعد میں ان کے فلمی کریئر کے لیے ایک مضبوط بنیاد بنی۔ یہ سیریل 2004 سے 2007 کے درمیان نشر ہوا۔
اگرچہ یش فلمی دنیا میں قدم رکھ چکے تھے، لیکن انھیں حقیقی شہرت 2008 میں ریلیز ہونے والی فلم موگینا مناسو سے ملی۔ دلچسپ بات یہ ہے کہ اس فلم میں ان کا کردار ابتدا میں کسی اور اداکار کو دیا گیا تھا، لیکن شوٹنگ شروع ہونے سے قبل وہ موٹر سائیکل حادثے میں زخمی ہو گئے اور کام جاری نہ رکھ سکے۔ یوں فلم سازوں کو متبادل اداکار کی تلاش کرنا پڑی اور یہ موقع یش کے حصے میں آ گیا۔
بعد ازاں ویک اینڈ ود رمیش میں یش نے اس واقعے کو اپنی زندگی کا اہم موڑ قرار دیا۔ اس فلم نے نہ صرف انھیں بڑے پردے پر نمایاں مقام دلایا بلکہ ان کی صلاحیتوں کو بھی صنعت کے سامنے اجاگر کر دیا۔
فلم کے ہدایتکار ششانک کے مطابق انھوں نے یش کو ایک ٹی وی سیریل میں دیکھا تھا اور ان کی اداکاری سے متاثر ہوئے تھے۔ ان کے بقول، جب وہ موگینا مناسو پر کام کر رہے تھے تو اس کردار کے لیے سب سے پہلا نام یش ہی کا ذہن میں آیا۔ ششانک کا کہنا تھا کہ یش اپنے کردار کو سمجھنے کے لیے غیر معمولی محنت کرتے تھے اور ہر منظر کے مختلف انداز تجویز کر کے اپنی لگن کا ثبوت دیتے تھے۔
یہ محنت رنگ لائی اور موگینا مناسو باکس آفس پر کامیاب ثابت ہوئی۔ اس فلم میں اپنی شاندار کارکردگی کے اعتراف میں یش کو فلم فیئر ایوارڈ برائے بہترین معاون اداکار بھی ملا۔ یہی وہ کامیابی تھی جس نے ایک ابھرتے ہوئے ٹی وی اداکار کو کنڑ سنیما کے مستقبل کے سپر سٹار بننے کی راہ پر گامزن کر دیا۔
کے جی ایف نے پورے انڈیا میں شناخت دلائی
فلمی ناقد اور پروڈیوسر سناینا سریش کے مطابق یش کے کریئر کو اہم موڑ دینے میں چند فلموں نے کلیدی کردار ادا کیا۔
ان کے مطابق یش کی پہلی نمایاں فلم ’موگینا مناسو‘ تھی، جس کے لیے انھیں فلم فیئر ایوارڈ بھی ملا اور فلمی دنیا میں ایک الگ شناخت بھی حاصل ہوئی۔
اس کے بعد ’گوگلی‘ ریلیز ہوئی، جس نے انھیں نوجوانوں، خصوصاً کالج کے طلبہ میں بے حد مقبول بنا دیا۔ سناینا سریش کے بقول یہی وہ فلم تھی جس نے یش کو ایک بڑے سٹار کے طور پر منوایا۔
اس کے بعد مسٹر اینڈ مسز راماچاری نے یش کو سپر سٹار بنانے میں اہم کردار ادا کیا۔ اس فلم میں ان کے ساتھ ان کی اہلیہ رادھیکا پنڈت بھی تھیں، اور یہ کنڑ فلم انڈسٹری کی بڑی ہٹ ثابت ہوئی۔ یش کے کریئر میں موگینا مناسو اور مسٹر اینڈ مسز راماچاری نے فیصلہ کن کردار ادا کیا۔
تاہم ’کے جی ایف‘ نے انھیں دوسرے اداکاروں سے منفرد مقام دلایا۔
فلمی تجزیہ کار سناینا سریش کے مطابق، ’کے جی ایف‘ کے ذریعے یش کو ہندی فلم بینوں میں بھی وسیع پہچان ملی، خاص طور پر اتر پردیش اور بہار جیسے ہندی بولنے والے علاقوں میں، جہاں ناظرین کو ایک ایسے ہیرو کی تلاش تھی جس کی کمی محسوس کی جا رہی تھی۔‘
راون کا کردار
کامک کان 2026 میں جب یش سے راون کے کردار کے بارے میں پوچھا گیا تو انھوں نے اسے ایک نہایت پیچیدہ اور چیلنجنگ کردار قرار دیا۔
انھوں نے کہا کہ ’ذرا سوچیں، ایک ایسا کردار جس سے لوگ اس قدر نفرت کرتے ہیں کہ ہر سال اس کا پتلا جلاتے ہیں۔ اس سے اندازہ ہوتا ہے کہ اس کردار نے لوگوں کے ذہنوں میں کتنی مضبوط شناخت قائم کر رکھی ہے۔‘
یش کے مطابق راون کا کردار ادا کرنا اس لیے بھی مشکل ہے کہ تقریباً ہر شخص اس سے واقف ہے اور اس کی ایک مخصوص شبیہ پہلے ہی عوام کے ذہنوں میں موجود ہے۔
انھوں نے مزید کہا کہ ’سب لوگ اس کردار کو جانتے ہیں، اس لیے اسے پردۂ سکرین پر پیش کرنا اور بھی زیادہ چیلنجنگ ہو جاتا ہے۔‘
وہ مزید کہتے ہیں کہ ’اگر کوئی نیا کردار ہو اور آپ پہلی بار اس کی کہانی سن رہے ہوں تو آپ اس کردار کو اپنی تشریح کے مطابق تشکیل دے سکتے ہیں۔ لیکن جب کسی کردار کے بارے میں لوگوں کے ذہنوں میں پہلے سے ایک مضبوط تصور موجود ہو اور ماضی میں متعدد اداکار اسے نبھا چکے ہوں تو نئے اداکار پر ذمہ داری اور دباؤ کہیں زیادہ بڑھ جاتا ہے۔‘
یش نے کہا کہ بطور اداکار انھوں نے یہ سمجھنے کی کوشش کی کہ راون نے جو کچھ کیا، وہ کیوں کیا اور اس کے پیچھے اس کا نقطۂ نظر کیا تھا۔
ان کے بقول، ’میرے خیال میں راون کو پردۂ سکرین پر پیش کرنے کا بہترین طریقہ یہی ہے کہ اس کی سوچ، محرکات اور اعمال کے پس منظر کو سمجھا جائے۔‘
