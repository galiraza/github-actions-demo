# ایران کے خلاف جنگ سمیت وہ پانچ وجوہات جو ٹرمپ کی ریپبلکن پارٹی کی شکست کا باعث بنیں

ایران کے خلاف جنگ سمیت وہ پانچ وجوہات جو ٹرمپ کی ریپبلکن پارٹی کی شکست کا باعث بنیں

امریکہ کے وسط مدتی انتخابات میں ڈیموکریٹس نے بڑی کامیابی حاصل کرتے ہوئے ایوانِ نمائندگان اور سینیٹ دونوں کا کنٹرول حاصل کر لیا ہے۔
اگرچہ ان انتخابات میں موجودہ صدر ٹرمپ کی جماعت کو عموماً مشکلات کا سامنا کرنا پڑتا ہے تاہم ریپبلکنز کی شکست کے حجم کو وائٹ ہاؤس میں ڈونلڈ ٹرمپ کے دو سالہ دور کے خلاف واضح ردعمل کے طور پر دیکھا جا رہا ہے۔
غصے سے بھرے ووٹرز سے لے کر ایک غیر مقبول صدر تک، یہ ہیں وہ پانچ وجوہات جن کی بنا پر ریپبلکنز نے اتنی زیادہ نشستیں گنوائیں۔۔۔
1۔ روزمرہ کے اخراجات میں اضافہ
امریکہ بھر میں ووٹرز کو روزمرہ کے اخراجات میں اضافے کا سامنا ہے۔ اہم ریاستوں میں یہ غصہ واضح طور پر محسوس کیا گیا۔
شمالی کیرولائنا کی ایک ووٹر نے انتخابات سے قبل بی بی سی کو بتایا کہ ’پیٹرول کی قیمتیں، اشیائے خورد و نوش کی قیمتیں، سب کچھ مہنگا ہو رہا ہے۔‘
انھوں نے مزید کہا کہ اس انتخاب میں انھوں نے اپنا ووٹ ریپبلکنز سے ڈیموکریٹس کی جانب منتقل کر دیا۔
ستمبر تک کے ایک سال میں قیمتوں میں 3.7 فیصد اضافہ ہوا جس کی بنیادی وجہ ایندھن اور توانائی کی بڑھتی ہوئی قیمتیں تھیں۔
سرکاری اعداد و شمار کے مطابق اکتوبر کے آغاز میں امریکہ میں ڈیزل کی اوسط خوردہ قیمت تقریباً 6.52 ڈالر فی گیلن کے اُس ریکارڈ کے برابر تھی جو ستمبر میں دیکھا گیا تھا۔
End of سب سے زیادہ پڑھی جانے والی
ووٹر پولز کے اعداد و شمار کے مطابق تقریباً ہر 10 میں سے آٹھ ووٹرز (79 فیصد) نے کہا کہ افراطِ زر نے گذشتہ ایک سال کے دوران انھیں اور ان کے خاندان کو شدید یا درمیانے درجے کی مشکلات سے دوچار کیا۔
اور ہر 10 میں سے چھ سے زیادہ افراد نے اس کا ذمہ دار ٹرمپ کو قرار دیا اور کہا کہ غیر معمولی طور پر بلند قیمتیں صدر کی پالیسیوں کا نتیجہ ہیں۔
2۔ ایران کے خلاف جنگ جو غیر مقبول ہوئی
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
قیمتوں میں اضافے کا ایک بڑا سبب ایران کے ساتھ وہ جنگ رہی ہے جو امریکہ اور اسرائیل نے فروری میں شروع کی تھی۔
اس کے بعد تہران نے جوابی کارروائی میں آبنائے ہرمز کو عملی طور پر بند کر دیا، جو دنیا کے تیل اور مائع قدرتی گیس (ایل این جی) کی ترسیل کی ایک اہم گزرگاہ ہے۔ اس کے نتیجے میں ایندھن کی قیمتیں تیزی سے بڑھ گئیں۔
تب سے امریکی ووٹرز پیٹرول پمپ پر اس کی قیمت ادا کر رہے ہیں۔
یہ جنگ مجموعی طور پر بیشتر امریکیوں میں غیر مقبول ثابت ہوئی ہے۔
ٹرمپ کے حامی ریپبلکنز کے لیے یہ ’امریکہ فرسٹ‘ کے انتخابی نعرے سے انحراف ثابت ہوا۔ ٹرمپ نے وعدہ کیا تھا کہ مزید جنگیں نہیں ہوں گی۔
ایک ووٹر پول میں شامل تقریباً ہر 10 میں سے چھ افراد (63 فیصد) نے کہا کہ امریکہ نے ایران کے خلاف فوجی کارروائی کر کے غلط فیصلہ کیا۔
تین چوتھائی ووٹرز (74 فیصد) نے کہا کہ یہ تنازع کانگریس کے لیے ان کے ووٹ کے فیصلے میں ایک اہم یا بہت بڑا عنصر تھا۔
ٹرمپ نے بھی تسلیم کیا کہ اس جنگ نے وسط مدتی انتخابات میں ریپبلکنز کی شکست میں کردار ادا کیا۔ انھوں نے فاکس نیوز کو بتایا کہ جب لوگوں کو ایندھن کے لیے زیادہ قیمت ادا کرنا پڑی تو انھیں یہ بات ’پسند نہیں آئی۔‘
ڈیموکریٹس کا منظم پیغام
ڈیموکریٹس کی اس وسیع کامیابی کو ممکن بنانے میں کچھ مضبوط امیدواروں کا بھی کردار تھا، جو خاص طور پر روایتی ریپبلکن علاقوں میں حقیقی اور قابلِ اعتبار نظر آئے۔
انھوں نے روزمرہ اخراجات اور صحت کی سہولتوں تک رسائی کے بارے میں ایک منظم اور واضح پیغام اپنائے رکھا۔
سابق استاد جیمز ٹالاریکو اور پادری ایڈم ہیملٹن نے بھی اپنے مذہبی عقائد کے بارے میں اس انداز میں بات کی جو بالترتیب ٹیکساس اور کنساس میں ووٹرز کو متاثر کرتی نظر آئی۔
جنوبی کیرولائنا میں ڈیموکریٹ اینی اینڈریوز نے غیر متوقع کامیابی حاصل کی۔
سابق ماہرِ اطفال ہونے کے ناطے، ان کی توجہ صحت کی سہولتوں کو قابلِ برداشت بنانے پر رہی، جو ایسے ملک میں اثر انداز ہوئی جہاں بہت سے لوگ نظام کو ناکام سمجھتے ہیں۔
صحت عامہ کے سابق عہدیدار عبدل السید، جنھوں نے مشیگن میں کامیابی حاصل کر کے یہ نشست ڈیموکریٹس کے پاس برقرار رکھی، نے ’پیسے کو سیاست کی بجائے آپ کی جیب میں ڈالا جائے‘ کا نعرہ لگایا۔
جارجیا سے دوبارہ منتخب ہونے والے ڈیموکریٹ سینیٹر جان اوسوف نے روزمرہ اخراجات اور بدعنوانی کو اپنی مہم کا مرکزی پیغام بنایا۔
39 سالہ امیدوار کی اس ریاست میں نسبتاً آسان کامیابی، جسے ٹرمپ دو بار جیت چکے ہیں، 2028 کے صدارتی انتخاب میں ممکنہ امیدوار بننے کی قیاس آرائیوں کو مزید تقویت دے گی۔
خود ٹرمپ نے بھی ڈیموکریٹس کی تعریف کرتے ہوئے فاکس نیوز سے کہا کہ انھوں نے ’اچھا کام کیا۔‘
اس کے برعکس، ٹیکساس میں کین پیکسٹن جیسے بعض ریپبلکن امیدواروں کو اس بات پر تنقید کا سامنا کرنا پڑا کہ ان کی انتخابی مہمات ووٹرز کے مسائل پر توجہ نہیں دے سکیں۔
ٹرمپ کی غیر مقبولیت ریپبلکنز کے لیے بوجھ بنی
جن دو مسائل سے ووٹرز سب سے زیادہ ناخوش تھے، یعنی بے قابو مہنگائی اور ایران کے خلاف جنگ، ان دونوں کا براہِ راست تعلق صدر سے جوڑا جا رہا تھا۔
ریپبلکن ووٹرز میں ان کی مقبولیت کی شرح میں کمی آئی تھی، جس کے باعث وہ ریاستیں جو 2024 میں ان کی حامی رہی تھیں خاص طور پر کمزور ہو گئیں۔
انھوں نے ووٹرز سے کہا کہ وہ تصور کریں کہ گویا ان کا نام بھی بیلٹ پیپر پر موجود ہے، تاکہ اپنے حامیوں کو متحرک کر سکیں۔
تاہم ایک دہائی سے زیادہ عرصے تک امریکی سیاست کی مرکزی شخصیت رہنے کے بعد، بعض لوگ ان نتائج کو اس بات کا اشارہ سمجھتے ہیں کہ ٹرمپ کے سیاسی برانڈ کی چمک ماند پڑ رہی ہے۔
بڑھتے ہوئے عوامی غصے کے ماحول میں، خود کو ریپبلکن مہم کا مرکز بنانا ایک خطرناک حکمتِ عملی تھی، اور بظاہر اس کا الٹا اثر ہوا۔
ووٹنگ سے پہلے کے مہینوں میں کیے گئے سرویز سے ظاہر ہوا تھا کہ بہت سے امریکی چاہتے تھے کہ ٹرمپ مہنگائی کے بارے میں ان کے خدشات پر زیادہ توجہ دیں۔
پیر کے روز واشنگٹن ڈی سی میں اپنی آخری انتخابی تقریر میں انھوں نے اپنے پیچھے واقع ریفلیکٹنگ پول کی متنازع تزئین و آرائش کے منصوبے پر تفصیل سے بات کی۔
ہسپانوی، سیاہ فام ووٹرز کی ڈیموکریٹ پارٹی کی طرف واپسی
دو سال قبل بعض ہسپانوی اور سیاہ فام ووٹرز کے ریپبلکن پارٹی کی جانب جھکاؤ نے ٹرمپ کو فیصلہ کن کامیابی دلانے میں مدد دی تھی۔
مگر وسط مدتی انتخابات میں یہ رجحان بدلتا ہوا دکھائی دیا۔
ڈیموکریٹس نے ٹیکساس اور میکسیکو کی سرحد کے ساتھ واقع نمایاں طور پر ہسپانوی آبادی والے اضلاع میں ایوانِ نمائندگان کی تینوں نشستیں جیت لیں۔
نیویارک ٹائمز کے تجزیے کے مطابق، 2024 کے صدارتی انتخاب کے مقابلے میں ہسپانوی اکثریت والے اضلاع سینیٹ کے ریپبلکن امیدواروں کے خلاف 22 فیصد پوائنٹس سے زیادہ منتقل ہوئے۔
اعداد و شمار کے رپورٹر جیف ایڈلسن کے مطابق ملک کی کسی بھی دوسری بڑی آبادیاتی جماعت کے مقابلے میں یہ سب سے بڑی تبدیلی تھی۔
یہ تبدیلی امیگریشن قوانین کے نفاذ پر وسیع عوامی ناراضی کے پس منظر میں سامنے آئی، جہاں پیو ریسرچ سینٹر کے مطابق امریکیوں کی اکثریت (56 فیصد) نے آئی سی ای کے بارے میں منفی رائے کا اظہار کیا۔
سیاہ فام اکثریت والے اضلاع بھی سینیٹ کے انتخابات میں نمایاں طور پر ڈیموکریٹ امیدواروں کی جانب مائل ہوئے۔
مسسپی میں سینیٹ کی دوڑ انتہائی سخت مقابلے کی شکل اختیار کر چکی ہے۔ یہ وہ امریکی ریاست ہے جہاں سیاہ فام ووٹرز کا تناسب سب سے زیادہ تقریباً 37 فیصد ہے۔
اگر سکاٹ کولم کامیاب ہو جاتے ہیں تو وہ 1982 کے بعد مسسپی سے سینیٹ کے لیے منتخب ہونے والے پہلے ڈیموکریٹ بن جائیں گے۔
