# کچھ لوگوں کے لیے صرف چار گھنٹے کی نیند بھی کافی کیوں ہوتی ہے؟

کچھ لوگوں کے لیے صرف چار گھنٹے کی نیند بھی کافی کیوں ہوتی ہے؟

ایمی پگ برنارڈ ان افراد میں شامل ہیں جنھیں کبھی زیادہ نیند کی ضرورت نہیں رہی۔
آج بھی وہ عموماً رات 10 بجے کے قریب سونے جاتی ہیں اور صبح سویرے ساڑھے چار بجے جاگ جاتی ہیں۔
وہ کہتی ہیں کہ ’اس وقت میں پوری طرح بیدار ہوتی ہوں اور مزید بستر میں نہیں رہ سکتی۔‘
یونیورسٹی آف کولوراڈو میں امیونولوجی کی پروفیسر برنارڈ کو بعض اوقات فوراً نیند بھی نہیں آتی۔ ایسے مواقع پر وہ بمشکل پانچ گھنٹے ہی سو پاتی ہیں۔
شاید سب سے حیران کن بات یہ ہے کہ اس کے باوجود برنارڈ اگلے دن بھی خود کو توانا محسوس کرتی ہیں۔ وہ اپنے کتے کو گھمانے لے جاتی ہیں، ایک گھنٹے تک جم میں ورزش کرتی ہیں اور صبح آٹھ بجے سے پہلے ڈیڑھ گھنٹے کام بھی کر چکی ہوتی ہیں۔
ان کے دوست کم نیند کے باوجود اتنا کچھ کر لینے کی ان کی صلاحیت پر حیرت کا اظہار کرتے ہیں۔ تاہم ان کے خاندان میں یہ کوئی غیرمعمولی بات نہیں کیونکہ ان کے والد اور بھائی بھی ایسے ہی ہیں۔
البتہ ہم میں سے بیشتر لوگوں کو اس سے زیادہ نیند درکار ہوتی ہے۔
امریکی نیشنل سلیپ فاؤنڈیشن کی جانب سے شواہد کے حالیہ جائزے کے مطابق 26 سے 64 سال عمر کے ایک اوسط بالغ فرد کو ہر 24 گھنٹوں میں تقریباً سات سے نو گھنٹے نیند کی ضرورت ہوتی ہے۔
End of سب سے زیادہ پڑھی جانے والی
یہ سوال طویل عرصے سے ایک معمہ بنا ہوا ہے کہ مختلف لوگوں کی نیند کی ضرورت میں اتنا بڑا فرق کیوں پایا جاتا ہے۔
تاہم گذشتہ چند برسوں میں سائنس دانوں نے متعدد ایسے طریقوں کی نشاندہی کی ہے جن سے معلوم ہوتا ہے کہ اس معاملے میں جینیات اہم کردار ادا کرتی ہیں۔
برنارڈ جیسے افراد میں پائی جانے والی نایاب جینیاتی تبدیلیوں سے لے کر سینکڑوں نسبتاً عام جینیاتی اقسام تک، محققین اب یہ سمجھنے کی کوشش کر رہے ہیں کہ بعض لوگ بہت کم نیند پر بھی کیسے گزارا کر لیتے ہیں۔
مستقبل میں ان نتائج کی مدد سے ایسے لوگوں کے تحفظ میں بھی مدد مل سکتی ہے جنھیں اپنی ضرورت کے مطابق نیند نہیں مل پاتی۔
پہلا اشارہ
بی بی سی اردو کی خبروں اور فیچرز کو اپنے فون پر حاصل کریں اور سب سے پہلے جانیں پاکستان اور دنیا بھر سے ان کہانیوں کے بارے میں جو آپ کے لیے معنی رکھتی ہیں
سبسکرائب کرنے کے لیے کلک کریں
مواد پر جائیں
سنہ 2019 میں برنارڈ کو اپنی اس غیرمعمولی صلاحیت کی ممکنہ وجہ معلوم ہوئی۔
سان فرانسسکو کی یونیورسٹی آف کیلیفورنیا کی ایک تحقیقی ٹیم نے انسانوں میں نیند کے دورانیے سے متعلق ایک جینیاتی تبدیلی دریافت کی تھی۔
اس جینیاتی تبدیلی کے حامل افراد اگرچہ نسبتاً معمول کے اوقات میں سونے جاتے تھے، لیکن بہت جلد بیدار ہو جاتے تھے اور اکثر ایک رات میں صرف چار سے ساڑھے چھ گھنٹے نیند لیتے تھے۔
برنارڈ نے تحقیقاتی ٹیم کو خون کا نمونہ بھیجا تو معلوم ہوا کہ وہ بھی انھی افراد میں شامل ہیں۔
اس نایاب موروثی کیفیت کو فیمیلیئل نیچرل شارٹ سلیپ (FNSS) کہا جاتا ہے۔ اس کے بعد کے برسوں میں ایسی کئی دیگر جینیاتی تبدیلیوں کی بھی نشاندہی کی گئی جو اس کیفیت سے منسلک ہیں۔
ایف این ایس ایس کے بارے میں سب سے بڑے سوالوں میں سے ایک یہ ہے کہ اس کے حامل افراد اتنی کم نیند کے باوجود بظاہر بالکل ٹھیک رہتے ہیں۔
تحقیقی ٹیم کی سربراہ اور مالیکیولر بایولوجسٹ ینگ ہوئی فو کے مطابق اس معمہ کی مکمل وضاحت ابھی سامنے نہیں آ سکی کہ ایسا کیوں ہوتا ہے؟
تاہم اس بات کے دلچسپ شواہد موجود ہیں کہ ایف این ایس ایس سے وابستہ جینیاتی تبدیلیاں نہ صرف کم نیند کا سبب بنتی ہیں بلکہ ان افراد کو کم نیند کے منفی اثرات سے بھی محفوظ رکھ سکتی ہیں۔
ان اثرات میں یادداشت کی کمزوری بھی شامل ہے۔
فو کی قیادت میں 2019 میں کیے گئے ایک تجربے میں ایف این ایس ایس سے منسلک ایک جینیاتی تبدیلی چوہوں میں منتقل کی گئی۔
محققین نے دیکھا کہ اس سے متعلقہ جینیاتی قسم رکھنے والے چوہے نہ صرف کم سوتے تھے بلکہ وہ یادداشت سے متعلق مسائل سے بھی کسی حد تک محفوظ دکھائی دیتے تھے۔
ایک اور تحقیق میں معلوم ہوا کہ ان میں سے بعض جینیاتی اقسام رکھنے والے چوہوں میں الزائمر کی چوہوں میں پائی جانے والی شکل پیدا ہونے کا امکان بھی کم تھا۔
سنہ 2014 میں ایک بین الاقوامی تحقیقی ٹیم نے جڑواں بچوں کے ایک جوڑے کا مطالعہ کیا جن میں سے ایک کے اندر کم نیند سے متعلق جینیاتی تبدیلی موجود تھی جبکہ دوسرے میں نہیں۔
محققین کے سامنے آیا کہ اس جینیاتی تبدیلی کا حامل جڑواں اپنے بہن بھائی کے مقابلے میں باقاعدگی سے تقریباً ایک گھنٹہ کم سوتا تھا۔
انھوں نے یہ بھی دیکھا کہ جب دونوں جڑواں بچوں کو مسلسل 38 گھنٹے تک جاگتے رکھا گیا تو جینیاتی تبدیلی کے حامل فرد میں کارکردگی میں کمی کے واقعات دوسرے جڑواں کے مقابلے میں تقریباً 40 فیصد کم تھے۔
محققین نے دونوں جڑواں بچوں کی نیند کے انداز میں کچھ اور دلچسپ فرق بھی نوٹ کیے۔
اگرچہ جینیاتی تبدیلی کا حامل بچہ کم سوتا تھا، لیکن اسے تقریباً اپنے جڑواں بھائی کے برابر ہی نان ریپڈ آئی موومنٹ (Non-REM) نیند حاصل ہوتی تھی۔
اس دوران وہ بچہ گہری نیند لیتا بھی پایا گیا جس سے یہ اشارہ ملتا ہے کہ ممکن ہے وہ اپنی نسبتاً کم نیند سے بھی بعض حوالوں سے زیادہ فائدہ حاصل کر رہا تھا۔
ادویات کے ذریعے نیند
تاہم اس شعبے میں تحقیق اب بھی جاری ہے اور ینگ ہوئی فو کے مطابق یہ تحقیق انتہائی اہم ہے۔
گذشتہ چند برسوں میں اس بات میں دلچسپی بڑھی ہے کہ ان جینیاتی تبدیلیوں کے بارے میں حاصل ہونے والی معلومات کو لوگوں کی مدد کے لیے کیسے استعمال کیا جا سکتا ہے۔
ادویہ سازی کی صنعت میں یہ امید پائی جاتی ہے کہ ایک دن کم نیند لینے والے افراد پر ہونے والی تحقیقات ایسی ادویات کی تیاری کا سبب بن سکتی ہیں جو نیند کے مسائل کے علاج میں مدد دیں یا مثلاً فوجی اہلکاروں کو کم نیند کے باوجود بہتر کارکردگی دکھانے کے قابل بنا سکیں۔
تاہم اس مقصد کے لیے سائنس دانوں کو نہ صرف یہ واضح طور پر سمجھنے کی ضرورت ہے کہ آیا یہ جینیاتی تبدیلیاں کم نیند لینے والے افراد کو اس کے منفی اثرات سے محفوظ رکھتی ہیں یا نہیں، بلکہ یہ بھی جاننا ضروری ہے کہ وہ ایسا کس طریقے سے کرتی ہیں۔
اس معلومات کے بغیر ممکن ہے ایسی دوا تیار ہو جائے جس کے غیر متوقع مضر اثرات سامنے آئیں۔
فو کہتی ہیں کہ ’آپ کو یہ سمجھنا ہو گا کہ یہ نظام کیسے کام کرتا ہے۔‘
ان کا دعویٰ ہے کہ ’آپ یہ نہیں چاہیں گے کہ آپ کم نیند لیں اور پھر پانچ، 10 یا 20 سال بعد الزائمر، پارکنسنز یا کینسر جیسے مرض کا شکار ہو جائیں۔‘
ماہرین کا اندازہ ہے کہ مجموعی طور پر ایک شخص کتنی دیر سوتا ہے، اس میں 30 سے 65 فیصد تک کردار موروثی عوامل کا ہو سکتا ہے۔
یعنی مختلف افراد کی نیند کے دورانیے میں پائے جانے والے فرق کا 65 فیصد تک حصہ جینیات سے سمجھا جا سکتا ہے۔
اب تک سائنس دان ڈی این اے کے سینکڑوں ایسے حصوں کی نشاندہی کر چکے ہیں جو اس بات پر اثر انداز ہوتے ہیں کہ ایک شخص کتنی دیر سوتا ہے۔
لندن کی کنگز کالج کی جینیات دان اِزابیل آسٹن-زمرمین اور ان کے دیگر محققین کے مطابق ایک اوسط فرد کے لیے جینیاتی فرق نیند کے دورانیے میں تقریباً 20 منٹ کے فرق کی وضاحت کر سکتا ہے۔
تاہم اب تک اس بارے میں زیادہ تحقیق نہیں ہوئی کہ ان نسبتاً عام جینیاتی تبدیلیوں میں سے کون سی ایسی ہیں جو لوگوں کو کم نیند کے منفی اثرات سے محفوظ رکھنے میں مدد دیتی ہیں۔
البتہ یونیورسٹی کالج لندن سے وابستہ ماہر زچہ و بچہ امراض ڈاکٹر کیارا پیٹروسیلینی کی سربراہی میں ہونے والی ایک حالیہ تحقیق میں ایسی ایک مثال کے شواہد سامنے آئے ہیں۔
تحقیقی ٹیم نے بائی پولر ڈس آرڈر میں مبتلا 343 ایسی خواتین کے جینیاتی ڈیٹا کا جائزہ لیا جو ماں بن چکی تھیں۔
ان خواتین میں کم نیند لینے والے افراد سے وابستہ سمجھی جانے والی جینیاتی تبدیلیوں کی موجودگی کی بنیاد پر ان کے کم سونے کے امکان کا اندازہ لگایا گیا۔
نتائج سے معلوم ہوا کہ جن خواتین میں کم نیند لینے والے افراد سے متعلق جینیاتی خصوصیات زیادہ پائی گئیں، ان میں بچے کی پیدائش کے بعد نفسیاتی عارضے (پوسٹ پارٹم سائیکوسِس) کا خطرہ ان خواتین کے مقابلے میں نصف تھا جن میں ایسی خصوصیات کم تھیں۔
بے خوابی کے معاملے میں بھی یہی رجحان دیکھا گیا۔
محققین کا کہنا ہے کہ اس حوالے سے مزید تحقیق کی ضرورت ہے تاہم ممکن ہے کہ ایسے افراد زندگی کے ابتدائی مرحلوں ہی سے نیند میں خلل برداشت کرنے کی زیادہ صلاحیت پیدا کر لیتے ہوں۔
یہی وجہ ولادت کے بعد نیند کے متاثر ہونے کے ان کی ذہنی کیفیت پر پڑنے والے اثرات کو کم کر سکتی ہے۔
تاہم کم نیند سے وابستہ جینیاتی تبدیلیاں ضروری نہیں کہ ہر صورت میں فائدہ پہنچائیں۔
درحقیقت آسٹن-زمرمین اور ان کی ٹیم کی ایک بڑی آبادی پر کی گئی تحقیق، جس میں صرف بائی پولر ڈس آرڈر میں مبتلا ماؤں کو شامل نہیں کیا گیا تھا، سے معلوم ہوا کہ مختصر نیند سے منسلک جینیاتی تبدیلیاں عمومی طور پر بے خوابی اور تھکاوٹ محسوس کرنے کے زیادہ خطرے سے وابستہ تھیں۔
یہ جینیاتی خصوصیات بعض ذہنی اور جسمانی مسائل سے بھی وابستہ پائی گئیں، جن میں اے ڈی ایچ ڈی، ڈپریشن کی علامات اور ٹائپ ٹو ذیابیطس شامل ہیں۔
تاہم اس کا یہ مطلب نہیں کہ کم نیند یا اس سے متعلق جینیاتی خصوصیات براہِ راست ان بیماریوں کا سبب بنتی ہیں۔
البتہ اس تحقیق سے ظاہر ہوتا ہے کہ کم نیند سے وابستہ جینیاتی عوامل انسان کو ان خطرات سے ہمیشہ محفوظ نہیں رکھتے۔
کم نیند کے باوجود خود کو ٹھیک سمجھنے کا فریب
تاہم معاملہ اتنا سادہ بھی نہیں۔ کم نیند کے باوجود خود کو ٹھیک محسوس کرنے والے افراد بھی ہمیشہ اپنی حالت کا درست اندازہ نہیں لگا پاتے۔
مثال کے طور پر ایک تحقیق میں معلوم ہوا کہ جب صحت مند بالغ افراد کی نیند کو محدود کر کے انھیں روزانہ صرف چار یا چھ گھنٹے بستر میں رہنے کا موقع دیا گیا تو مختلف ذہنی صلاحیتوں کے ٹیسٹوں میں ان کی کارکردگی مسلسل خراب ہوتی گئی۔
تاہم دو ہفتوں کے بعد بھی شرکا کا کہنا تھا کہ انھیں صرف معمولی سی غنودگی محسوس ہو رہی ہے۔
اس تحقیق کے شریک مصنف اور واشنگٹن سٹیٹ یونیورسٹی کے سلیپ اینڈ پرفارمنس ریسرچ سینٹر سے وابستہ نیند کے ماہر ہانس وین ڈونگن کہتے ہیں کہ انسان کی کارکردگی بتدریج متاثر ہوتی جاتی ہے، لیکن ساتھ ہی وہ اس خرابی سے بھی کم آگاہ رہتا ہے۔
اگرچہ یہ بات درست ہے کہ بعض لوگوں کو دوسروں کے مقابلے میں کم نیند کی ضرورت ہوتی ہے۔
وین ڈونگن کی دوسری تحقیقات سے بھی معلوم ہوا ہے کہ بعض افراد نیند کی کمی کو دوسروں کے مقابلے میں بہتر انداز میں برداشت کر سکتے ہیں۔
تاہم اس کا مطلب یہ بھی ہے کہ لوگوں کے درمیان موجود ان فرقوں کو سمجھنا محض یہ جان لینے سے کہیں زیادہ پیچیدہ ہے کہ کوئی شخص کتنے گھنٹے سوتا ہے یا اگلے دن خود کو کیسا محسوس کرتا ہے۔
طویل مدت میں آسٹن-زمرمین اور فو جیسے محققین کی جینیاتی تحقیقات یہ سمجھنے میں مدد دے سکتی ہیں کہ کون سے لوگ واقعی کم نیند پر بھی ٹھیک رہتے ہیں۔
یہ نتائج ان لوگوں کی نشاندہی میں بھی مددگار ثابت ہو سکتے ہیں جو خود کو تو ٹھیک سمجھتے ہیں لیکن درحقیقت ان کی کارکردگی متاثر ہونے کا خطرہ زیادہ ہوتا ہے۔
محققین کو امید ہے کہ مستقبل میں ان تحقیقات کی بنیاد پر ایسی ادویات تیار کی جا سکیں گی جو کم نیند لینے والے افراد کو اس کے اثرات بہتر طور پر برداشت کرنے میں مدد دیں۔
تاہم فی الحال یہ تحقیقات لوگوں کو یہ سمجھنے میں مدد دے سکتی ہیں کہ ان کی نیند کے انداز دوسروں سے مختلف کیوں ہیں۔
برنارڈ کہتی ہیں کہ جب انھیں معلوم ہوا کہ وہ ایف این ایس ایس سے وابستہ جینیاتی تبدیلیوں میں سے ایک کی حامل ہیں تو ان پر سے کافی ذہنی دباؤ کم ہو گیا۔
ایک سائنس دان ہونے کے ناطے وہ پہلے ہی جانتی تھیں کہ انسانوں کے درمیان کئی فطری فرق پائے جاتے ہیں، جن میں نیند بھی شامل ہے۔
وہ یہ بھی سمجھتی ہیں کہ کم نیند کی وجہ سے انھیں دن کے آغاز میں اضافی وقت مل جاتا ہے، خاص طور پر ایسے وقت میں جب ان کے شوہر اور بچے ابھی سو رہے ہوتے ہیں۔
اس کے باوجود برنارڈ کبھی کبھی یہ سوچتی تھیں کہ شاید وہ آرام کرنے میں غیرمعمولی طور پر کمزور ہیں یا پھر زیادہ دیر تک سو نہ پانا اس بات کی علامت ہے کہ وہ اپنی زندگی میں بہت زیادہ ذمہ داریاں اٹھائے ہوئے ہیں۔
