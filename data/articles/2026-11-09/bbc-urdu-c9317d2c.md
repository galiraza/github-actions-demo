# محمد الفائد کے بھائی نے مجھے نشہ آور ادویات دیں اور جنسی حملوں کا نشانہ بنایا، خاتون کا دعویٰ

محمد الفائد کے بھائی نے مجھے نشہ آور ادویات دیں اور جنسی حملوں کا نشانہ بنایا، خاتون کا دعویٰ

ایک خاتون نے بی بی سی سکاٹ لینڈ کو بتایا ہے کہ لگژری سٹور ’ہیرڈز‘ کے سابق مالک محمد الفائد کے بھائی صلاح فائد انھیں سکاٹ لینڈ کے علاقے ہائی لینڈز لے کر آئے، نشہ آور ادویات دیں، جنسی حملوں کا نشانہ بنایا اور قید میں رکھا۔
جل، جن کا نام قانونی وجوہات کی بنا پر تبدیل کیا گیا ہے، کا کہنا ہے کہ 1990 کی دہائی میں وہ ’ہیرڈز‘ میں کام کرنے والی ایک نوجوان ورکر تھیں جب انھیں ایک ایسے پروگرام میں شامل ہونے کے لیے بھرتی کیا گیا، جسے وہ ایک انتظامی تربیتی سکیم سمجھتی تھیں۔
سکاٹ لینڈ میں ہونے والے واقعات کے بارے میں پہلی بار گفتگو کرتے ہوئے جل نے بتایا کہ انھیں فائد خاندان کے نجی طیارے کے ذریعے بالناگاؤن، جو ایسٹر راس میں اُن کی جاگیر ہے، لے جایا گیا اور وہاں ان پر بار بار جنسی حملے کیے گئے۔
جل اُن سینکڑوں خواتین میں سے ایک ہیں، جنھوں نے فائد برادران پر بدسلوکی اور جنسی استحصال کے الزامات عائد کیے ہیں۔
سنہ 2023 میں محمد الفائد کی وفات کے بعد سے سینکڑوں خواتین اور لڑکیوں نے ان پر جنسی استحصال کے الزامات عائد کیے ہیں۔
ان کے بھائی صلاح فائد، جو 2010 میں لبلبے کے کینسر کے باعث وفات پا گئے تھے، پر بھی متعدد خواتین کی جانب سے بدسلوکی اور جنسی استحصال کے الزامات لگائے گئے ہیں۔
سکاٹ لینڈ پولیس کا کہنا ہے کہ بالناگاؤن میں متعدد متاثرین کے خلاف جرائم کے سلسلے میں تین افراد کی شناخت کی جا چکی ہے اور تحقیقات تاحال جاری ہیں۔

بی بی سی سکاٹ لینڈ سے گفتگو کرتے ہوئے جل نے صلاح فائد کو ’انتہائی ظالم شخص‘ قرار دیا، جنھوں نے ان کی کمزوریوں کا فائدہ اٹھایا اور ان کی زندگی پر تقریباً مکمل کنٹرول قائم کر لیا تھا۔
انھوں نے بتایا کہ ہیرڈز میں اپنے کیریئر کا آغاز انھوں نے محمد الفائد کے لیے کام کرتے ہوئے کیا تھا۔
اُن کے مطابق وہ ایک ایسے دفتر میں کام کرتی تھیں جہاں کڑی نگرانی رکھی جاتی تھی، عملے میں زیادہ تر نوجوان خواتین شامل تھیں، رسائی سختی سے محدود تھی اور ملازمین کے دوسروں کے ساتھ میل جول رکھنے کی حوصلہ شکنی کی جاتی تھی۔
جل نے کہا کہ اس کے کچھ ہی عرصے بعد انھیں صلاح فائد کے لیے کام کرنے پر مامور کر دیا گیا۔
انھوں نے مزید کہا کہ ’مجھے لگا تھا کہ شاید یہ بہتر ہو گا۔ محمد الفائد مجھے مشکوک شخص لگتے تھے۔‘
تاہم ان کے بقول اس کے برعکس ان کی صورتحال تیزی سے بگڑتی چلی گئی۔
انھوں نے کہا ’مجھے دو مرتبہ سکاٹ لینڈ لے جایا گیا۔ دونوں بار فائد خاندان کے نجی طیارے کے ذریعے۔‘
’جب آپ وہاں جاتے تھے تو انورنیس ہوائی اڈے پر اترتے تھے اور پھر ان کی گاڑیاں آپ کو ہوائی اڈے سے لے کر فائد برادرز کی جاگیر تک پہنچاتی تھیں۔‘
بالناگاؤن سٹیٹ میں ہر بھائی کے لیے تین الگ الگ رہائش گاہیں تھیں۔
جل نے صلاح فائد کے گھر کے بارے میں بات کرتے ہوئے کہا کہ وہ محمد الفائد کی رہائش گاہ سے چھوٹا تھا، اس کا رُخ وادی کی جانب تھا، اس کے عقب میں جنگل تھا اور وہ کافی حد تک الگ تھلگ تھا۔
انھوں نے الزام عائد کیا کہ اس مرحلے تک صلاح فائد انھیں غیر قانونی منشیات دینا شروع کر چکے تھے۔
انھوں نے کہا ’پہلی بار جب انھوں نے مجھے منشیات لینے پر مجبور کیا تو اُنھوں نے دھوکے سے مجھے یہ چیزیں لینے پر آمادہ کرنے کی کوشش کی۔‘
جل کہتی ہیں کہ ’انھوں (صلاح فائد) نے مجھے ایک کمرے میں بند کر رکھا تھا۔ وہ کہتے تھے کہ ’تمہیں یہ (منشیات) لینی چاہییں، یہ تمہارے لیے اچھی چیز ہے اور جب میں نے انکار کیا تو وہ مجھ پر چیخے اور مجھے زبردستی یہ لینے پر مجبور کیا۔‘
’یہ منشیات مجھ پر حملہ کرنے کے ان کے طریقۂ کار کا حصہ تھیں۔‘
جل نے کہا کہ وہ اس بات پر بھی کنٹرول رکھتے تھے کہ وہ کیا کھائیں، کیا پہنیں اور کن لوگوں سے بات کریں۔
ان کا کہنا تھا کہ ان کے ساتھ جو کچھ ہوا، اس کا بیشتر حصہ ان کی مرضی کے خلاف تھا۔
انھوں نے کہا ’زیادہ تر وقت مجھے تنہا اور الگ تھلگ رکھا جاتا تھا۔ وہ ایک بہت ظالم آدمی تھے۔‘
’وہ جانتے تھا کہ میری کچھ کمزوریاں ہیں اور انھوں نے ان کا فائدہ اٹھایا۔ انھوں نے مجھے یہ احساس دلایا کہ میں ایک بہت بُری انسان ہوں۔‘
جل نے کہا کہ اس صورتحال سے نکلنا آسان نہیں تھا اور انھیں رفتہ رفتہ اس بات کا ادراک ہوا کہ ان کے ساتھ درحقیقت کیا کچھ ہوا تھا۔
انھوں نے مزید کہا ’جب آخرکار میں وہاں سے نکلنے میں کامیاب ہوئی تو مجھے پوری طرح اندازہ ہو چکا تھا کہ وہ میرے ساتھ کیا کر رہے تھے اور ان سے جان چھڑانے کے لیے مجھے کافی جدوجہد کرنا پڑی۔‘
’اس وقت تک مجھے یہ احساس بھی ہو چکا تھا کہ میں ان کی واحد شکار نہیں تھی۔‘
’میں نے خطرے کی گھنٹی بجانے کی کوشش کی، میں نے لوگوں کو خبردار کرنے کی کوشش کی۔‘
صلاح فائد ان تین فائد برادران میں سے ایک تھے جنھوں نے سنہ 1985 میں نائٹس برج کے معروف پرتعیش ڈپارٹمنٹ سٹور ہیرڈز کو خریدا تھا۔
ان کی وفات کے بعد 2024 میں نشر ہونے والی بی بی سی کی دستاویزی فلم ’پریڈیٹر ایٹ ہیرڈز‘ میں 20 سے زائد خواتین نے محمد الفائد پر جنسی حملوں کے الزامات عائد کیے تھے۔
ان الزامات کے منظرِ عام پر آنے کے بعد 400 سے زائد مبینہ متاثرین یا گواہ قانونی نمائندوں سے رابطہ کر چکے ہیں۔
اس سال کے آغاز میں سکاٹ لینڈ پولیس نے کہا تھا کہ وہ بالناگاؤن سٹیٹ سے متعلق الزامات کے حوالے سے مزید تحقیقات کر رہی ہے۔
ایک بیان میں پولیس کے ترجمان نے کہا کہ ’متعدد متاثرین کے خلاف جرائم کے سلسلے میں تین افراد کی شناخت کی جا چکی ہے۔‘
’ہماری تحقیقات جاری ہیں اور اس مرحلے پر مزید تبصرہ کرنا مناسب نہیں ہو گا۔‘
دوسری جانب میٹروپولیٹن پولیس الگ سے محمد الفائد اور ان افراد کے خلاف الزامات کی تحقیقات کر رہی ہے، جن پر ان کے مبینہ جرائم میں سہولت کاری کرنے کا شبہ ہے۔
پولیس نے کہا کہ یہ ان سب سے بڑی اور پیچیدہ تحقیقات میں سے ایک ہے جو وہ اس وقت انجام دے رہی ہے۔
میٹروپولیٹن پولیس کے ایک بیان میں کہا گیا کہ ’اب تک سات مشتبہ افراد سے احتیاطی بنیاد پر پوچھ گچھ کی جا چکی ہے۔ ان پر جن جرائم کا شبہ ہے ان میں جنسی حملے میں معاونت اور سہولت کاری، جنسی جرائم کے ارتکاب میں مدد اور جنسی استحصال کے مقصد سے انسانی سمگلنگ شامل ہیں۔‘
بیان میں مزید کہا گیا ہے کہ ’اب تک 168 متاثرین سامنے آ چکے ہیں۔ ہم انسانی سمگلنگ سے متعلق شواہد کی تلاش کے لیے متاثرین کی گواہیوں کا جائزہ لیتے رہیں گے۔‘
فائد اور ہیرڈز کے متاثرین کے لیے قائم آل پارٹی پارلیمانی گروپ اور میٹروپولیٹن پولیس کے درمیان ہونے والی ایک ملاقات کے خفیہ نوٹ، جو متاثرین کے ساتھ شیئر کیا گیا، کے مطابق تقریباً 70 افراد کو انسانی سمگلنگ کے ممکنہ متاثرین کے طور پر ہوم آفس کے حوالے کیا گیا تھا۔
جل نے بی بی سی سکاٹ لینڈ کو بتایا کہ وہ دیگر ایسی خواتین کے بارے میں بھی جانتی ہیں جنھوں نے الزام عائد کیا ہے کہ سکاٹ لینڈ میں صلاح فائد نے ان کے ساتھ بدسلوکی کی، اگرچہ ان میں سے بعض نے کبھی عوامی طور پر اپنے تجربات بیان نہیں کیے۔
ان کا کہنا تھا کہ وہ بعض متاثرین کی جانب سے قانونی اختیارات کے تحت عوامی تحقیقاتی کمیشن کے قیام کے مطالبے سے متفق ہیں۔
انھوں نے کہا کہ وہ اس لیے سامنے آ رہی ہیں کیونکہ انھیں تشویش ہے کہ پولیس کی تحقیقات معاملے کے تمام پہلوؤں تک نہیں پہنچ رہیں اور کافی دور تک نہیں جا رہی۔
جل نے بتایا کہ انھوں نے میٹروپولیٹن پولیس سے رابطہ کیا ہے اور ساتھ ہی قانونی کارروائی بھی جاری ہے۔
ہیرڈز، جو اب فائد خاندان کی ملکیت نہیں ہے، نے کہا ہے کہ وہ سامنے آنے والے متاثرین کی ’غیر معمولی جرات‘ کو تسلیم کرتا ہے اور انصاف کے حصول کی ان کی کوششوں میں ان کی حمایت کرتا ہے۔
اس ریٹیل ادارے نے اس سے قبل محمد الفائد سے منسوب بدسلوکی کے متاثرین کے ازالے کے لیے ایک پروگرام بھی قائم کیا تھا۔
جل کا کہنا ہے کہ کسی بھی قسم کا معاوضہ ان کی زندگی پر پڑنے والے اثرات کا صرف جزوی ازالہ ہی ہو سکتا ہے۔
انھوں نے کہا ’آپ اپنی زندگی کے کئی سال کھو چکے ہوتے ہیں۔‘
’علاج معالجے پر اخراجات کرنے پڑتے ہیں، اس کا آپ کے کیریئر پر اثر پڑتا ہے، آپ کے خاندان پر اثر پڑتا ہے، اس لیے ہاں، انھیں اس نقصان کا ازالہ کرنا چاہیے۔‘
جل کہتی ہیں کہ ’وہ لڑکی، جو میں اس وقت تھی جب مجھے اس شخص کے پاس بھیجا گیا تھا، اب باقی نہیں رہی۔‘
’میں نہیں جانتی کہ وہ آج کون ہوتی یا اس کی زندگی کیسی ہوتی۔‘
’میری پوری زندگی کا کام اس تجربے سے خود کو سنبھالنے اور بحال کرنے میں گزرا ہے۔ یہ ایک تلخ، انتہائی تلخ، اور دہائیوں پر محیط جدوجہد رہی ہے۔‘
بی بی سی نے مؤقف جاننے کے لیے دو مرتبہ بالناگاؤن سٹیٹ سے رابطہ کیا، تاہم تاحال وہاں سے کوئی جواب موصول نہیں ہوا۔
