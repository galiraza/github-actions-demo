’سیکس کنٹریکٹس‘: کولمبیا کے سابق صدر ایک خاتون کے انٹرویو کے بعد تحقیقات کی زد میں کیسے آئے؟
کولمبیا کے سابق صدر گستاوو پیٹرو ایک ایسی سیاسی ہلچل کا مرکز بن گئے ہیں، جو ایک خاتون کے الزامات کے باعث پیدا ہوئی ہے۔ خاتون کا دعویٰ ہے کہ انھوں نے سابق صدر کے ساتھ جنسی تعلقات کی بدولت سرکاری ٹھیکے حاصل کیے۔
یہ سکینڈل 25 اکتوبر کو منظرِ عام پر آیا۔ ڈیسی خیمنیز، جو اس سے قبل عوام کے لیے غیر معروف تھیں، نے صحافی وکی ڈیویلا کو 22 منٹ کا ایک انٹرویو دیا۔ وکی ڈیویلا ریویسٹا سیمانا نامی جریدے کی ڈائریکٹر اور سابق صدارتی امیدوار ہیں اور پیٹرو کی مخالف سمجھی جاتی ہیں۔
خاتون نے انٹرویو میں دعویٰ کیا کہ ان کا پیٹرو کے ساتھ رومانوی تعلق تھا اور یہ کہ انھیں ان کی مدتِ صدارت سے پہلے اور اس کے دوران کئی سرکاری اداروں میں ملازمت دی گئی۔
انھوں نے یہ بھی دعویٰ کیا کہ ایسے فوائد حاصل کرنے والوں وہ اکیلی نہیں تھیں۔ ان کے مطابق ’30 سے 40 خواتین‘ پر مشتمل ایک نیٹ ورک تھا، جس نے سرکاری ٹھیکوں اور عہدوں سے فائدہ اٹھایا اور اس کے بدلے میں وہ سابق صدر کے ساتھ ’سیکس پارٹیوں‘ میں ’جاتی تھیں۔‘
فی الحال دیگر خواتین کی جانب سے ایسی کوئی شہادت سامنے نہیں آئی ہے اور نہ ہی اس بات کا کوئی ثبوت موجود ہے کہ یہ ایک منظم نیٹ ورک تھا۔
اس انٹرویو کے بعد خیمنیز نے اپنے سوشل میڈیا اکاؤنٹ پر ایک ویڈیو پوسٹ کی، جس کے بارے میں ان کا کہنا تھا کہ اس سے ثابت ہوتا ہے کہ ان کا پیٹرو کے ساتھ واقعی تعلق تھا۔
کولمبیا کے سابق صدر خاتون کے ساتھ تعلق سے انکار نہیں کرتے، تاہم وہ اس الزام کی تردید کرتے ہیں کہ انھوں نے جنسی تعلقات کے بدلے سرکاری ٹھیکے دیے تھے۔ انھوں نے اپنے ایکس اکاؤنٹ پر ایک پوسٹ میں کہا ’اگر میں نے کبھی ایسا کوئی کام کیا ہوتا تو مجھے خود پر ہی قے آ جاتی۔‘
پیٹرو کا دعویٰ ہے کہ یہ ڈی لا ایسپریئیلا حکومت کی جانب سے شروع کیا گیا ایک آپریشن ہے اور وہ تاثر دیتے ہیں کہ خیمنیز کو یہ بیانات دینے کے لیے رقم ادا کی گئی تھی۔
End of سب سے زیادہ پڑھی جانے والی
خاتون کے دعوؤں اور ویڈیو کی بنیاد پر سابق صدر کے خلاف مختلف عدالتی فورمز پر پہلے ہی تحقیقات شروع کی جا چکی ہیں۔
یہ اس سکینڈل کے چار اہم نکات ہیں جسے کولمبیا کے میڈیا نے ’سیکس کنٹریکٹس‘ کا نام دیا ہے۔
مبینہ ’سیکس کنٹریکٹس‘ کے الزامات
وہ تمام خِبریں جو آپ جاننا چاہتے ہیں بی بی سی ورلڈ سروس ایپ پر موجود ہیں
ایپ ڈاؤن لوڈ کریں
بی بی سی اُردو موبائل ایپ نهاية
سیمانا میگزین کو دیے گئے بیان کے مطابق ڈیسی خیمنیز کی ملاقات پیٹرو سے ایک ویب کیم ماڈلنگ پلیٹ فارم کے ذریعے ہوئی۔ ان کے بقول ’وہ بطور ایک عام گاہک سامنے آئے تھے۔‘
خیمنیز کے مطابق پیٹرو نے ابتدا میں اپنا تعارف ایک فرضی نام سے کرایا، لیکن ایک موقع پر انھوں نے اپنا کیمرا آن کیا اور وہ انھیں پہچان گئیں۔
خاتون کے مطابق ایک ماہ بعد دونوں نے کالی کے ایک ہوٹل میں بالمشافہ ملاقات پر اتفاق کیا اور ان کے درمیان رومانوی تعلق کا آغاز ہوا۔
ان کے بقول اس کے کچھ ہی عرصے بعد انھوں نے بوگوٹا منتقل ہونے کا فیصلہ کیا۔ ان کا کہنا ہے کہ اسی وقت سے انھیں سرکاری اداروں کے ساتھ معاہدوں کا فائدہ ملنا شروع ہوا، تاہم انھوں نے یہ واضح نہیں کیا کہ آیا پیٹرو نے انھیں یہ ٹھیکے ان کے جنسی تعلقات سے پہلے ملے تھے یا بعد میں۔
خیمنیز نے دعویٰ کیا کہ ان کی طرح پیٹرو کے ساتھ اسی طرز کے معاہدے رکھنے والی ’کافی زیادہ خواتین‘ تھیں۔ ان کے اپنے الفاظ میں ان کی تعداد ’30 سے 40 کے درمیان‘ تھی۔
انھوں نے مزید کہا ’ہم سب بہت کچھ جانتے ہیں۔‘
ان کے یہ بیانات اس شکایت کے چند روز بعد سامنے آئے جو پیٹرو کے خلاف موجودہ ڈائریکٹر، ایڈمنسٹریٹو ڈیپارٹمنٹ آف دی پریزیڈنسی کارلوس ریوس نے کانگریس میں جمع کرائی تھی۔
صدر ایبیلارڈو ڈی لا ایسپریئیلا کے قریبی حلقے کے ارکان میں شامل ریوس نے ’48 خواتین‘ کا ذکر کیا تھا جنھیں، ان کے بقول، سابق صدر گستاوو پیٹرو نے ایوانِ صدر میں رکھا ہوا تھا اور جنھیں انھوں نے بدقسمتی سے او پی ایس (سروس پروویژن کنٹریکٹس) کے ذریعے اپنے مقاصد کے لیے استعمال کیا۔
ان بیانات اور ڈیسی خیمنیز کے انٹرویو کے ردعمل میں، پیٹرو کے وکیل نے استغاثہ کے دفتر میں کارلوس ریوس کے خلاف شکایت دائر کی اور ان پر الزام عائد کیا کہ انھوں نے ان کے خلاف خواتین سے من گھڑت گواہیاں تیار کرانے کا ایک منصوبہ بنایا۔
اس سے قبل اینجی روڈریگیز، جو پیٹرو کی حکومت کے دوران ڈی اے پی آر ای کی ڈائریکٹر تھیں، ایک انٹرویو میں کاسا دی نارینو میں مبینہ جسم فروشی کے ایک نیٹ ورک کے بارے میں بھی بات کر چکی تھیں۔
روڈریگیز کے بارے میں پیٹرو نے کہا ہے کہ وہ ’نفسیاتی عارضے میں مبتلا شخص’ ہیں۔
منظر عام پر آنے والے مبینہ شواہد
سیکوپ II (الیکٹرانک پبلک پروکیورمنٹ سسٹم) میں ڈیسی خیمنیز کے تین سرکاری اداروں کے ساتھ کیے گئے خدمات کی فراہمی کے 10 معاہدے درج ہیں۔
یہ معاہدے 2021 سے 2026 کے درمیان کیے گئے اور ان کی مجموعی مالیت 12 کروڑ 65 لاکھ 44 ہزار سے زائد کولمبین پیسو ہے (موجودہ شرح تبادلہ کے مطابق تقریباً 38,000 امریکی ڈالر)۔
خیمنیز نے اپنی گواہی میں جو بعض تفصیلات بیان کیں وہ معاہدوں میں درج معلومات سے مطابقت رکھتی ہیں، تاہم بعض دیگر تفصیلات مطابقت نہیں رکھتیں۔
مثال کے طور پر ان کا کہنا ہے کہ جب وہ بوگوٹا پہنچیں تو انھوں نے سینیٹ میں کام شروع کیا۔ حقیقتاً ریکارڈ میں موجود ان کا پہلا کنٹریکٹ اسی ادارے کے ساتھ ہے۔
تاہم ان کے مطابق یہ 2019 میں ہوا تھا، جبکہ کنٹریکٹ میں تاریخ 29 جولائی 2021 درج ہے۔
2018 سے 2022 کے درمیان گستاوو پیٹرو سینیٹر تھے۔
خیمنیز کے بیان کے مطابق انھوں نے سینیٹ میں براہِ راست پیٹرو کے ساتھ کام نہیں کیا، بلکہ سینیٹر ماریا خوسے پیزارو کے ساتھ کام کیا، جو اسی جماعت کی رکن اور سابق صدر کی قریبی اتحادی ہیں۔
خاتون نے بتایا کہ اس کے بعد انھیں سپرنٹنڈنسی آف سرویلنس اینڈ پرائیویٹ سکیورٹی اور وزارتِ داخلہ کے ساتھ بھی کانٹریکٹ ملے، جیسا کہ سیکوپ II میں درج ہے۔
یہ دونوں ادارے ایگزیکٹو برانچ کا حصہ ہیں اور خیمنیز نے وہاں اسی دوران کام کیا جب پیٹرو صدر تھے۔
ریکارڈ میں موجود سب سے حالیہ معاہدے پر رواں سال 21 جنوری کو دستخط کیے گئے تھے اور یہ اب بھی اپنے نفاذ کی مدت میں ہے۔ اس معاہدے کے تحت 2026 کے دوران ڈیسی خیمنیز کو ماہانہ 26 لاکھ 48 ہزار سے زائد کولمبین پیسو (800 امریکی ڈالر) ادا کیے جانے ہیں۔
یہ واضح نہیں کیا گیا کہ آیا خیمنیز اب بھی ریاست سے رقم وصول کر رہی ہیں یا نہیں۔
اب تک خیمنیز اور صدر کے درمیان مبینہ تعلق کے بارے میں جو تفصیلات سامنے آئی ہیں، وہ ایک ویڈیو سے حاصل ہوئی ہیں جو خود اُنھوں نے شائع کی تھی اور ایک دوسری ویڈیو سے جو صحافی ڈی آرسی کوئن نے جاری کی تھی۔
پہلی ویڈیو میں مبینہ طور پر پیٹرو اور خیمنیز ایک کمرے میں دکھائی دیتے ہیں۔ وہ ایک جیکٹ پہنتے ہیں اور وہ ان سے کہتی ہیں ’آپ بھی بہت خوبصورت لگ رہے ہیں۔‘ اس پر پیٹرو جواب دیتے ہیں ’کیا واقعی آپ ایسا سمجھتی ہیں؟‘
اس کے بعد گفتگو ایک ایسے سیکس ٹوئے کی طرف مڑ جاتی ہے جو ویڈیو میں نظر نہیں آتا اور پھر سابق صدر خیمنیز کا سر پکڑ کر انھیں بوسہ دیتے ہیں۔
ڈیسی خیمنیز نے یہ ویڈیو اپنے انسٹاگرام اکاؤنٹ پر یہ کہتے ہوئے پوسٹ کی: ’میں یہ ویڈیو اس بات کے ثبوت کے طور پر شائع کر رہی ہوں کہ میں سچ بول رہی ہوں۔‘
دوسری ویڈیو ایک سکرین ریکارڈنگ ہے، جو بظاہر خیمنیز کے فون سے بنائی گئی ہے اور اس میں مبینہ طور پر پیٹرو کے ساتھ ایک ویڈیو کال دکھائی گئی ہے۔
ریکارڈنگ کے مطابق پیٹرو ایک فرضی نام استعمال کرتے ہوئے خیمنیز سے بات کر رہے ہیں: ’آپ بہت خوبصورت لگ رہی ہیں۔ کھڑی ہو جائیں۔‘ وہ انھیں بوسے بھی بھیجتے ہیں اور خاموش رہنے کا اشارہ بھی کرتے ہیں۔
بی بی سی نیوز منڈو ان ویڈیوز یا سوشل میڈیا پر گردش کرنے والی ان چیٹس کی آزادانہ طور پر تصدیق نہیں کر سکا، جو مبینہ طور پر خیمنیز اور پیٹرو کی جانب سے چلائے جانے والے جعلی پروفائلز کے درمیان تھیں۔
پیٹرو اور ان کے قریبی حلقے کا کیا کہنا ہے؟
پیٹرو نے خیمنیز کے الزامات کی تردید کی ہے اور اپنی نجی زندگی کے احترام کا مطالبہ کیا ہے۔
انھوں نے اسی روز جاری کیے گئے ایک بیان میں، جس دن خاتون کی گواہی منظرِ عام پر آئی، کہا ’میں نے کبھی جنسی تعلقات حاصل کرنے کے بدلے کوئی کانٹریکٹ نہیں دیا۔‘
انھوں نے مزید کہا ’غریب خواتین، اکیلی ماؤں اور سیکس ورکرز کی مدد کرنا تاکہ انھیں زندگی میں مختلف مواقع میسر آ سکیں (...) جنسی تعلقات کے بدلے کانٹریکٹس بانٹنے سے بالکل مختلف بات ہے۔‘
حال ہی میں پیٹرو نے دعویٰ کیا کہ بطور صدر وہ ڈیسی خیمنیز کبھی نہیں ملے۔ انھوں نے سوال اٹھایا ’کیا تقریباً آٹھ سال پرانی ایک ویڈیو ثبوت سمجھی جا سکتی ہے؟‘
کارلوس ریوس کے خلاف اپنی شکایت میں پیٹرو کے وکیل نے اس مؤقف کا دفاع کیا ہے کہ ڈی لا ایسپریئیلا حکومت کی جانب سے سابق صدر کے خلاف خواتین کی گواہیاں گھڑنے کے لیے ایک کارروائی چلائی جا رہی ہے۔
سابق صدر کی جماعت ہسٹوریکل پیکٹ نے بھی اسی مؤقف کی تائید کی ہے۔
پیٹرو نے اسی موضوع پر اپنے پہلے بیان میں کہا ’میری نجی زندگی کو بے نقاب کرنے کی یہ کارروائی ایبیلارڈو ڈی لا ایسپریئیلا کی حکومت کی جانب سے کی جا رہی ہے اور اس میں پیسہ استعمال ہو رہا ہے۔‘
ریکارڈ شدہ ویڈیو کال کے ایک حصے کا حوالہ دیتے ہوئے، جس میں وہ بوسے بھیجتے ہوئے دکھائی دیتے ہیں، سابق صدر نے اپنے ایکس اکاؤنٹ پر لکھا ’اب بوسوں کو بھی جرم قرار دیا جا رہا ہے، حالانکہ وہ بنیادی اہمیت رکھتے ہیں۔‘
سابق سینیٹر ماریا خوسے پیزارو، جو پیٹرو کی قریبی اتحادی ہیں، نے تسلیم کیا کہ ڈیسی خیمنیز اس وقت سینیٹ میں کام کرتی تھیں جب وہ نائب صدر تھیں۔
تاہم انھوں نے کہا کہ انھیں کبھی یہ معلوم نہیں تھا کہ خیمنیز کا پیٹرو کے ساتھ کوئی قریبی یا ذاتی تعلق تھا۔ انھوں نے ایک بار پھر اس بات پر زور دیا کہ کسی بھی قسم کے ’احسانات‘ کے بدلے کانٹریکٹس نہیں دیے گئے تھے۔
ہسٹوریکل پیکٹ سے تعلق رکھنے والے سابق صدارتی امیدوار ایوان سیپیڈا نے پوڈکاسٹ ایل ریپورٹے کورونیل کو دیے گئے ایک انٹرویو میں کہا کہ پیٹرو کے اس طرزِ عمل کا، جو اب منظرِ عام پر آیا ہے، جماعت کی جانب سے جائزہ لیا جائے گا۔
تحقیقات کہاں تک پہنچیں؟
اس سکینڈل کے باعث استغاثہ کے دفتر، سپریم کورٹ آف جسٹس اور اٹارنی جنرل کے دفتر میں مختلف نوعیت کی تحقیقات پہلے ہی جاری ہیں۔
جیسا کہ خود ڈیسی خیمنیز نے اعلان کیا اور مقامی ذرائع ابلاغ نے بھی اس کی تصدیق کی، انھوں نے اپنا موبائل فون استغاثہ کے دفتر کے حوالے کر دیا، جہاں کئی گھنٹوں تک اس کا فرانزک معائنہ کیا گیا۔
توقع ہے کہ وہ اس ادارے کے سامنے اپنا بیان بھی ریکارڈ کرائیں گی۔
دوسری جانب سپریم کورٹ کے ایک جج نے سینیٹ میں خیمنیز کے کانٹریکٹس کا جائزہ لینے کے لیے ان کی تحقیقات کا حکم دیا ہے۔
جج کارلوس روبرتو سولورسانو نے صحافیوں سے گفتگو کرتے ہوئے کہا ’ذاتی تفصیلات سے قطع نظر یہاں اہم بات یہ ہے کہ ان ممکنہ مجرمانہ پہلوؤں کا تعین کیا جائے کہ آیا ریاستی فنڈز سے کوئی کانٹریکٹس کیے گئے تھے یا نہیں۔‘
پیٹرو کے صدر بننے سے پہلے پیش آنے والے واقعات کی فوجداری تحقیقات استغاثہ کے دفتر اور سپریم کورٹ آف جسٹس کر سکتے ہیں۔
ان کی صدارتی مدت کے دوران پیش آنے والے معاملات کمیٹی آف ایکیوزیشنز کے دائرہ اختیار میں آتے ہیں، جو ارکانِ کانگریس پر مشتمل ہے اور عام طور پر تحقیقات میں پیش رفت نہ ہونے کے حوالے سے مشہور ہے۔
اس کے باوجود حکمران جماعت نیشنل سالویشن کے دو ارکانِ کانگریس نے پیٹرو کے خلاف اسی کمیٹی میں شکایت دائر کی ہے، جس میں ان پر ممکنہ ’غلط طرزِ عمل کے باعث عہدے کے لیے نااہلی‘ کا الزام عائد کیا گیا ہے۔
