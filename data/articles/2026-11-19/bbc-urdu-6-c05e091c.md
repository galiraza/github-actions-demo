’جی ٹی اے 6‘ کی ریلیز اور پاکستان سمیت دنیا بھر میں گیمرز کا جنون: کسی نے چھٹی لی تو کسی نے نیا پلے سٹیشن خرید لیا
دنیا بھر کے گیمرز گذشتہ 13 سال سے جس گیم کا انتظار کر رہے تھے، وہ ویڈیو گیمز کی دنیا کی مشہور ترین گیم ’گرینڈ تھیفٹ آٹو 6‘ یعنی ’جی ٹی اے 6‘ ریلیز ہو چکی ہے اور اس کے آتے ہی دنیا بھر میں گیمرز کا جوش دیکھنے والا ہے۔
گیم سب سے پہلے نیوزی لینڈ میں دستیاب ہوئی، جہاں مداحوں کو دنیا کے دوسرے حصوں سے پہلے اسے کھیلنے کا موقع ملا۔
گیم کی ریلیز سے پہلے ہی شائقین نے اسے کھیلنے کے لیے خوب تیاریاں کر رکھی تھیں۔ کسی نے دفتر سے چھٹی لے لی، کسی نے نیا پلے سٹیشن خرید لیا، تو کچھ ایسے بھی تھے جنھوں نے جلد گیم کھیلنے کے لیے اپنے کنسول کی لوکیشن ہی نیوزی لینڈ کی کر دی تاکہ وہ دوسرے ممالک کے مقابلے میں پہلے کھیل سکیں۔
لیکن بات صرف مداحوں کے جوش و خروش تک محدود نہیں۔ گیم کے ابتدائی ریویوز نے بھی اسے خاصا مقبول بنا دیا ہے۔
گیم کے گرافکس، کہانی، کھیلنے کے انداز اور اس کی وسیع ورچوئل دنیا کو ناقدین کی جانب سے خوب سراہا گیا ہے۔
’جی ٹی اے 6‘ کا جنون صرف بیرونِ ملک ہی نہیں بلکہ پاکستان میں بھی دیکھا جا رہا ہے اور پاکستانی گیمرز بھی اس گیم کو کھیلنے کے لیے خاصے پُرجوش ہیں۔
پاکستان میں بھی جی ٹی اے 6 کا جنون، پلے سٹیشن کی قیمتوں میں اضافہ، گیمنگ زونز میں پیشگی بکنگ
پاکستانی گیمنگ کمیونٹی سے وابستہ عامر شہزاد نے بی بی سی سے گفتگو کرتے ہوئے بتایا کہ پاکستان میں ’جی ٹی اے 6‘ کی دستیابی کے بعد پلے سٹیشن 5 کی قیمتوں میں بھی نمایاں اضافہ ہوا ہے۔
ان کے مطابق جو ’پی ایس 5‘ پہلے تقریباً ایک لاکھ 40 ہزار روپے میں فروخت ہو رہا تھا، اس کی قیمت اب بڑھ کر ایک لاکھ 80 ہزار روپے تک پہنچ گئی ہے، جبکہ مارکیٹ میں یہ کنسول بلیک میں بھی فروخت کیا جا رہا ہے۔
End of سب سے زیادہ پڑھی جانے والی
وہ تمام خِبریں جو آپ جاننا چاہتے ہیں بی بی سی ورلڈ سروس ایپ پر موجود ہیں
ایپ ڈاؤن لوڈ کریں
بی بی سی اُردو موبائل ایپ نهاية
عامر شہزاد کا کہنا تھا کہ ’جی ٹی اے 6‘ تقریباً 100 ڈالر کے واؤچر کے ذریعے خریدی جا سکتی ہے، تاہم صارفین کو اپنے پلے اسٹیشن اکاؤنٹ کے ریجن کے مطابق ہی واؤچر خریدنا ہوتا ہے۔ یعنی جس ملک کا پلے اسٹیشن اکاؤنٹ ہوگا، اسی ملک کا واؤچر استعمال کرنا پڑے گا اور ادائیگی بھی متعلقہ ملک کی قیمت کے مطابق کرنا ہوگی۔
انھوں نے مزید بتایا کہ پاکستانی گیمرز اور یوٹیوبرز بھی ’جی ٹی اے 6‘ کی لائیو اسٹریمنگ کر رہے ہیں، جنہیں ہزاروں افراد براہِ راست دیکھ رہے ہیں۔
اس کے علاوہ پاکستان کے کئی گیمنگ زونز نے بھی صارفین کو اپنے مراکز پر آ کر ’جی ٹی اے 6‘ کھیلنے کی سہولت فراہم کرنے کا اعلان کیا ہے۔
پاکستان میں ایک گیمنگ سٹور کے مالک نے بی بی سی کو بتایا کہ ’جی ٹی اے 6‘ کی آمد کے بعد پلے سٹیشن 5 کی قیمتوں میں بھی کافی اضافہ ہوا ہے۔
ان کے مطابق پہلے پی ایس 5 تقریباً دو لاکھ 20 ہزار روپے میں فروخت ہو رہا تھا، لیکن اب اس کی قیمت بڑھ کر دو لاکھ 80 ہزار روپے تک پہنچ گئی ہے۔
انھوں نے بتایا کہ ’جی ٹی اے‘ کے شوقین افراد گیم کی ریلیز کے بعد بڑی تعداد میں پلے سٹیشن خرید رہے ہیں۔ پہلے ایک دن میں ایک کنسول فروخت ہوتا تھا، لیکن اب روزانہ پانچ سے چھ کنسولز بھی فروخت ہو رہے ہیں۔
پاکستان میں گیمنگ زونز چلانے والے نوجوانوں نے بھی بی بی سی سے گفتگو کرتے ہوئے بتایا کہ گیم کی ریلیز کے بعد اسے کھیلنے کے خواہش مند افراد کی تعداد میں اضافہ ہو رہا ہے۔
پاکستان میں گیمنگ زونز کے مالکان سے گفتگو کے بعد یہ بات سامنے آئی کہ جو افراد مہنگے پلے سٹیشن خریدنے کی استطاعت نہیں رکھتے، وہ بھی اس گیم سے لطف اندوز ہو سکتے ہیں۔ ایسے افراد تقریباً 800 سے 1500 روپے فی گھنٹہ ادا کرکے گیمنگ زونز میں ’جی ٹی اے 6‘ کھیل سکتے ہیں۔
لاہور کے ’پلے اِن گیمنگ زون‘ کے مالک میاں شہزاد نے بی بی سی کو بتایا کہ ’جمعرات کی صبح سے جی ٹی اے 6 کھیلنے کے خواہش مند افراد کے لیے دستیاب ہوگی۔ بہت سے نوجوان ابھی سے گیم کھیلنے کے لیے بکنگ کروا رہے ہیں اور چاہتے ہیں کہ وہ اسے سب سے پہلے کھیلنے والوں میں شامل ہوں۔‘
میاں شہزاد کے مطابق ’جو لوگ پاکستان میں مہنگا پلے سٹیشن نہیں خرید سکتے، وہ جی ٹی اے 6 کھیلنے کے لیے گیمنگ زونز کا رخ کرتے ہیں۔‘
اسی طرح حمزہ فاضل نے بی بی سی سے گفتگو کرتے ہوئے کہا کہ ’پاکستان میں بھی ’جی ٹی اے 6‘ کے شوقین افراد کی تعداد کافی زیادہ ہے، جس کا اندازہ اس بات سے لگایا جا سکتا ہے کہ نوجوان ابھی سے گیمنگ زونز کا رخ کر رہے ہیں اور یہ جاننے کی کوشش کر رہے ہیں کہ وہ کب یہ گیم کھیل سکیں گے۔‘
گیم کے ابتدائی جائزوں کی بات کریں تو ریویوز جمع کرنے والی ویب سائٹ ’میٹا کریٹک‘ پر اسے 100 میں سے 99 نمبر دیے گئے ہیں، جس کے بعد یہ بہترین ریٹنگ حاصل کرنے والی ویڈیو گیمز میں شامل ہو گئی ہے۔
معروف گیمنگ ویب سائٹ ’آئی جی این‘ نے اسے ’غیر معمولی‘ اور ’ناقابل یقین‘ قرار دیتے ہوئے ایسی گیم کہا ہے جسے ہر گیمر کو ضرور کھیلنا چاہیے۔
اسی طرح 'یوروگیمر' کے مطابق گیم بنانے والی کمپنی ’راک سٹار‘ نے گرافکس، کہانی اور گیم کی دنیا کو جس انداز میں پیش کیا ہے، اس سے ویڈیو گیمز کا معیار ایک نئی سطح پر پہنچ گیا ہے۔ تاہم بعض ناقدین نے گیم میں دکھائی گئی حقیقت پسندی اور اس کے ممکنہ اثرات پر سوالات بھی اٹھائے ہیں۔
جی ٹی اے 6 کا جنون برطانیہ میں بھی سر چڑھ کر بول رہا ہے۔ ویسٹ سسیکس سے تعلق رکھنے والی وکٹوریہ ایونز نے تو گیم کھیلنے کے لیے ریلیز والے دن صبح پانچ بجے کا الارم لگا رکھا تھا اور باقاعدہ دفتر سے چھٹی بھی لے لی۔
دلچسپ بات یہ ہے کہ وکٹوریہ نے چھٹی کی درخواست میں صاف لکھ دیا تھا کہ وہ جی ٹی اے 6 کھیلنے کے لیے چھٹی چاہتی ہیں۔ ان کے باس نے بھی درخواست منظور کر لی، حالانکہ انھیں یہ تک معلوم نہیں تھا کہ جی ٹی اے 6 آخر ہے کیا!
اور وکٹوریہ اکیلی نہیں ہیں۔ برطانیہ میں بڑی تعداد میں لوگوں نے اس گیم کے لیے اپنی چھٹیاں پہلے ہی طے کر لی تھیں۔
برطانوی ہیومن ریسورس سافٹ ویئر کمپنی 'برائٹ ایچ آر' نے 70 ہزار کاروباری اداروں کے 12 لاکھ سے زیادہ ملازمین کی چھٹیوں اور غیر حاضری کے اعداد و شمار کا جائزہ لیا۔
رپورٹ کے مطابق ملازمین کی 4.6 فیصد چھٹیوں کی درخواستوں میں جی ٹی اے کا ذکر موجود تھا، جبکہ نومبر اور دسمبر کے دوران اس گیم کے لیے مجموعی طور پر تقریباً 83 ہزار دن کی چھٹیاں لی گئیں۔
یعنی کسی نے باس سے چھٹی منظور کروائی، کسی نے نئے کنسول پر پیسے لگائے اور کسی نے گیم جلد کھیلنے کے لیے نیوزی لینڈ پہنچنے کا ڈیجیٹل راستہ نکال لیا۔
گیم کی غیر معمولی مانگ
مارکیٹ انٹیلی جنس فرم ’سینسر ٹاور‘ کے اندازے کے مطابق ’جی ٹی اے 6‘ کی اب تک ایک کروڑ 20 لاکھ سے زائد پری آرڈرز (پیشگی بکنگ) ہوچکی ہیں، جن سے کمپنی کو تقریباً ایک ارب 18 کروڑ ڈالر (894 ملین پاؤنڈ) کی آمدنی ہوئی ہے۔ تاہم گیم بنانے والی کمپنی راک سٹار نے ان اعداد و شمار کی تصدیق نہیں کی۔
گیم کا سائز ’212 جی بی‘ ہے اور اسے ریلیز سے پہلے ڈاؤن لوڈ کرنے کی سہولت بھی موجود ہے، اس کے باوجود برطانیہ میں انٹرنیٹ کی بڑھتی ہوئی طلب کے پیش نظر کئی انٹرنیٹ فراہم کرنے والی کمپنیوں نے اپنی سروس کی گنجائش بڑھا دی ہے۔
لندن انٹرنیٹ ایکسچینج کی چیف ایگزیکٹیو جینیفر ہومز کے مطابق برطانیہ میں ’بی ٹی‘ اور ’ورجن میڈیا‘ سمیت کئی انٹرنیٹ کمپنیوں نے گیم کی ریلیز کے موقع پر صارفین کی زیادہ تعداد اور انٹرنیٹ کے بڑھتے استعمال کو مدنظر رکھتے ہوئے اضافی انتظامات کیے ہیں۔
گیم کی بڑی ریلیز کے موقع پر ملازمت سے چھٹی لینا کوئی نئی بات نہیں۔ اس سے پہلے بھی کئی مشہور ویڈیو گیمز کے مداح ایسا کرتے رہے ہیں۔
اکتوبر سنہ 2018 میں جب ’ریڈ ڈیڈ ریڈمپشن 2‘ ریلیز ہوئی تو سوشل میڈیا پر کئی صارفین نے گیم کھیلنے کے لیے دفتر سے چھٹی لینے کے منصوبے شیئر کیے۔ ایک تجزیاتی ادارے کے مطابق اس سال کسی بھی دوسری گیم کے مقابلے میں زیادہ لوگوں نے اس گیم کے لیے بیماری کا بہانہ کرکے چھٹی لینے کی بات کی تھی۔
اسی طرح دسمبر 2020 میں ’سائبرپنک 2077‘ کی ریلیز سے پہلے بھی مداح کئی دن کی چھٹیاں لینے اور مسلسل گیم کھیلنے کے منصوبے بناتے دکھائی دیے۔
کچھ اداروں نے بھی ’جی ٹی اے 6‘ کی مقبولیت کو دیکھتے ہوئے گیم کے مداحوں کے لیے خصوصی مراعات کا اعلان کیا ہے، جن میں امریکی ریاست جارجیا کے فورٹ سٹیورٹ میں تعینات امریکی فوج کی تھرڈ انفنٹری ڈویژن بھی شامل ہے۔
ڈویژن نے اپنے بعض اہلکاروں کو پیشکش کی کہ اگر وہ یکم اگست سے 12 نومبر کے درمیان کم از کم دو سال کے لیے دوبارہ فوج میں ملازمت کا معاہدہ کریں تو انھیں گیم کی ریلیز کے موقع پر چار دن کی چھٹی دی جائے گی۔
ڈویژن کی ترجمان لیفٹیننٹ کرنل اینجل ٹومکو نے اگست میں بی بی سی کو بتایا تھا کہ اس پیشکش کا مقصد فوجیوں کی دلچسپی کو مدنظر رکھتے ہوئے ایک منفرد مراعاتی پروگرام متعارف کرانا تھا۔
ان کے مطابق متعلقہ بٹالین کے 130 فوجی اس پیشکش کے اہل تھے، جن میں سے 20 نے دوبارہ ملازمت کے معاہدے کے ساتھ ’جی ٹی اے 6‘ کھیلنے کے لیے چار دن کی چھٹی لینے کی سہولت منتخب کی۔
نیوزی لینڈ منتقل ہونے کا طریقہ
ویڈیو گیم کے شوقین افراد نے ’جی ٹی اے 6‘ کی باضابطہ ریلیز سے پہلے گیم کھیلنے کے لیے ایک تکنیکی طریقہ تلاش کر لیا ہے، جس کے ذریعے وہ اپنے گیمنگ کنسولز کو یہ باور کروا سکتے ہیں کہ وہ نیوزی لینڈ میں موجود ہیں۔
رپورٹس کے مطابق ’جی ٹی اے 6‘ دنیا کے مختلف خطوں میں مقامی وقت کے مطابق رات 12 بجے ریلیز ہوگی۔ نیوزی لینڈ بین الاقوامی تاریخ کی لکیر (انٹرنیشنل ڈیٹ لائن) کے قریب واقع ہونے کے باعث ان اولین ممالک میں شامل ہوگا جہاں یہ گیم دستیاب ہوگی۔
اسی وجہ سے دنیا بھر کے گیمرز اپنے کنسولز کی لوکیشن تبدیل کرکے نیوزی لینڈ کے صارفین کی طرح گیم تک جلد رسائی حاصل کرنے کے طریقے تلاش کر رہے ہیں۔
گوگل ٹرینڈز کے مطابق برطانیہ میں گذشتہ ایک ماہ کے دوران ’’جی ٹی اے 6 نیوزی لینڈ ٹرک‘ سے متعلق سرچز میں پانچ ہزار فیصد سے زائد اضافہ ریکارڈ کیا گیا ہے۔
گیمز کی صحافی وِک ہُڈ کا کہنا ہے کہ ’بہت سے گیمرز جلد از جلد گیم کھیلنے کے خواہش مند ہوں گے تاکہ وہ دوسروں سے پہلے اس کا تجربہ حاصل کر سکیں اور گیم کی کہانی سے متعلق ممکنہ سپائلرز سے بھی محفوظ رہ سکیں۔‘
ان کے مطابق نیوزی لینڈ برطانیہ سے 13 گھنٹے آگے ہے، جس کا مطلب ہے کہ برطانوی گیمرز کے پاس گیم کی ریلیز کے بعد 13 گھنٹے اضافی کھیلنے کا موقع ہو سکتا ہے، اس دوران گیم کی کہانی کے اہم راز اور دلچسپ تفصیلات انٹرنیٹ پر پھیلنے کا خدشہ بھی موجود ہے۔
ایک مداح نے سوشل میڈیا پلیٹ فارم ریڈٹ پر اس طریقے کے بارے میں تبصرہ کرتے ہوئے کہا کہ جب انھیں اس کا علم ہوا تو انھوں نے ایک اضافی دن کی چھٹی بھی لے لی۔
اس طریقے کا ایک اور فائدہ یہ ہے کہ برطانوی وقت کے مطابق رات بھر جاگنے کے بجائے گیمرز صبح 11 بجے ہی گیم کھیل سکیں گے۔
تاہم ایکس باکس کے ایک صارف نے ریڈٹ پر اس طریقے کے ممکنہ خطرات سے خبردار کرتے ہوئے کہا کہ اگرچہ ماضی میں دیگر گیمز کی ریلیز کے موقع پر اس طریقے سے فائدہ اٹھایا جاتا رہا ہے، لیکن ممکن ہے کہ مائیکروسافٹ ’جی ٹی اے 6‘ کے معاملے میں اس مبینہ خامی کے خلاف کارروائی کرے، جس میں صارفین کے اکاؤنٹس پر پابندی بھی شامل ہو سکتی ہے۔
ابھی تک ’جی ٹی اے 6‘ کے ڈویلپرز راک سٹار گیمز، پبلشر ٹیک ٹو انٹرایکٹو، سونی اور مائیکروسافٹ کی جانب سے اس طریقے کے حوالے سے کوئی باضابطہ بیان سامنے نہیں آیا۔
ایمپائر اینالیسس سے وابستہ لوئس وولڈریج کے مطابق مائیکروسافٹ اور سونی، دونوں کمپنیوں کے ایکس باکس اور پلے سٹیشن پلیٹ فارمز پر علاقائی اکاؤنٹس اور خریداری سے متعلق مخصوص قواعد و ضوابط موجود ہیں، تاہم سونی کی پالیسیاں نسبتاً زیادہ سخت ہیں۔
ان کا کہنا ہے کہ اس طریقے کو استعمال کرنے والے کھلاڑی کمپنیوں کی شرائط و ضوابط کی خلاف ورزی کے مرتکب ہو سکتے ہیں، جس کے نتیجے میں ان کے اکاؤنٹس پر پابندیاں عائد ہونے یا ڈیجیٹل طور پر خریدی گئی گیمز تک رسائی کھونے کا خدشہ بھی موجود ہے۔
کچھ گیمرز کو ’جی ٹی اے 6‘ کھیلنے کے لیے کنسول صارفین کے مقابلے میں مزید طویل انتظار کرنا پڑے گا، کیونکہ ماضی کی جی ٹی اے گیمز کی طرح اس بار بھی گیم ابتدائی ریلیز کے موقع پر پی سی کے لیے دستیاب نہیں ہوگی۔
گیمر سٹیفن فن نے بی بی سی سے گفتگو کرتے ہوئے کہا کہ ’پی سی صارفین کے لیے بھی افسوس ہونا چاہیے، جنھیں فی الحال دوسروں کی یوٹیوب ویڈیوز اور ٹوئچ لائیو سٹریمز دیکھ کر ہی گیم کا تجربہ کرنا پڑے گا۔‘
دوسری جانب ’پلے سٹیشن 5‘ کی قیمت 520 برطانوی پاؤنڈ سے شروع ہوتی ہے، جبکہ زیادہ طاقتور ’پلے سٹیشن 5‘ پرو کی قیمت تقریباً 790 پاؤنڈ تک پہنچ جاتی ہے۔
اتنی بھاری قیمت کے باوجود بعض گیمرز خود کو اس گیم سے دور نہیں رکھ پا رہے، جسے کئی افراد رواں سال بلکہ گذشتہ ایک دہائی کی سب سے بڑی ویڈیو گیم ریلیز قرار دے رہے ہیں۔
ریڈٹ پر شیئر کی گئی پوسٹس اور ویڈیوز، جن میں امریکی یوٹیوبر آئسوش کی ویڈیو بھی شامل ہے، میں دیکھا جا سکتا ہے کہ مداح صرف ’جی ٹی اے 6‘ کھیلنے کے لیے نئے گیمنگ کنسولز خرید رہے ہیں۔
یوٹیوبر نے اپنی ویڈیو میں کنسول خریدنے کی وجہ بتاتے ہوئے کہا کہ ’اس کنسول پر میرے پاس صرف ایک ہی گیم ہوگی اور وہ جی ٹی اے ہے۔ پورا ’پی ایس 5‘ صرف ’جی ٹی اے 6‘ کے لیے ہے۔‘
ایک دہائی سے زائد عرصے کے انتظار کے بعد ایسا محسوس ہوتا ہے کہ جی ٹی اے کے بعض مداح اس گیم کی ریلیز کا ایک لمحہ بھی ضائع نہیں کرنا چاہتے، چاہے اس کے لیے انہیں کتنی ہی رقم کیوں نہ خرچ کرنی پڑے یا اضافی کوشش کیوں نہ کرنا پڑے۔
