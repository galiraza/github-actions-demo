# کالعدم بلوچ لبریشن آرمی کی قیادت میں تبدیلی کے پس منظر میں کیا عوامل کارفرما ہو سکتے ہیں؟

کالعدم بلوچ لبریشن آرمی کی قیادت میں تبدیلی کے پس منظر میں کیا عوامل کارفرما ہو سکتے ہیں؟

رواں ماہ کے آغاز میں کالعدم عسکریت پسند تنظیم بلوچ لبریشن آرمی (بی ایل اے) نے اعلان کیا کہ بشیر زیب اب تنظیم کے سربراہ نہیں رہے، جبکہ چند روز بعد رحمان گل عرف مرید کو تنظیم کا نیا ’کمانڈر اِن چیف‘ مقرر کرنے کا اعلان کیا گیا۔
بی ایل اے نے گذشتہ دو دہائیوں کے دوران پاکستان میں متعدد حملوں کی ذمہ داری قبول کی ہے۔ پاکستان، برطانیہ اور امریکہ سمیت کئی ممالک اسے دہشت گرد تنظیم قرار دے چکے ہیں۔
بی ایل اے کی قیادت میں یہ تبدیلی ایک ایسے وقت میں سامنے آئی ہے جب شدت پسند تنظیم حالیہ برسوں میں متعدد بڑے حملوں کی ذمہ داری قبول کر چکی ہے اور بلوچستان میں شورش ایک مرتبہ پھر قومی سطح پر توجہ کا مرکز بنی ہوئی ہے۔
کالعدم تنظیم کے ترجمان اس تبدیلی کو ایک معمول کی داخلی منتقلی قرار دیتے ہیں تاہم بعض سکیورٹی تجزیہ کاروں کا خیال ہے کہ اس کے پس منظر میں تنظیمی، عسکری یا دیگر عوامل بھی ہو سکتے ہیں۔
بی بی سی اردو نے سکیورٹی امور کے ماہرین، تجزیہ کاروں اور تنظیم کی سرگرمیوں سے واقف افراد سے بات کر کے یہ جاننے کی کوشش کی کہ اس تبدیلی کے بارے میں مختلف آرا کیا ہیں اور کیا اس کے نتیجے میں تنظیم کی حکمت عملی پر کوئی اثر پڑ سکتا ہے۔
بشیر زیب کی جگہ رحمان گل کیوں لائے گئے؟
رحمان گل عرف مرید بلوچ، بلوچ لبریشن آرمی (بی ایل اے) کے چھٹے سربراہ بنے ہیں۔
1970 کی دہائی میں قائم ہونے والی یہ تنظیم فوجی آمر جنرل ضیاالحق کی جانب سے اقتدار پر قبضے اور بلوچ قوم پرست رہنماؤں سے مذاکرات کے نتیجے میں پس منظر میں چلی گئی تھی مگر پھر سابق صدر پرویز مشرف کے دور اقتدار میں پیش آئے واقعات کے بعد اس تنظیم کی جانب سے عسکریت پسندی پر مبنی نمایاں کارروائیوں کا سلسلہ شروع ہوا۔
نواب خیر بخش مری اس تنظیم کی بنیاد رکھنے والوں میں شامل تھے، جن کے بعد اس کالعدم تنظیم کی قیادت اُن کے بیٹے بالاچ مری کے پاس آئی۔ سنہ 2007 میں بالاچ مری کی ہلاکت کے بعد بی ایل اے کی قیادت اُن کے چھوٹے بھائی حربیار مری کو سونپی گئی۔

مگر جلد ہی تنظیم میں اندرونی اختلافات کی وجہ سے دو دھڑے قائم ہو گئے اور مرکزی دھڑے کی قیادت اسلم عرف استاد اسلم بلوچ کے پاس آئی۔ سنہ 2018 میں ایک حملے میں اسلم بلوچ کی ہلاکت کے بعد بلوچ اسٹوڈنٹس آرگنائزیشن کے سابق چیئرمین بشیر زیب اِس کالعدم تنظیم کے سربراہ بنے اور لگ بھگ آٹھ سال تک اس عہدے پر رہے۔
اب تنظیم کی نئی قیادت کے اعلان کے بعد یہ سوال سامنے آیا ہے کہ تقریباً آٹھ برس تک سربراہ رہنے والے بشیر زیب کی جگہ رحمان گل کو کیوں لایا گیا۔ تاحال بی ایل اے کی جانب سے بشیر زیب کی تبدیلی کی تفصیلی وجہ بیان نہیں کی گئی اور نہ ہی آزاد ذرائع سے اس بارے میں معلومات کی تصدیق ہو سکی ہے۔
پاکستان انسٹیٹیوٹ آف پیس سٹڈیز کے سابق ڈائریکٹر اور سکیورٹی تجزیہ کار عامر رانا کے مطابق شورش میں ملوث گروہوں کو ایک جانب ریاستی دباؤ کا سامنا ہوتا ہے جبکہ دوسری جانب انہیں اپنی اندرونی ساخت اور ہم آہنگی برقرار رکھنا پڑتی ہے۔
ان کے بقول یہ ممکن ہے کہ قیادت کی منتقلی باہمی مشاورت سے ہوئی ہو، تاہم یہ بھی ممکن ہے کہ اس کے پس منظر میں تنظیم کے اندر مختلف آرا موجود رہی ہوں۔ ان کا کہنا ہے کہ فی الحال ایسی کوئی مصدقہ اطلاع سامنے نہیں آئی جس سے یہ ظاہر ہو کہ اس فیصلے کے نتیجے میں تنظیم کسی بڑے داخلی بحران یا نئی دھڑے بندی کا شکار ہوئی ہے۔
بلوچستان کے امور پر تحقیق کرنے والے محمد عادل بلوچ (نام تبدیل کیا گیا ہے) کہتے ہیں کہ بشیر زیب کے دور میں تنظیم نے اپنی سرگرمیوں اور حکمت عملی کو ایک مخصوص سمت دینے کی کوشش کی۔
تاہم ان کے مطابق قیادت کی حالیہ تبدیلی اس بات کی نشاندہی کر سکتی ہے کہ تنظیم اپنے طریقۂ کار یا حکمت عملی میں تبدیلی لانا چاہتی ہو، اگرچہ اس بارے میں کوئی باضابطہ اعلان سامنے نہیں آیا۔
بلوچ طلبہ سیاست سے وابستہ رہنے والے پردھان بلوچ کا کہنا ہے کہ بشیر زیب تنظیم سازی اور ساتھیوں سے مشاورت کے حوالے سے جانے جاتے تھے۔ ان کے مطابق ’اگر قیادت کی تبدیلی کے پس منظر میں کوئی بڑا تنازع یا زبردستی شامل ہوتی تو اس کے آثار اب تک سامنے آ سکتے تھے۔‘
پردھان بلوچ کے خیال میں اسلم بلوچ کی ہلاکت کے بعد ممکنہ طور پر تنظیم کے اندر یہ احساس پیدا ہوا ہو گا کہ تمام اختیارات کسی ایک شخصیت تک محدود نہ رہیں۔
تاہم اس بارے میں تنظیم نے عوامی سطح پر کوئی وضاحت نہیں دی اور بی ایل اے کے ترجمان جیئند بلوچ نے بی بی سی کے سوالات کے جواب میں کہا کہ قیادت میں تبدیلی کو تنظیم کے اندر جاری ایک ادارہ جاتی عمل کے تناظر میں دیکھا جانا چاہیے۔
ترجمان نے یہ بھی نہیں بتایا کہ بشیر زیب اس وقت کس حیثیت میں کام کر رہے ہیں یا تنظیم میں ان کا مستقبل کا کردار کیا ہو گا۔
کیا ہیروف ٹو سے متعلق اختلافات نے جنم لیا؟
بشیر زیب کے دور میں کالعدم بلوچ لبریشن آرمی نے کئی بڑے حملوں کی ذمہ داری قبول کی، جن میں کراچی میں چینی قونصل خانے، گوادر کے پرل کانٹی نینٹل ہوٹل، کراچی سٹاک ایکسچینج اور کراچی یونیورسٹی میں چینی اساتذہ کو لے جانے والی گاڑی پر حملے شامل ہیں اور ان حملوں میں درجنوں افراد ہلاک اور زخمی ہوئے تھے۔
ان کارروائیوں میں 2024 میں ’ہیروف‘ اور جنوری 2026 میں ’ہیروف ٹو‘ کے نام سے کارروائیاں بھی شامل تھیں جن کے بارے میں تنظیم کا کہنا تھا کہ یہ بلوچستان کے مختلف علاقوں میں کئی روز تک جاری رہیں۔
ہیروف ٹو کے دوران بی ایل اے کو جو جانی نقصان اٹھانا پڑا اس کے بارے میں متضاد دعوے موجود ہیں۔ بی ایل اے نے اپنے جنگجوؤں کی ہلاکتوں کی تعداد 93 بتائی جبکہ پاکستانی حکام کا دعویٰ تھا کہ مرنے والے شدت پسندوں کی تعداد 200 سے زیادہ تھی تاہم ان دعوؤں کی آزادانہ تصدیق ممکن نہیں ہو سکی۔
تجزیہ کاروں کا خیال ہے کہ بی ایل اے کی قیادت کی تبدیلی کو ہیروف ٹو کے نتائج اور اس کے بعد پیدا ہونے والی صورت حال کے تناظر میں بھی دیکھا جا سکتا ہے۔
بلوچستان حکومت کے سابق ترجمان اور تجزیہ کار جان اچکزئی کہتے ہیں کہ ہیروف ٹو کے بعد تنظیم کو نمایاں جانی نقصان اٹھانا پڑا، جس کے اثرات داخلی سطح پر بھی محسوس کیے گئے ہوں گے۔
ان کے مطابق ’بہت سارے دیگر معاملات بھی تھے‘ جو اُن کے بقول بشیر زیب کو ہٹانے اور تنظیم کے اندر اختلافات کا باعث بنے۔ تاہم اس بارے میں غیر جانبدار ذرائع سے تصدیق شدہ معلومات دستیاب نہیں۔
دوسری جانب سکیورٹی تجزیہ کار عامر رانا کہتے ہیں کہ بعض حلقوں میں یہ بحث ضرور رہی کہ ہیروف ٹو کے دوران ہونے والے جانی نقصان اور حکمت عملی کے بعض پہلوؤں کا ازسرنو جائزہ لیا جانا چاہیے، تاہم تنظیم کے اندرونی مباحث یا اختلافات کے بارے میں مصدقہ معلومات محدود ہیں۔
سنگاپور میں مقیم دہشت گردی اور سکیورٹی امور کے محقق عبدالباسط کے مطابق ایک ہی کارروائی کے بعد بڑی تعداد میں جنگجوؤں کی ہلاکت کسی بھی مسلح گروہ کے لیے دھچکے کا باعث بن سکتی ہے۔ ان کی رائے میں بشیر زیب کی صحت، تنظیمی حکمت عملی اور عوامی سطح پر ان کی بڑھتی ہوئی نمایاں موجودگی سمیت مختلف عوامل قیادت کی تبدیلی میں کردار ادا کر سکتے ہیں۔
عبدالباسط کا کہنا ہے کہ جدید نگرانی اور انٹیلیجنس صلاحیتوں کے دور میں بڑی تعداد میں جنگجوؤں کا ایک ہی علاقے میں متحرک ہونا سکیورٹی خطرات بڑھا سکتا ہے، جس سے بعد ازاں انسدادِ عسکریت پسندی کی کارروائیوں میں نقصان اٹھانے کا امکان پیدا ہوتا ہے۔
ادھر بی ایل اے کے ترجمان جیئند بلوچ ان تجزیوں سے اتفاق نہیں کرتے۔ ان کا کہنا ہے کہ تنظیم کی نظر میں ہیروف ٹو اپنے مقاصد کے حصول کے اعتبار سے ایک اہم کارروائی تھی اور قیادت کی تبدیلی کو اس کے نتائج سے جوڑنا درست نہیں۔
کیا کالعدم تنظیم کی قیادت کی تبدیلی سے حکمتِ عملی میں تبدیلی آئے گی؟
بی ایل اے کے قیام کے بعد سے اس تنظیم کے طریقۂ کار اور کارروائیوں کی نوعیت وقت کے ساتھ تبدیل ہوتی رہی ہے۔ تاہم یہ ابھی واضح نہیں کہ رحمان گل کی قیادت میں تنظیم کی مجموعی حکمتِ عملی میں کوئی بنیادی تبدیلی آئے گی یا نہیں۔
سکیورٹی تجزیہ کار عامر رانا کے مطابق اگر آئندہ بھی تنظیم ماضی جیسی مربوط کارروائیاں جاری رکھتی ہے تو یہ اس بات کا اشارہ ہو سکتا ہے کہ موجودہ قیادت سابقہ حکمتِ عملی کے تسلسل کو برقرار رکھنا چاہتی ہے۔
ان کے مطابق رحمان گل کا ماضی یا پیشہ ورانہ پس منظر اپنے طور پر اس بات کا قابلِ اعتماد اشارہ نہیں کہ تنظیم کی سمت میں کوئی بڑی تبدیلی آئے گی، کیونکہ دنیا کے مختلف شورش زدہ علاقوں میں مسلح گروہوں کی قیادت مختلف پس منظر کے افراد کے ہاتھ میں جاتی رہی ہے۔
’دنیا میں عسکریت پسندوں کی تحریکوں میں ملٹری بیک گراؤنڈ سے لوگ آتے رہے ہیں اور طالبان کے اندر بھی افغان فورسز کو چھوڑ کر لوگ جاتے رہے ہیں۔ سابقہ مشرقی پاکستان میں بھی بہت سارے لوگ فوج سے مکتی باہنی میں گئے، تو یہ کوئی اچنبھے کی بات نہیں ہے۔‘
پردھان بلوچ کے مطابق رحمان گل کے ماضی کے تجربات کو دیکھتے ہوئے کارروائیوں کی نوعیت میں کچھ تبدیلی آ سکتی ہے، تاہم یہ کہنا قبل از وقت ہوگا کہ یہ تبدیلی کس حد تک نمایاں ہوگی۔
اسی طرح سکیورٹی امور کے محقق عبدالباسط کا کہنا ہے کہ قیادت کی تبدیلی کے بعد تنظیم اپنے طریقۂ کار کا ازسرِنو جائزہ لے سکتی ہے، خاص طور پر ایسے وقت میں جب حالیہ برسوں میں سکیورٹی فورسز اور عسکریت پسند گروہوں دونوں کو نمایاں نقصانات کا سامنا رہا ہے۔
اُن کے مطابق ’بجائے اس کے کہ ایک بہت بڑی تعداد اور بہت بڑے وسائل ایک اور ہیروف میں جھونک دیا جائے اور نقصان اٹھایا جائے، اس کے برعکس چھوٹے چھوٹے آپریشن ہو سکتے ہیں اور یہ ہارڈ اہداف کی طرف واپس جا سکتے ہیں۔‘
حکمتِ عملی میں تبدیلی کا ایک مظہر ڈرونز کا استعمال بھی دکھائی دیتا ہے۔ بشیر زیب کی قیادت کے آخری دور میں بی ایل اے کی جانب سے ڈرون ٹیکنالوجی کے استعمال سے متعلق دعوے سامنے آئے تھے۔
جنوری 2026 میں تنظیم نے اپنی کارروائیوں میں ڈرونز کے استعمال کا دعویٰ کیا جبکہ پاکستانی فوج کے شعبۂ تعلقاتِ عامہ نے نومبر 2026 میں شدت پسندوں کی جانب سے کواڈ کاپٹرز کے ذریعے سکیورٹی اہلکاروں کو نشانہ بنانے کی تصدیق کی۔
سکیورٹی تجزیہ کار عبدالباسط کہتے ہیں کہ ’ڈرونز سستے ہیں، آسانی سے دستیاب ہیں، حکمتِ عملی کے لحاظ سے مؤثر ہیں، اگرچہ یہ گیم چینجر نہیں ہیں لیکن کافی مؤثر ہیں اور نفسیاتی جنگ میں ان کا کافی زیادہ اثر ہے۔ وقت کے ساتھ اِن کا استعمال بڑھتا جائے گا، اس کا کسی تنظیم میں قیادت کی تبدیلی سے زیادہ تعلق ٹیکنالوجی کے اپنے پھیلاؤ کے ساتھ ہے۔‘
نئی قیادت کے تحت تنظیم کی سرگرمیوں کی نوعیت میں کوئی نمایاں تبدیلی آتی ہے یا نہیں، اس کا اندازہ آنے والے مہینوں میں زیادہ واضح ہو سکے گا تاہم مبصرین اور تجزیہ کار اس بات پر متفق دکھائی دیتے ہیں کہ بشیر زیب کی جگہ رحمان گل کی تقرری کسی ایک مخصوص وجہ کا نتیجہ نہیں اور جہاں تنظیم اسے ایک معمول کی داخلی منتقلی قرار دیتی ہے وہیں بعض سکیورٹی ماہرین حالیہ کارروائیوں کے نتائج، ممکنہ تنظیمی مباحث اور حکمت عملی میں تبدیلی جیسے عوامل کی نشاندہی کرتے نظر آتے ہیں۔
’BBC News اردو‘ ایڈیشن کا انتخاب کریں اور نیوز الرٹس، تحقیقاتی اور تفصیلی رپورٹس، سب کچھ ایک ہی جگہ پر حاصل کریں۔
