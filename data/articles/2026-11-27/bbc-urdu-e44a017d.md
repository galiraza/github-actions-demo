25 نومبر، 2026 کو خلائی سائنس کے شعبے میں پاکستان کے لیے ایک اہم دن قرار دیا جا رہا ہے۔
بدھ کے روز جب چین کے جیوچوان سیٹلائٹ لانچ سینٹر سے ’شینژو 24‘ خلائی مشن چین کے خلائی سٹیشن کے لیے روانہ ہوا تو اس پر چینی خلابازوں کے ساتھ ساتھ ایک پاکستانی خلا باز محمد ذیشان علی بھی موجود تھے۔
اسی روز روسی وزیرِ توانائی نے ایک بیان میں کہا کہ ان کا ملک پاکستان کے ساتھ خلائی ٹیکنالوجی شیئر کرنے کے لیے تیار ہے۔
روس کی تعاون کی پیشکش
روس پاکستان کے ساتھ سیٹلائٹ نیویگیشن اور مواصلات، ریموٹ سینسنگ، انسانی خلائی پروازوں اور خلائی ٹیکنالوجی کے شعبوں میں اپنی مہارت اور تجربہ شیئر کرنے کے لیے تیار ہے۔
یہ بات روس کے وزیرِ توانائی سرگئی تسیویلیوف نے بدھ کے روز ماسکو میں پاکستان اور روس کے درمیان تجارتی، اقتصادی، سائنسی اور تکنیکی تعاون سے متعلق بین الحکومتی کمیشن کے 11ویں اجلاس سے خطاب کرتے ہوئے کہی۔
سرگئی تسیویلیوف کا کہنا تھا کہ ’روس ایک بڑی خلائی طاقت ہے۔ ہمارے ملک کے پاس منفرد تجربہ، مضبوط سائنسی و تکنیکی بنیاد اور خلائی تحقیق میں وسیع مہارت موجود ہے۔‘
یاد رہے کہ چین اور امریکہ کے علاوہ روس وہ ملک ہے جو انسانوں کو خلا بھیج چکا ہے۔
سنہ 1961 میں جب روس (جو اس وقت سوویت یونین تھا) کے ووسٹوک-1 پر سوار روسی خلاباز یوری گگارن خلا میں جانے والے پہلے انسان بن گئے تھے۔
End of سب سے زیادہ پڑھی جانے والی
ان کا مزید کہنا تھا کہ ’ہم سیٹلائٹ نیویگیشن اور مواصلات، زمین کے مشاہدے (ریموٹ سینسنگ)، انسانی خلائی پروازوں اور خلائی ٹیکنالوجی کے شعبوں میں اپنی پیش رفت اور مہارت پاکستان کے ساتھ شیئر کرنے کے لیے تیار ہیں۔‘
وہ تمام خِبریں جو آپ جاننا چاہتے ہیں بی بی سی ورلڈ سروس ایپ پر موجود ہیں
ایپ ڈاؤن لوڈ کریں
بی بی سی اُردو موبائل ایپ نهاية
دوسری جانب پاکستان کی وزارت برائے معاشی امور کی جانب سے جاری بیان کے مطابق ماسکو میں ہونے والی بین الحکومتی کمیشن کے اجلاس کے دوران معاشی تعاون کے پروٹوکول پر بھی دستخط ہوئے۔ اس پروٹوکول کے تحت تجارت، سرمایہ کاری، توانائی، صنعت، زراعت، دواسازی، ٹرانسپورٹ رابطوں، آئی سی ٹی، تعلیم، ثقافت، سائنس اور خلائی شعبوں میں تعاون کو فروغ دینے کی بات کی گئی ہے۔
روسی وزیر کا بیان پاکستان کے وزیرِ اعظم شہباز شریف کے متوقع دورہ ماسکو سے محض چند دن قبل آیا ہے۔
اس سے قبل رواں ہفتے روسی خبر رساں ادارے سپوتنک کو دیے گئے ایک انٹرویو میں روس میں پاکستان کے سفیر فیصل نیاز ترمذی کا کہنا تھا کہ وزیرِ اعظم شہباز شریف کے متوقع دورۂ روس کا مقصد دونوں ممالک کے درمیان اقتصادی، تجارتی، ثقافتی، تعلیمی اور عوامی روابط کو فروغ دینا ہے، جبکہ اس موقع پر تقریباً 15 سے 16 معاہدوں پر دستخط کیے جانے کی توقع ہے۔
فیصل ترمذی نے کہا کہ اس دورے کا مقصد محض سفارتی روابط تک محدود رہنے کے بجائے ٹھوس اور طویل مدتی اقتصادی نتائج حاصل کرنا ہے۔
ان کا مزید کہنا تھا کہ روس کے ساتھ پاکستان کی شمولیت اس کی وسیع تر خارجہ پالیسی کا حصہ ہے اور اس بات پر زور دیا کہ سیاسی تعاون سے آگے بڑھ کر تعلقات کو مضبوط بنانا ضروری ہے۔
روس پاکستان کو خلائی ٹیکنالوجی کے شعبے میں تعاون کی پیشکش کیوں کر رہا ہے؟
بی بی سی اردو نے تجزیہ کاروں سے بات کر کے یہ سمجھنے کی کوشش کی کہ روس کی جانب سے پاکستان کے ساتھ خلائی ٹیکنالوجی کے شعبے میں تعاون کی پیشکش کی کیا اہمیت ہے اور اس سے پاکستان کو کیا فائدہ پہنچ سکتا ہے۔
دفاعی تجریہ کار سید محمد علی کہتے ہیں کہ دنیا کے خلائی پروگرام کی تاریخ کا جائزہ لیں تو روس اور امریکہ سب سے آگے ہیں تاہم اب چین اور یورپی یونین بھی ان تک پہنچنے کی کوشش کر رہے ہیں۔
ان کے مطابق اب بھی وسائل، تجربے، خلائی اثاثوں اور ٹیکنالوجی کے اعتبار سے روس اور امریکہ کا اثرورسوخ چین اور یورپی یونین کے مقابلے میں کہیں زیادہ ہے۔
وہ کہتے ہیں چین اس شعبے میں ترقی اور اپنے وسائل میں اضافہ کر کے اس فاصلے کو کم کرنے کی کوشش کر رہا ہے۔
سید محمد علی کے مطابق پاکستان کا سپیس ٹیکنالوجی کے حصول کے لیے زیادہ تر چین پر انحصار رہا ہے۔ ’اب اگر روس کے ساتھ بھی اس شعبے میں تعاون بڑھتا ہے تو ٹیکنالوجی کے حصول، مشترکہ منصوبے، سیتلائٹ لنچز، ان کی ڈیزائننگ، نیوی گیشن، ڈیٹا شیئرنگ، سائنسدانوں اور انجینیئرز کی ٹریننگ اور ٹیکنالوجی کی منتقلی کی بہت سی راہیں کھل سکتی ہیں۔‘
ان کا مزید کہنا تھا کہ اس سے یہ بھی ثابت ہوتا ہے کہ روس پاکستان کو اس خطے اور اسلامی دنیا کی ایک اہم ترین ریاست سمجھتا ہے جس کے ساتھ وہ اب ٹیکنالوجی اور سائنس کے شعبوں میں بھی تعاون کو فروغ دینا اپنے مفادات کے مترادف سمجھتا ہے۔
ڈاکٹر عادل سلطان ایئر یونیورسٹی میں بطور ڈین فیکلٹی آف ایرو سپیس اینڈ سٹریٹیجک سٹڈیز وابستہ ہیں۔
روس کی جانب پاکستان کو خلائی ٹیکنالوجی میں مدد کی پیشکش کے متعلق بات کرتے ہوئے ڈاکٹر عادل سلطان کا کہنا تھا کہ وہ اس کی تفصیلات سے تو آگاہ نہیں تاہم عمومی طور پر خلائی ٹیکنالوجی کے شعبے میں تعاون کی حوصلہ افزائی کی جاتی ہے۔
ان کے مطابق اگر امریکہ کا انٹرنیشنل سپیس سٹیشن دیکھیں تو اس میں بھی روس سمیت کئی ممالک شامل ہیں۔
ان کا کہنا ہے کہ اسی طرح اگر پاکستان کی چین کے ساتھ شراکت داری ہے بھی تو اس میں کوئی برائی نہیں کیونکہ یہ پر امن مقاصد کے لیے ہوتا ہے۔
ان کا کہنا ہے کیونکہ امریکہ اور دیگر مغربی ممالک نے چین کو شامل نہیں کیا ہے تو مجبورا چین نے اپنا خود کا خلائی لروگر بنایا جس میں اس نے پاکستان کو بھی شامل کیا ہے۔
ڈاکٹر عادل سلطان کے مطابق اس سے پاکستان نے فائدہ اٹھایا اور ماضی میں سیٹلائٹس لانچ کیے ہیں۔
’اگر ایسا ہی ارینجمنٹ روس کے ساتھ ہوتا ہے تو اس میں کوئی مضائقہ نہیں۔‘
وہ کہتے ہیں کہ ’اگر روس کے ساتھ کام کرنے کا موقع ملتا ہے تو اس سے ہماری سیٹلائٹ بلڈنگ کپیسٹی میں اضافہ ہو گا اور دونوں ممالک کے لیے خلائی ٹیکنالوجی کے شعبے میں تعاون کے لیے ایک اچھا موقع ہوگا۔‘
روس کی جانب سے مختلف قسم کے سیٹلائٹ ٹیکنالوجی میں تعاون کے مطابق بات کرتے ہوئے ان کا کہنا تھا کہ ریموٹ سینسنگ ماحولیاتی تبدیلیوں کا بروقت پتا لگانے اور زراعت کے شعبے میں مددگار ثابت ہو سکتی ہے، کہ کون سی فصل زیادہ ہوئی ہے اور کونسی لگائی جائے۔
ان کے مطابق ’اس سے آپ کو سپیس سے آپ کو ایک بڑا برڈ آئی ویو مل جاتا ہے اور کافی ڈیٹیل میں مل جاتا ہے کہ آپ منصوبہ بندی کر سکیں۔‘
کمیونیکیشن سیٹلائٹ کے حوالے سے ان کا کہنا تھا کہ یہ ٹی وی چینلز کی نشریات اور آپس کی موبائل کمیونیکیشن کے لیے استعمال ہوتے ہیں۔ اس کے علاوہ نیویگیشن راستوں کی نشاندہی جیسے میپس وغیرہ کے لیے استعمال ہوتے ہیں۔
ڈاکٹر عادل سلطان کے مطابق اس نوعیت کے تعاون میں کسی بھی ملک کی سائنسی شراکت کے ساتھ ساتھ تحقیق، تجربات اور تکنیکی عمل سے حاصل ہونے والا علم اور مہارت بھی آگے منتقل ہوتی ہے۔
اس کے علاوہ اس تعاون کا ایک سیاسی پس منظر بھی ہوتا ہے، روس بھی اپنا اثرورسوخ کا ایک حلقہ بنانا چاہ رہا ہے۔
عادل سلطان کہتے ہیں کہ ’ہمارا سپیس پروگرام روس سے بہت پیچھے ہے، اور ہمیں زیادہ فائدہ ہوگا اگر ہم ان کے ساتھ کولیبریٹ کریں لیکن ان کو ایک طرح سے پولیٹیکل فائدہ بھی ہو سکتا ہے اور ان کا 'سفیئر آف انفلوئنس' بڑھے گا۔‘
’روس کے لیے پاکستان زیادہ اہم ہو گیا ہے‘
ڈاکٹر اشتیاق احمد کمیشن آن سائنس اینڈ ٹیکنالوجی فار سسٹینیبل ڈیولپمنٹ اِن دی ساؤتھ (کامسیٹس) کے ایگزیکٹو ڈائریکٹر ہیں۔
وہ کہتے ہیں کہ ’ہمیں یہ سمجھنے کی ضرورت ہے کہ روس ہم سے تعلقات بڑھانے میں کیوں دلچسپی رکھتا ہے۔‘ ان کا کہنا ہے کہ پچھلے تقریباً ایک دہائی میں پاکستان نے اپنی خارجہ پالیسی کو وسعت دینے کی کوشش کی ہے۔
ڈاکٹر اشتیاق کے مطابق پاکستان کے چین، سعودی عرب اور ترکی کے ساتھ تاریخی تعلقات ہیں، لیکن امریکہ کے ساتھ تعلقات سرد و گرم رہے ہیں لیکن اس کے باوجود پاکستان امریکہ کا اہم شراکت دار رہا اور صدر ٹرمپ کے حالیہ دور میں بھی اس میں اضافہ ہوا ہے۔ اسی طرح ہم نے روس کے ساتھ بھی دفاع اور توانائی کے شعبوں میں تعاون کو وسعت دینے کی کوشش کی ہے۔
ان کا کہنا ہے کہ افغانستان اور وسطی ایشیائی ریاستوں کے تناظر میں ہمارا روس کے ساتھ کوئی ٹکراؤ نہیں بلکہ ہمارے مفادات ہم آہنگ ہیں۔
ڈاکٹر اشتیاق کہتے ہیں کہ 90 کی دہائی سے اور بالخصوص سابق امریکی صدر اوباما کے دور سے انڈیا کو چین کے مقابلے کے لیے تیار کر رہے تھے۔
وہ دعویٰ کرتے ہیں کہ 90 کی دہائی کے اواخر سے امریکہ اور انڈیا کے درمیان جو سٹریٹیجک تعلقات قائم ہوئے، اس سے روس کافی خفا ہے۔ ’انھوں نے محسوس کیا کہ انڈیا امریکہ کے کیمپ میں چلا گیا ہے۔‘
ڈاکٹر اشتیاق کہتے ہیں پاکستان اور روس کے انسدادِ دہشت گردی اور خطے کی سلامتی کے حوالے سے خدشات ایک جیسے ہیں۔
ان کا کہنا ہے کہ ایک دہائی کے دوران پاکستان کی خارجہ پالیسی میں آنے والی وسعت کے علاوہ روس کی جنوبی ایشیا کے حوالے سے پالیسی بھی تبدیل ہوئی ہے۔ ’وہ انڈیا کے ساتھ سٹریٹیجک تعلقات رکھ رہے ہیں لیکن پاکستان زیادہ اہم ہو گیا ہے۔‘
ڈاکٹر اشتیاق بھی سید محمد علی کی اس بات سے متفق ہیں کہ پاکستان کی دفاعی اور سیٹلائٹ ٹیکنالوجی چین سے جڑی ہوئی ہے۔ تاہم وہ اس جانب اشارہ کرتے ہیں کہ روس بھی چین کا اتحادی ہے۔
وہ کہتے ہیں کہ روس پہلا ملک ہے جس نے خلا میں انسانی مشن بھیجا تھا اور کمیونزم کے مسائل سے قطع نظر روس کا خلائی پروگرام اتنا ہی قابل ہے جتنا امریکہ یا اب چین کا ہوتا جا رہا ہے۔
ان کا کہنا ہے کہ ’ایسے وقت میں جب پاکستانی خلاباز چینی مشن کے ساتھ خلا میں گیا ہے، روس کی جانب سے خلائی ٹیکنالوجی شیئر کرنے کا اعلان کیا ہے۔‘ ان کے مطابق اس کی ایک وجہ یہ ہے کہ ’روس کو لگتا ہے کہ اگر چین تعاون کر رہا ہے تو ہم بھی کریں۔‘
ڈاکٹر اشتیاق کے مطابق روس اور پاکستان کے ساتھ ہمارے تعلقات سے امریکی صدر ڈونلڈ کو کوئی اعتراض نہیں ہو گا کیونکہ ’اس وقت تینوں (چین، روس اور امریکہ کے) رہنما اکٹھے ہیں۔‘
روس پاکستان کے ساتھ شراکت داری سے انڈیا کو کیا پیغام دینا چاہتا ہے؟
اسلام آباد میں قائم صنوبر انسٹی ٹیوٹ کے ایگزیکٹیو ڈائریکٹر ڈاکٹر قمر چیمہ کہتے ہیں کہ روس کی جانب سے خلائی ٹیکنالوجی میں تعاون کی جو بات کی گئی ہے اس کے مختلف پہلو ہیں۔
’مثال کے طور پر انڈیا روسی ساختہ ففتھ جنریشن ایس یو-57 طیارے اور امریکی ایف-35 کی خریداری کے حوالے سے فیصلہ نہیں کر پا رہا۔‘
ان کے مطابق ان حالات میں ’روسی محسوس کر رہے ہیں کہ گو کہ انڈیا ہمارا بہت مضبوط پارٹنر ہے لیکن اس وقت انڈیا کو مغرب میں اتنی رسائی اور رعایتیں ہیں کہ انڈیا سمجھتا ہے کہ اسے مغرب کی جانب جانا چاہیے۔ لہذا یہ انڈیا کے لیے بھی ایک پیغام ہے۔‘
قمر چیمہ کہتے ہیں کہ اس کے علاوہ انڈیا نے روس سے سستے داموں تیل تو ضرور لیا لیکن جس طرح کی رسائی امریکہ کو انڈیا میں حاصل ہے وہ روس کو حاصل نہیں۔
'اس حوالے سے روس کے اندر یہ بات ضرور آئی ہوگی کہ ہمیں پاکستان کے ساتھ تعلقات کو سپیس اینڈ ٹیکنالوجی کے شعبے میں بھی تعاون کرنا چاہیے خاص طور پر جب پاکستان اور چین کے پہلے ہی شعبے میں کافی کام کر چکے ہیں۔'
تاہم وہ کہتے ہیں کہ اس پاکستان اور روس کے درمیان قربت کے باوجود وہ زیادہ قریب نہیں آ سکے اور اس کی وجہ وہ روس پر عائد پابندیوں کو مانتے ہیں۔
ڈاکٹر قمر چیمہ کہتے ہیں پاکستان افغانستان کے حوالے سے روس کا تعاون چاہتا ہے۔
ان کا کہنا ہے کہ پاکستان نے جب افغانستان پر حملہ کیا تو اس وقت روس نے بھی کہا تھا کہ افغانستان میں پانچ سے سات ہزار شدت پسند موجود ہیں۔
’پاکستان اور روس کا افغانستان کے حوالے سے سرحد پار خطرات کے بارے میں یکساں مؤقف ہے۔‘
ان کا کہنا ہے وزیرِ اعظم شہباز شریف کے دورہ روس کے دوران خطے کے ممالک کے درمیان تعاون کی اہمیت کو اجاگر کیا جائے گا۔
ڈاکٹر قمر چیمہ کے مطابق پاکستان روس کے ساتھ ایک وسیع تر پیمانے پر تعاون بھی چاہتا ہے۔
’ظاہر سی بات ہے کہ روس چاہتا ہے کہ پاکستان کے لیے مزید آپشنز کھولنے چاہیے تاکہ پاکستان کو مزید قریب بھی کیا جا سکے اور انڈیا کو بھی احساس دلایا جا سکے کہ اگر آپ ’ڈبل مائنڈڈ‘ ہیں تو ہم پاکستان کو ایک اہم شراکت دار کے طور پر دیکھتے ہیں۔‘
